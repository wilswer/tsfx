8/// Sum of `value(0), ..., value(n - 1)` exactly as numpy's `add.reduce` sums
9/// a contiguous `float64` array (numpy 2.x): pairwise summation with eight
10/// accumulators for blocks of up to 128 values, split in halves above that,
11/// added to an initial `0.0`. Results equal numpy's bit for bit, which a
12/// straight or Welford sum does not.
13pub fn numpy_sum(n: usize, value: impl Fn(usize) -> f64) -> f64 {
14    0.0 + pairwise_sum(0, n, &value)
15}
16
17fn pairwise_sum(start: usize, n: usize, value: &impl Fn(usize) -> f64) -> f64 {
18    if n < 8 {
19        (start..start + n).fold(0.0, |res, i| res + value(i))
20    } else if n <= 128 {
21        let mut r: [f64; 8] = std::array::from_fn(|j| value(start + j));
22        let mut i = 8;
23        while i < n - n % 8 {
24            for (j, acc) in r.iter_mut().enumerate() {
25                *acc += value(start + i + j);
26            }
27            i += 8;
28        }
29        let mut res = ((r[0] + r[1]) + (r[2] + r[3])) + ((r[4] + r[5]) + (r[6] + r[7]));
30        for k in i..n {
31            res += value(start + k);
32        }
33        res
34    } else {
35        let mut n2 = n / 2;
36        n2 -= n2 % 8;
37        pairwise_sum(start, n2, value) + pairwise_sum(start + n2, n - n2, value)
38    }
39}

41/// Arithmetic mean, exactly as `np.mean` (see [`numpy_sum`]). NaN for an
42/// empty series or if any value is NaN.
43pub fn numpy_mean(x: &[f64]) -> f64 {
44    numpy_sum(x.len(), |i| x[i]) / x.len() as f64
45}

47/// Population variance (`ddof = 0`) of a series.
48///
49/// Exactly `np.var` as used by tsfresh: the mean, then the mean of the
50/// squared deviations, both summed as numpy does (see [`numpy_sum`]).
51/// Returns the raw value; how a zero variance is handled is up to each
52/// feature.
53pub fn population_var(arr: &ArrayView1<f64>) -> f64 {
54    let n = arr.len();
55    let mean = numpy_sum(n, |i| arr[i]) / n as f64;
56    numpy_sum(n, |i| {
57        let d = arr[i] - mean;
58        d * d
59    }) / n as f64
60}

62/// Population standard deviation (`ddof = 0`) of a series: the square root
63/// of [`population_var`], exactly `np.std` as used by tsfresh. Returns the
64/// raw value; how a zero standard deviation is handled is up to each
65/// feature.
66pub fn population_std(arr: &ArrayView1<f64>) -> f64 {
67    population_var(arr).sqrt()
68}

70/// Pairwise leaf of pandas' moment sums: 128 SSE2 batches of two lanes.
71const PANDAS_MOMENTS_LEAF: usize = 256;
72
73/// The two lane sums and the non-NaN count of an even-length `x`, as pandas'
74/// `accumulate_mean_pairwise` (x86_64 build: SSE2, two `f64` lanes): within a
75/// leaf, lane `j` adds the values at positions `j, j + 2, ...`; longer slices
76/// are split near the middle and their lane sums added. A NaN adds `+0.0` with
77/// `skipna`, and ends the sum (`None`) without it.
78fn pandas_mean_lanes(x: &[f64], skipna: bool) -> Option<([f64; 2], usize)> {
79    if x.len() > PANDAS_MOMENTS_LEAF {
80        let mid = x.len() / 2 / 2 * 2;
81        let (left, left_count) = pandas_mean_lanes(&x[..mid], skipna)?;
82        let (right, right_count) = pandas_mean_lanes(&x[mid..], skipna)?;
83        let sum = [left[0] + right[0], left[1] + right[1]];
84        return Some((sum, left_count + right_count));
85    }
86    let mut sum = [0.0; 2];
87    let mut count = 0;
88    for pair in x.as_chunks::<2>().0 {
89        if !skipna && pair.iter().any(|v| v.is_nan()) {
90            return None;
91        }
92        for (lane, &v) in pair.iter().enumerate() {
93            sum[lane] += if v.is_nan() { 0.0 } else { v };
94            count += usize::from(!v.is_nan());
95        }
96    }
97    Some((sum, count))
98}

100/// Lane sums of $d, d^2, d^3, d^4$ with $d = x_i - \text{mean}$ (0 for a NaN)
101/// of an even-length `x`, as pandas' `accumulate_central_diffs_pairwise`, in
102/// the same lanes and leaves as [`pandas_mean_lanes`].
103fn pandas_central_lanes(x: &[f64], mean: f64) -> [[f64; 2]; 4] {
104    if x.len() > PANDAS_MOMENTS_LEAF {
105        let mid = x.len() / 2 / 2 * 2;
106        let left = pandas_central_lanes(&x[..mid], mean);
107        let right = pandas_central_lanes(&x[mid..], mean);
108        return std::array::from_fn(|k| [left[k][0] + right[k][0], left[k][1] + right[k][1]]);
109    }
110    let mut acc = [[0.0; 2]; 4];
111    for pair in x.as_chunks::<2>().0 {
112        for (lane, &v) in pair.iter().enumerate() {
113            let d = if v.is_nan() { 0.0 } else { v - mean };
114            let d2 = d * d;
115            acc[0][lane] += d;
116            acc[1][lane] += d2;
117            acc[2][lane] += d2 * d;
118            acc[3][lane] += d2 * d2;
119        }
120    }
121    acc
122}

124/// `std::max(v, 0.0)`: a negative becomes 0, NaN stays NaN (unlike
125/// [`f64::max`]).
126fn clamp_negative(v: f64) -> f64 {
127    if v < 0.0 { 0.0 } else { v }

130/// The number of non-NaN values of `x` and its central moment sums
131/// $m_k = \sum_i (x_i - \mu)^k$ for $k = 2, 3, 4$, bit for bit as pandas 3.1
132/// computes them for `Series.skew` and `Series.kurtosis` (`moments_reduce`,
133/// x86_64 build).
134///
135/// pandas sums in two passes: a trial mean, then the powers of the
136/// deviations from it, both in two SSE2 lanes with pairwise sums (see
137/// [`pandas_mean_lanes`]) and the odd last value added on its own. The
138/// trial mean's rounding error is then corrected (Pébay et al., 2016), so a
139/// constant series gives exactly $m_2 = 0$. With `skipna`, NaN values are
140/// skipped where they are; without it, a NaN makes every sum NaN.
141fn pandas_moments(x: &[f64], skipna: bool) -> (usize, [f64; 3]) {
142    let undefined = (x.len(), [f64::NAN; 3]);
143    let (front, last) = x.split_at(x.len() / 2 * 2);
144    let Some((lanes, front_count)) = pandas_mean_lanes(front, skipna) else {
145        return undefined;
146    };
147    if !skipna && last.iter().any(|v| v.is_nan()) {
148        return undefined;
149    }
150    // The odd last value, unless it is a skipped NaN; pandas sums it on its
151    // own from 0.0 and adds that to the two lanes
152    let last = last.first().copied().filter(|v| !v.is_nan());
153    let count = front_count + usize::from(last.is_some());
154    if count == 0 {
155        return (0, [0.0; 3]);
156    }
157    let n = count as f64;
158    let mean = (lanes[0] + lanes[1] + last.map_or(0.0, |v| 0.0 + v)) / n;
159
160    let lanes = pandas_central_lanes(front, mean);
161    let last_acc = last.map_or([0.0; 4], |v| {
162        let d = v - mean;
163        let d2 = d * d;
164        [0.0 + d, 0.0 + d2, 0.0 + d2 * d, 0.0 + d2 * d2]
165    });
166    let [m1, m2, m3, m4]: [f64; 4] =
167        std::array::from_fn(|k| lanes[k][0] + lanes[k][1] + last_acc[k]);
168
169    // Correct for the trial mean's error c = m1 / n, as pandas'
170    // compute_moments_with_correction does
171    let c = m1 / n;
172    let c2 = c * c;
173    let term1 = m1 * c;
174    let corrected = [
175        clamp_negative(m2 - term1),
176        m3 - 3.0 * m2 * c + 2.0 * c * term1,
177        clamp_negative(m4 - 4.0 * m3 * c + 6.0 * m2 * c2 - 3.0 * term1 * c2),
178    ];
179    (count, corrected)
180}

188/// Bias-adjusted sample skewness (Fisher-Pearson G1) of a series.
189///
190/// Matches `pandas.Series.skew(skipna=False)` (pandas 3.1), which tsfresh
191/// uses, bit for bit on x86_64:
192/// $$ G_1 = \frac{n \sqrt{n - 1}}{n - 2} \frac{m_3}{m_2^{3/2}}, $$
193/// with $m_k = \sum_{i=1}^{n} (x_i - \mu)^k$ computed as pandas does.
194/// Returns NaN for fewer than 3 values, for a constant series ($0 / 0$) and
195/// if any value is NaN.
196pub fn sample_skewness(arr: &ArrayView1<f64>) -> f64 {
197    let (count, [m2, m3, _]) = pandas_moments(&contiguous(arr), false);
198    if count < 3 {
199        return f64::NAN;
200    }
201    let n = count as f64;
202    m3 / (m2 * m2.sqrt()) * (n * (n - 1.0).sqrt() / (n - 2.0))
203}

205/// Bias-adjusted sample excess kurtosis of a series.
206///
207/// Matches `pandas.Series.kurtosis()` (pandas 3.1, `skipna=True`), which
208/// tsfresh uses, bit for bit on x86_64:
209/// $$ G_2 = \frac{n - 1}{(n - 2) (n - 3)}
210///        \left( \frac{n (n + 1) m_4}{m_2^2} - 3 (n - 1) \right), $$
211/// with $m_k = \sum_{i=1}^{n} (x_i - \mu)^k$ computed as pandas does over
212/// the $n$ non-NaN values. A normal distribution scores 0. Returns NaN for
213/// fewer than 4 non-NaN values and for a constant series ($0 / 0$).
214pub fn sample_excess_kurtosis(arr: &ArrayView1<f64>) -> f64 {
215    let (count, [m2, _, m4]) = pandas_moments(&contiguous(arr), true);
216    if count < 4 {
217        return f64::NAN;
218    }
219    let n = count as f64;
220    let inner = n * (n + 1.0) * (m4 / (m2 * m2)) - 3.0 * (n - 1.0);
221    (n - 1.0) / ((n - 2.0) * (n - 3.0)) * inner
222}

224/// Ordinary least squares intercept and slope of `y` against its index
225/// $0, 1, \dots, n - 1$, as `scipy.stats.linregress` (which tsfresh uses)
226/// computes them: centred sums,
227/// $b = \sum (i - \bar{i})(y_i - \bar{y}) / \sum (i - \bar{i})^2$ and
228/// $a = \bar{y} - b\,\bar{i}$, summed as numpy does. (A one-pass
229/// $\sum i\,y_i - n\,\bar{i}\,\bar{y}$ loses digits to cancellation.)
230/// NaN for fewer than 2 values.
231pub(crate) fn calculate_sequential_ols(y: &[f64]) -> (f64, f64) {
232    let n = y.len();
233    if n < 2 {
234        return (f64::NAN, f64::NAN);
235    }
236    let mean_x = (n as f64 - 1.0) / 2.0;
237    let mean_y = numpy_mean(y);
238    let ss_xx = numpy_sum(n, |i| (i as f64 - mean_x) * (i as f64 - mean_x));
239    let ss_xy = numpy_sum(n, |i| (i as f64 - mean_x) * (y[i] - mean_y));
240    let slope = ss_xy / ss_xx;
241    (mean_y - slope * mean_x, slope)
242}

254/// The `q`-quantile of `sorted` (ascending, not empty), exactly as
255/// `np.quantile` computes it (method "linear"): index $h = (n - 1)\,q$,
256/// neighbours $a = x_{(\lfloor h \rfloor)}$, $b = x_{(\lfloor h \rfloor + 1)}$
257/// and weight $t = h - \lfloor h \rfloor$, then numpy's `_lerp`:
258/// $a + (b - a)\,t$ for $t < 0.5$, $b - (b - a)(1 - t)$ otherwise. At or
259/// beyond the last index numpy uses the last value for both neighbours with
260/// $t = h + 1$ (it computes the weight from the replacement index $-1$).
261/// So an infinite neighbour can give NaN, exactly as in numpy.
262pub fn numpy_quantile(sorted: &[f64], q: f64) -> f64 {
263    let n = sorted.len();
264    let h = (n as f64 - 1.0) * q;
265    let (i, j, t) = if h >= (n - 1) as f64 {
266        (n - 1, n - 1, h + 1.0)
267    } else {
268        let previous = h.floor();
269        (previous as usize, previous as usize + 1, h - previous)
270    };
271    let (a, b) = (sorted[i], sorted[j]);
272    let diff = b - a;
273    if t >= 0.5 {
274        b - diff * (1.0 - t)
275    } else {
276        a + diff * t
277    }
278}

280/// The median of `values` (not empty, reordered in place), exactly as
281/// `np.median` computes it: the `np.mean` of the middle value, or of the two
282/// middle values for an even count. Found with a selection, $O(n)$.
283pub fn numpy_median(values: &mut [f64]) -> f64 {
284    let n = values.len();
285    let (left, &mut upper, _) = values.select_nth_unstable_by(n / 2, f64::total_cmp);
286    if n % 2 == 1 {
287        return numpy_mean(&[upper]);
288    }
289    let lower = left.iter().copied().max_by(f64::total_cmp).unwrap_or(upper);
290    numpy_mean(&[lower, upper])
291}

293/// `num` evenly spaced values from `start` to `stop` (both included),
294/// exactly as `np.linspace(start, stop, num)` computes them: $i \cdot s +
295/// \text{start}$ with step $s = (\text{stop} - \text{start}) / (\text{num} - 1)$
296/// (or $(i / (\text{num} - 1)) \cdot \Delta + \text{start}$ if the step
297/// underflows to 0), the last value set to `stop`.
298pub(crate) fn numpy_linspace(start: f64, stop: f64, num: usize) -> Vec<f64> {
299    let delta = stop - start;
300    let mut y: Vec<f64> = if num > 1 {
301        let div = (num - 1) as f64;
302        let step = delta / div;
303        if step == 0.0 {
304            (0..num).map(|i| i as f64 / div * delta + start).collect()
305        } else {
306            (0..num).map(|i| i as f64 * step + start).collect()
307        }
308    } else {
309        (0..num).map(|i| i as f64 * delta + start).collect()
310    };
311    if num > 1 {
312        y[num - 1] = stop;
313    }
314    y
315}

317/// numpy's ordering of floats for sorting and searching: NaN sorts last.
318fn numpy_less(a: f64, b: f64) -> bool {
319    a < b || (b.is_nan() && !a.is_nan())
320}

322/// `np.searchsorted(arr, keys, side="left")`, reproducing numpy's binary
323/// search step for step (including how it reuses the bounds of the previous
324/// key), so that the result also equals numpy's when `arr` is not sorted
325/// (for example edges that came out NaN).
326pub(crate) fn numpy_searchsorted_left(arr: &[f64], keys: &[f64]) -> Vec<usize> {
327    let n = arr.len();
328    let mut out = Vec::with_capacity(keys.len());
329    let (mut min_idx, mut max_idx) = (0, n);
330    let Some(&first) = keys.first() else {
331        return out;
332    };
333    let mut last_key = first;
334    for &key in keys {
335        if numpy_less(last_key, key) {
336            max_idx = n;
337        } else {
338            min_idx = 0;
339            max_idx = if max_idx < n { max_idx + 1 } else { n };
340        }
341        last_key = key;
342        while min_idx < max_idx {
343            let mid = min_idx + ((max_idx - min_idx) >> 1);
344            if numpy_less(arr[mid], key) {
345                min_idx = mid + 1;
346            } else {
347                max_idx = mid;
348            }
349        }
350        out.push(min_idx);
351    }
352    out
353}

367fn np_arg_extreme(x: &ArrayView1<f64>, better: fn(f64, f64) -> bool) -> Option<usize> {
368    if let Some(i) = x.iter().position(|v| v.is_nan()) {
369        return Some(i);
370    }
371    let mut best: Option<(usize, f64)> = None;
372    for (i, &v) in x.iter().enumerate() {
373        match best {
374            Some((_, b)) if !better(v, b) => {}
375            _ => best = Some((i, v)),
376        }
377    }
378    best.map(|(i, _)| i)
379}
