6/// Sum of `value(0), ..., value(n - 1)` exactly as numpy's `add.reduce` sums
7/// a contiguous `float64` array (numpy 2.x): pairwise summation with eight
8/// accumulators for blocks of up to 128 values, split in halves above that,
9/// added to an initial `0.0`. Results equal numpy's bit for bit, which a
10/// straight or Welford sum does not.
11pub fn numpy_sum(n: usize, value: impl Fn(usize) -> f64) -> f64 {
12    0.0 + pairwise_sum(0, n, &value)
13}
14
15fn pairwise_sum(start: usize, n: usize, value: &impl Fn(usize) -> f64) -> f64 {
16    if n < 8 {
17        (start..start + n).fold(0.0, |res, i| res + value(i))
18    } else if n <= 128 {
19        let mut r: [f64; 8] = std::array::from_fn(|j| value(start + j));
20        let mut i = 8;
21        while i < n - n % 8 {
22            for (j, acc) in r.iter_mut().enumerate() {
23                *acc += value(start + i + j);
24            }
25            i += 8;
26        }
27        let mut res = ((r[0] + r[1]) + (r[2] + r[3])) + ((r[4] + r[5]) + (r[6] + r[7]));
28        for k in i..n {
29            res += value(start + k);
30        }
31        res
32    } else {
33        let mut n2 = n / 2;
34        n2 -= n2 % 8;
35        pairwise_sum(start, n2, value) + pairwise_sum(start + n2, n - n2, value)
36    }
37}

39/// Arithmetic mean, exactly as `np.mean` (see [`numpy_sum`]). NaN for an
40/// empty series or if any value is NaN.
41pub fn numpy_mean(x: &[f64]) -> f64 {
42    numpy_sum(x.len(), |i| x[i]) / x.len() as f64
43}

45/// Population variance (`ddof = 0`) of a series.
46///
47/// Exactly `np.var` as used by tsfresh: the mean, then the mean of the
48/// squared deviations, both summed as numpy does (see [`numpy_sum`]).
49/// Returns the raw value; how a zero variance is handled is up to each
50/// feature.
51pub fn population_var(arr: &ArrayView1<f64>) -> f64 {
52    let n = arr.len();
53    let mean = numpy_sum(n, |i| arr[i]) / n as f64;
54    numpy_sum(n, |i| {
55        let d = arr[i] - mean;
56        d * d
57    }) / n as f64
58}
59
60/// Population standard deviation (`ddof = 0`) of a series: the square root
61/// of [`population_var`], exactly `np.std` as used by tsfresh. Returns the
62/// raw value; how a zero standard deviation is handled is up to each
63/// feature.
64pub fn population_std(arr: &ArrayView1<f64>) -> f64 {
65    population_var(arr).sqrt()
66}

68/// Treat a sum of `k`-th power deviations as zero when it is within
69/// floating-point noise, as pandas 3 does (`_zero_out_fperr` with tolerance
70/// `(eps * max|x|)^k * n`).
71fn zero_out_fperr(m: f64, k: i32, max_abs: f64, n: f64) -> f64 {
72    let tol = (f64::EPSILON * max_abs).powi(k) * n;
73    if m.abs() < tol { 0.0 } else { m }
74}

76/// Mean and largest absolute value of a series, as pandas computes them.
77fn mean_and_max_abs(arr: &ArrayView1<f64>) -> (f64, f64) {
78    let mean = arr.sum() / arr.len() as f64;
79    let max_abs = arr.fold(0.0_f64, |acc, &x| acc.max(x.abs()));
80    (mean, max_abs)
81}

83/// Bias-adjusted sample skewness (Fisher-Pearson G1) of a series.
84///
85/// Matches `pandas.Series.skew` (pandas 3), which tsfresh uses:
86/// $$ G_1 = \frac{n \sqrt{n - 1}}{n - 2} \frac{m_3}{m_2^{3/2}}, $$
87/// with $m_k = \sum_{i=1}^{n} (x_i - \mu)^k$. Returns NaN for fewer than 3
88/// values and 0 for a (numerically) constant series.
89pub fn sample_skewness(arr: &ArrayView1<f64>) -> f64 {
90    if arr.len() < 3 {
91        return f64::NAN;
92    }
93    let n = arr.len() as f64;
94    let (mean, max_abs) = mean_and_max_abs(arr);
95    let (m2, m3) = arr.fold((0.0, 0.0), |(m2, m3), &x| {
96        let d2 = (x - mean).powi(2);
97        (m2 + d2, m3 + d2 * (x - mean))
98    });
99    let m2 = zero_out_fperr(m2, 2, max_abs, n);
100    let m3 = zero_out_fperr(m3, 3, max_abs, n);
101    if m2 == 0.0 {
102        return 0.0;
103    }
104    (n * (n - 1.0).sqrt() / (n - 2.0)) * (m3 / m2.powf(1.5))
105}

107/// Bias-adjusted sample excess kurtosis of a series.
108///
109/// Matches `pandas.Series.kurtosis` (pandas 3), which tsfresh uses:
110/// $$ G_2 = \frac{n (n + 1) (n - 1) m_4}{(n - 2) (n - 3) m_2^2}
111///        - \frac{3 (n - 1)^2}{(n - 2) (n - 3)}, $$
112/// with $m_k = \sum_{i=1}^{n} (x_i - \mu)^k$. A normal distribution scores 0.
113/// Returns NaN for fewer than 4 values and 0 for a (numerically) constant
114/// series.
115pub fn sample_excess_kurtosis(arr: &ArrayView1<f64>) -> f64 {
116    if arr.len() < 4 {
117        return f64::NAN;
118    }
119    let n = arr.len() as f64;
120    let (mean, max_abs) = mean_and_max_abs(arr);
121    let (m2, m4) = arr.fold((0.0, 0.0), |(m2, m4), &x| {
122        let d2 = (x - mean).powi(2);
123        (m2 + d2, m4 + d2 * d2)
124    });
125    let m2 = zero_out_fperr(m2, 2, max_abs, n);
126    let m4 = zero_out_fperr(m4, 4, max_abs, n);
127    let denominator = (n - 2.0) * (n - 3.0) * m2 * m2;
128    if denominator == 0.0 {
129        return 0.0;
130    }
131    let adj = 3.0 * (n - 1.0).powi(2) / ((n - 2.0) * (n - 3.0));
132    n * (n + 1.0) * (n - 1.0) * m4 / denominator - adj
133}

135/// Ordinary least squares intercept and slope of `y` against its index
136/// $0, 1, \dots, n - 1$, as `scipy.stats.linregress` (which tsfresh uses)
137/// computes them: centred sums,
138/// $b = \sum (i - \bar{i})(y_i - \bar{y}) / \sum (i - \bar{i})^2$ and
139/// $a = \bar{y} - b\,\bar{i}$, summed as numpy does. (A one-pass
140/// $\sum i\,y_i - n\,\bar{i}\,\bar{y}$ loses digits to cancellation.)
141/// NaN for fewer than 2 values.
142pub(crate) fn calculate_sequential_ols(y: &[f64]) -> (f64, f64) {
143    let n = y.len();
144    if n < 2 {
145        return (f64::NAN, f64::NAN);
146    }
147    let mean_x = (n as f64 - 1.0) / 2.0;
148    let mean_y = numpy_mean(y);
149    let ss_xx = numpy_sum(n, |i| (i as f64 - mean_x) * (i as f64 - mean_x));
150    let ss_xy = numpy_sum(n, |i| (i as f64 - mean_x) * (y[i] - mean_y));
151    let slope = ss_xy / ss_xx;
152    (mean_y - slope * mean_x, slope)
153}

155pub(crate) fn get_length_sequences_where(x: &ndarray::Array1<bool>) -> Vec<usize> {
156    let mut group_lengths = Vec::new();
157    for (key, group) in &x.into_iter().chunk_by(|elt| *elt) {
158        if *key {
159            group_lengths.push(group.count());
160        }
161    }
162    group_lengths
163}

165/// The `q`-quantile of `sorted` (ascending, not empty), exactly as
166/// `np.quantile` computes it (method "linear"): index $h = (n - 1)\,q$,
167/// neighbours $a = x_{(\lfloor h \rfloor)}$, $b = x_{(\lfloor h \rfloor + 1)}$
168/// and weight $t = h - \lfloor h \rfloor$, then numpy's `_lerp`:
169/// $a + (b - a)\,t$ for $t < 0.5$, $b - (b - a)(1 - t)$ otherwise. At or
170/// beyond the last index numpy uses the last value for both neighbours with
171/// $t = h + 1$ (it computes the weight from the replacement index $-1$).
172/// So an infinite neighbour can give NaN, exactly as in numpy.
173pub fn numpy_quantile(sorted: &[f64], q: f64) -> f64 {
174    let n = sorted.len();
175    let h = (n as f64 - 1.0) * q;
176    let (i, j, t) = if h >= (n - 1) as f64 {
177        (n - 1, n - 1, h + 1.0)
178    } else {
179        let previous = h.floor();
180        (previous as usize, previous as usize + 1, h - previous)
181    };
182    let (a, b) = (sorted[i], sorted[j]);
183    let diff = b - a;
184    if t >= 0.5 {
185        b - diff * (1.0 - t)
186    } else {
187        a + diff * t
188    }
189}

191/// The median of `values` (not empty, reordered in place), exactly as
192/// `np.median` computes it: the `np.mean` of the middle value, or of the two
193/// middle values for an even count. Found with a selection, $O(n)$.
194pub fn numpy_median(values: &mut [f64]) -> f64 {
195    let n = values.len();
196    let (left, &mut upper, _) = values.select_nth_unstable_by(n / 2, f64::total_cmp);
197    if n % 2 == 1 {
198        return numpy_mean(&[upper]);
199    }
200    let lower = left.iter().copied().max_by(f64::total_cmp).unwrap_or(upper);
201    numpy_mean(&[lower, upper])
202}

204/// `num` evenly spaced values from `start` to `stop` (both included),
205/// exactly as `np.linspace(start, stop, num)` computes them: $i \cdot s +
206/// \text{start}$ with step $s = (\text{stop} - \text{start}) / (\text{num} - 1)$
207/// (or $(i / (\text{num} - 1)) \cdot \Delta + \text{start}$ if the step
208/// underflows to 0), the last value set to `stop`.
209pub(crate) fn numpy_linspace(start: f64, stop: f64, num: usize) -> Vec<f64> {
210    let delta = stop - start;
211    let mut y: Vec<f64> = if num > 1 {
212        let div = (num - 1) as f64;
213        let step = delta / div;
214        if step == 0.0 {
215            (0..num).map(|i| i as f64 / div * delta + start).collect()
216        } else {
217            (0..num).map(|i| i as f64 * step + start).collect()
218        }
219    } else {
220        (0..num).map(|i| i as f64 * delta + start).collect()
221    };
222    if num > 1 {
223        y[num - 1] = stop;
224    }
225    y
226}

228/// numpy's ordering of floats for sorting and searching: NaN sorts last.
229fn numpy_less(a: f64, b: f64) -> bool {
230    a < b || (b.is_nan() && !a.is_nan())
231}

233/// `np.searchsorted(arr, keys, side="left")`, reproducing numpy's binary
234/// search step for step (including how it reuses the bounds of the previous
235/// key), so that the result also equals numpy's when `arr` is not sorted
236/// (for example edges that came out NaN).
237pub(crate) fn numpy_searchsorted_left(arr: &[f64], keys: &[f64]) -> Vec<usize> {
238    let n = arr.len();
239    let mut out = Vec::with_capacity(keys.len());
240    let (mut min_idx, mut max_idx) = (0, n);
241    let Some(&first) = keys.first() else {
242        return out;
243    };
244    let mut last_key = first;
245    for &key in keys {
246        if numpy_less(last_key, key) {
247            max_idx = n;
248        } else {
249            min_idx = 0;
250            max_idx = if max_idx < n { max_idx + 1 } else { n };
251        }
252        last_key = key;
253        while min_idx < max_idx {
254            let mid = min_idx + ((max_idx - min_idx) >> 1);
255            if numpy_less(arr[mid], key) {
256                min_idx = mid + 1;
257            } else {
258                max_idx = mid;
259            }
260        }
261        out.push(min_idx);
262    }
263    out
264}

266/// Index of the first maximum, like `np.argmax`: if the series contains NaN,
267/// the index of the first NaN. `None` for an empty series.
268pub(crate) fn np_argmax(x: &ArrayView1<f64>) -> Option<usize> {
269    np_arg_extreme(x, |a, b| a > b)
270}
271
272/// Index of the first minimum, like `np.argmin`: if the series contains NaN,
273/// the index of the first NaN. `None` for an empty series.
274pub(crate) fn np_argmin(x: &ArrayView1<f64>) -> Option<usize> {
275    np_arg_extreme(x, |a, b| a < b)
276}

278fn np_arg_extreme(x: &ArrayView1<f64>, better: fn(f64, f64) -> bool) -> Option<usize> {
279    if let Some(i) = x.iter().position(|v| v.is_nan()) {
280        return Some(i);
281    }
282    let mut best: Option<(usize, f64)> = None;
283    for (i, &v) in x.iter().enumerate() {
284        match best {
285            Some((_, b)) if !better(v, b) => {}
286            _ => best = Some((i, v)),
287        }
288    }
289    best.map(|(i, _)| i)
290}
