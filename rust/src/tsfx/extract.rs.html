36/// Collect `lf` on the in-memory engine, as `collect()` did before polars 2.0.
37/// Since 2.0 a plain `collect()` runs lazy queries on the streaming engine,
38/// which costs tsfx more per query and per group.
39pub fn collect_in_memory(lf: LazyFrame) -> PolarsResult<DataFrame> {
40    Ok(lf.collect_with_engine(Engine::InMemory)?.unwrap_single())
41}

43/// Parse the `field` duration of the dynamic group-by settings. An invalid
44/// string is a settings error (a ValueError in Python) rather than the panic
45/// of `Duration::parse`.
46fn parse_window_duration(field: &str, value: &str) -> Result<Duration, ExtractionError> {
47    Duration::try_parse(value).map_err(|e| {
48        ExtractionError::InvalidSettings(format!("invalid {field} duration {value:?}: {e}"))
49    })
50}

52pub fn lazy_feature_df(
53    df: LazyFrame,
54    opts: ExtractionSettings,
55) -> Result<LazyFrame, ExtractionError> {
56    if let Some(scaling) = &opts.scaling {
57        scaling.validate()?;
58    }
59    let features = aggregators(&opts)?;
60    // Columns identifying a group (or window), which scaling leaves alone
61    let mut key_cols = opts.grouping_cols.clone();
62    if let Some(dynamic_settings) = &opts.dynamic_settings {
63        key_cols.push(dynamic_settings.time_col.clone());
64    }
65    let grouping_cols: Vec<Expr> = opts.grouping_cols.into_iter().map(col).collect();
66    let mut selected_cols = grouping_cols.clone();
67    for val_col in &opts.value_cols {
68        selected_cols.push(col(val_col));
69    }
70    let gdf = if let Some(dynamic_settings) = opts.dynamic_settings {
71        let datetime_format = if let Some(dt_fmt) = dynamic_settings.datetime_format {
72            dt_fmt
73        } else {
74            "%Y-%m-%d %H:%S:%M".to_string()
75        };
76        let time_options = StrptimeOptions {
77            format: Some(datetime_format.into()),
78            ..Default::default()
79        };
80        selected_cols.push(col(&dynamic_settings.time_col));
81        df.select(&selected_cols).group_by_dynamic(
82            col(&dynamic_settings.time_col).str().to_date(time_options),
83            &grouping_cols,
84            DynamicGroupOptions {
85                index_column: dynamic_settings.time_col.into(),
86                every: parse_window_duration("every", &dynamic_settings.every)?,
87                period: parse_window_duration("period", &dynamic_settings.period)?,
88                offset: parse_window_duration("offset", &dynamic_settings.offset)?,
89                ..Default::default()
90            },
91        )
92    } else {
93        df.select(&selected_cols).group_by(&grouping_cols)
94    };
95    let mut fdf = collect_in_memory(gdf.agg(features.aggregations))?;
96    for (column, plan) in &features.kernels {
97        plan.unpack(&mut fdf, column)?;
98    }
99    // Scale in memory too, whatever engine collects the result: the streaming
100    // engine merges a column's partial means and variances in thread order, so
101    // its scaled features differ in the last bits from run to run
102    Ok(match &opts.scaling {
103        Some(scaling) => collect_in_memory(scale_features(fdf, scaling, &key_cols))?.lazy(),
104        None => fdf.lazy(),
105    })
106}

262    fn test_invalid_window_duration_is_an_error() {
263        // Regression: Duration::parse panicked on a duration without a unit
264        for field in ["every", "period", "offset"] {
265            let mut dynamic = DynamicGroupBySettings {
266                time_col: "time".to_string(),
267                every: "1y".to_string(),
268                period: "1y".to_string(),
269                offset: "0y".to_string(),
270                datetime_format: Some("%Y-%m-%d".to_string()),
271            };
272            let target = match field {
273                "every" => &mut dynamic.every,
274                "period" => &mut dynamic.period,
275                _ => &mut dynamic.offset,
276            };
277            *target = "0".to_string();
278            let df = df![
279                "id" => ["a", "a"],
280                "time" => ["2001-01-01", "2002-01-01"],
281                "value" => [1.0, 2.0],
282            ]
283            .unwrap()
284            .lazy();
285            let opts = ExtractionSettings {
286                grouping_cols: vec!["id".to_string()],
287                value_cols: vec!["value".to_string()],
288                feature_setting: FeatureSetting::Minimal,
289                config_path: None,
290                dynamic_settings: Some(dynamic),
291                scaling: None,
292            };
293            let Err(err) = lazy_feature_df(df, opts) else {
294                panic!("{field} = \"0\" was accepted");
295            };
296            assert!(matches!(err, ExtractionError::InvalidSettings(_)), "{err}");
297            let expected = format!("invalid {field} duration \"0\"");
298            assert!(err.to_string().contains(&expected), "{err}");
299        }

302    /// The float columns of `df` sorted by `id`, as bit patterns (any NaN as
303    /// one value), so that equal means bit-for-bit equal.
304    fn float_bits(df: DataFrame) -> Vec<(String, Vec<Option<u64>>)> {
305        let df = df.sort(["id"], SortMultipleOptions::default()).unwrap();
306        df.columns()
307            .iter()
308            .filter(|c| c.dtype() == &DataType::Float64)
309            .map(|c| {
310                let values = c.f64().unwrap().to_vec();
311                let bits = values.into_iter();
312                let bits = bits.map(|v| v.map(|x| if x.is_nan() { u64::MAX } else { x.to_bits() }));
313                (c.name().to_string(), bits.collect())
314            })
315            .collect()
316    }

318    /// Scaled features do not depend on the engine that collects the result.
319    /// Polars 2's streaming engine reduces a column's mean and standard
320    /// deviation in parallel and merges the partial results in whatever order
321    /// its threads finish, so a scaling run there differed in the last bits
322    /// from run to run.
323    #[test]
324    fn test_scaling_does_not_depend_on_engine() {
325        let df = df![
326            "id" => ["a", "b", "c", "d"],
327            "value" => [1.0, 2.0, 3.0, 4.0],
328        ]
329        .unwrap()
330        .lazy();
331        let opts = || ExtractionSettings {
332            grouping_cols: vec!["id".to_string()],
333            value_cols: vec!["value".to_string()],
334            feature_setting: FeatureSetting::Minimal,
335            config_path: None,
336            dynamic_settings: None,
337            scaling: Some(Scaling::Standard),
338        };
339        let expected = collect_in_memory(lazy_feature_df(df.clone(), opts()).unwrap()).unwrap();
340        let expected = float_bits(expected);
341        for _ in 0..100 {
342            let lf = lazy_feature_df(df.clone(), opts()).unwrap();
343            let streamed = lf
344                .collect_with_engine(Engine::Streaming)
345                .unwrap()
346                .unwrap_single();
347            assert_eq!(float_bits(streamed), expected);
348        }
349    }
