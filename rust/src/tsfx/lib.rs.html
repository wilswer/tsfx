15/// Defines the complexity level of the feature extraction process.
16///
17/// Attributes:
18///     Minimal: Extracts a small, basic set of features.
19///     Efficient: Extracts a balanced set of features optimized for performance.
20///     Comprehensive: Extracts an exhaustive set of features.
21#[pyclass(from_py_object, name = "FeatureSetting", eq, eq_int)]
22#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord)]
23enum PyFeatureSetting {
24    Minimal,
25    Efficient,
26    Comprehensive,
27}

29/// How to scale the extracted features, column-wise across groups.
30///
31/// Matches scikit-learn's `StandardScaler` and `MinMaxScaler`: NaN and null
32/// are ignored when fitting and kept in the output, as are infinities (which
33/// scikit-learn rejects), and (near-)constant features map to 0 (`Standard`)
34/// or the lower end of the range (`MinMax`). Grouping and time columns are
35/// not scaled.
36///
37/// Attributes:
38///     Standard: Zero mean and unit (population) variance.
39///     MinMax: Maps each feature's [min, max] onto `feature_range`, (0, 1) by default.
40#[pyclass(from_py_object, name = "ScalerSetting", eq)]
41#[derive(Clone, Debug, PartialEq)]
42enum PyScalerSetting {
43    Standard {},
44    #[pyo3(constructor = (feature_range = (0.0, 1.0)))]
45    MinMax {
46        feature_range: (f64, f64),
47    },
48}

73#[pymethods]
74impl PyExtractionSettings {
75    /// Initialize the extraction settings.
76    ///
77    /// Args:
78    ///     grouping_cols (list[str]): The columns used to group the data (e.g., IDs).
79    ///     value_cols (list[str]): The columns containing the numerical values to extract features from.
80    ///     feature_setting (FeatureSetting): The complexity/depth of features to calculate.
81    ///     config_path (str | None, optional): Path to a TOML feature config file, which must exist. Defaults to None: `.tsfx-config.toml` in the working directory if present, else the default config.
82    ///     dynamic_settings (DynamicGroupBySettings | None, optional): Settings for rolling/dynamic time windows. Defaults to None.
83    ///     scaling (ScalerSetting | None, optional): Scale the features across groups after extraction. Defaults to None: no scaling.
84    #[new]
85    #[pyo3(signature = (grouping_cols, value_cols, feature_setting, config_path=None, dynamic_settings=None, scaling=None))]
86    fn new(
87        grouping_cols: Vec<String>,
88        value_cols: Vec<String>,
89        feature_setting: PyFeatureSetting,
90        config_path: Option<String>,
91        dynamic_settings: Option<PyDynamicGroupBySettings>,
92        scaling: Option<&Bound<PyAny>>,
93    ) -> PyResult<Self> {
94        let scaling = match scaling {
95            // A variant class without arguments, e.g. `ScalerSetting.Standard`
96            Some(s)
97                if s.cast::<PyType>()
98                    .is_ok_and(|t| t.is_subclass_of::<PyScalerSetting>().unwrap_or(false)) =>
99            {
100                Some(s.call0()?.extract()?)
101            }
102            Some(s) => Some(s.extract::<PyScalerSetting>()?),
103            None => None,
104        };
105        if let Some(s) = &scaling {
106            Scaling::from(s.clone()).validate()?;
107        }
108        Ok(PyExtractionSettings {
109            grouping_cols,
110            value_cols,
111            feature_setting,
112            config_path,
113            dynamic_settings,
114            scaling,
115        })
116    }
117
118    /// Columns the extraction reads: grouping, value and (if set) time
119    /// column. Internal, used to collect only these before calling Rust.
120    #[getter]
121    fn _input_columns(&self) -> Vec<String> {
122        let mut cols = self.grouping_cols.clone();
123        cols.extend(self.value_cols.iter().cloned());
124        if let Some(dynamic) = &self.dynamic_settings {
125            cols.push(dynamic.time_col.clone());
126        }
127        cols
128    }
129}

207/// Extract time-series features from a Polars LazyFrame.
208///
209/// This function computes features based on the provided settings. It evaluates
210/// the lazy computation graph and returns an in-memory DataFrame.
211///
212/// Args:
213///     lf (polars.LazyFrame): The input data to extract features from.
214///     settings (ExtractionSettings): The configuration controlling grouping and feature complexity.
215///     streaming (bool, optional): If True, executes the query using Polars' streaming engine
216///         for out-of-core processing. Defaults to False.
217///
218/// Returns:
219///     polars.DataFrame: A new DataFrame containing the grouped IDs and their extracted features.
220///
221/// Raises:
222///     Exception: If an underlying Polars error occurs during collection.
223#[pyfunction]
224#[pyo3(signature = (lf, settings, streaming=false))]
225fn extract_features(
226    lf: PyLazyFrame,
227    settings: PyExtractionSettings,
228    streaming: bool,
229) -> PyResult<PyDataFrame> {
230    let lf = lf.into();
231    let settings = settings.into();
232    let lf = if !streaming {
233        lazy_feature_df(lf, settings)?
234            .collect()
235            .map_err(ExtractionError::PolarsError)?
236    } else {
237        lazy_feature_df(lf, settings)?
238            .with_streaming(true)
239            .collect()
240            .map_err(ExtractionError::PolarsError)?
241    };
242    Ok(PyDataFrame(lf))
243}

245/// Extract features from an in-memory DataFrame.
246///
247/// Internal counterpart of `extract_features` for the non-streaming path: a
248/// DataFrame crosses from Python via the Arrow C interface without copying,
249/// while a LazyFrame's plan (and any in-memory data in it) is serialised.
250#[pyfunction]
251#[pyo3(signature = (df, settings))]
252fn extract_features_df(df: PyDataFrame, settings: PyExtractionSettings) -> PyResult<PyDataFrame> {
253    let df = lazy_feature_df(df.0.lazy(), settings.into())?
254        .collect()
255        .map_err(ExtractionError::PolarsError)?;
256    Ok(PyDataFrame(df))
257}

259/// Time Series Feature Extraction module.
260///
261/// This module provides high-performance feature extraction capabilities for
262/// time series data, leveraging a Rust core and Polars dataframes.
263#[pymodule]
264fn tsfx(_py: Python, m: &Bound<PyModule>) -> PyResult<()> {
265    m.add_class::<PyFeatureSetting>()?;
266    m.add_class::<PyExtractionSettings>()?;
267    m.add_class::<PyDynamicGroupBySettings>()?;
268    m.add_class::<PyScalerSetting>()?;
269    m.add_function(wrap_pyfunction!(extract_features, m)?)?;
270    m.add_function(wrap_pyfunction!(extract_features_df, m)?)?;
271    Ok(())
272}
