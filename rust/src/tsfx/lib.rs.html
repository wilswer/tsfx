11/// Defines the complexity level of the feature extraction process.
12///
13/// Attributes:
14///     Minimal: Extracts a small, basic set of features.
15///     Efficient: Extracts a balanced set of features optimized for performance.
16///     Comprehensive: Extracts an exhaustive set of features.
17#[pyclass(from_py_object, name = "FeatureSetting", eq, eq_int)]
18#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord)]
19enum PyFeatureSetting {
20    Minimal,
21    Efficient,
22    Comprehensive,
23}

142/// Extract time-series features from a Polars LazyFrame.
143///
144/// This function computes features based on the provided settings. It evaluates
145/// the lazy computation graph and returns an in-memory DataFrame.
146///
147/// Args:
148///     lf (polars.LazyFrame): The input data to extract features from.
149///     settings (ExtractionSettings): The configuration controlling grouping and feature complexity.
150///     streaming (bool, optional): If True, executes the query using Polars' streaming engine
151///         for out-of-core processing. Defaults to False.
152///
153/// Returns:
154///     polars.DataFrame: A new DataFrame containing the grouped IDs and their extracted features.
155///
156/// Raises:
157///     Exception: If an underlying Polars error occurs during collection.
158#[pyfunction]
159#[pyo3(signature = (lf, settings, streaming=false))]
160fn extract_features(
161    lf: PyLazyFrame,
162    settings: PyExtractionSettings,
163    streaming: bool,
164) -> PyResult<PyDataFrame> {
165    let lf = lf.into();
166    let settings = settings.into();
167    let lf = if !streaming {
168        lazy_feature_df(lf, settings)?
169            .collect()
170            .map_err(ExtractionError::PolarsError)?
171    } else {
172        lazy_feature_df(lf, settings)?
173            .with_new_streaming(true)
174            .collect()
175            .map_err(ExtractionError::PolarsError)?
176    };
177    Ok(PyDataFrame(lf))
178}

180/// Time Series Feature Extraction module.
181///
182/// This module provides high-performance feature extraction capabilities for
183/// time series data, leveraging a Rust core and Polars dataframes.
184#[pymodule]
185fn tsfx(_py: Python, m: &Bound<PyModule>) -> PyResult<()> {
186    m.add_class::<PyFeatureSetting>()?;
187    m.add_class::<PyExtractionSettings>()?;
188    m.add_class::<PyDynamicGroupBySettings>()?;
189    m.add_function(wrap_pyfunction!(extract_features, m)?)?;
190    Ok(())
191}
