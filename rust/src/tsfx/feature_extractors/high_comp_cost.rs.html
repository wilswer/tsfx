5pub fn high_comp_cost_aggregators(value_cols: &[String]) -> Vec<Expr> {

13fn _into_subchunks(x: &Array1<f64>, chunk_size: usize) -> Vec<Array1<f64>> {
14    let mut subchunks = Vec::with_capacity(x.len());
15    for chunk in x.axis_windows(Axis(0), chunk_size) {
16        subchunks.push(chunk.to_owned());
17    }
18    subchunks
19}

21fn _get_matches(templates: Vec<Array1<f64>>, r: f64) -> usize {
22    let mut matches = 0;
23    for combo in templates.into_iter().combinations(2) {
24        let a = combo[0].to_owned();
25        let b = combo[1].to_owned();
26        let diff = a - b;
27        let dist_check = diff.mapv(|x| if x.abs() < r { 1 } else { 0 }).sum();
28        if dist_check == diff.len() {
29            matches += 1;
30        }
31    }
32    matches
33}

39fn _sample_entropy(s: Column) -> Result<Column, PolarsError> {
40    if s.is_empty() {
41        return Ok(Column::new("".into(), &[f64::NAN]));
42    }
43    let arr = s
44        .into_frame()
45        .to_ndarray::<Float64Type>(IndexOrder::C)
46        .unwrap();
47    let arr = arr
48        .remove_axis(Axis(1))
49        .into_dimensionality::<Ix1>()
50        .unwrap();
51    let m = 2;
52    let r = 0.2 * arr.std(1.0);
53    let templates_m = _into_subchunks(&arr, m);
54    let matches_m = _get_matches(templates_m, r);
55    let templates_m_plus_1 = _into_subchunks(&arr, m + 1);
56    let matches_m_plus_1 = _get_matches(templates_m_plus_1, r);
57    let out = ((matches_m as f64) / (matches_m_plus_1 as f64)).ln();
58    let s = Column::new("".into(), &[out]);
59    Ok(s)
60}
