8/// Coefficients of the polynomial $h$ of degree `m` (highest power first,
9/// as `np.polyfit`) fitted to the deterministic dynamics
10/// $\dot{x}(t) = h(x(t)) + \mathcal{N}(0, R)$ of a Langevin model
11/// (Friedrich et al., 2000), exactly as tsfresh's
12/// `_estimate_friedrich_coefficients` computes them:
13///
14/// 1. pair each value $x_t$ (all but the last) with the change
15///    $x_{t+1} - x_t$;
16/// 2. bin the values into `r` quantile bins, as `pd.qcut(x[:-1], r)`;
17/// 3. average values and changes per bin, as pandas' group-by mean;
18/// 4. fit the polynomial through the bin averages by least squares, as
19///    `np.polyfit(..., deg=m)`.
20///
21/// `None` where tsfresh returns NaN coefficients: duplicate quantile bin
22/// edges (`pd.qcut` raises), or a fit that fails (`np.linalg.lstsq` raises
23/// on non-finite input). Also `None` (tsfresh raises instead) for fewer than
24/// 2 values, `m = 0`, `r = 0`, or when every bin average is NaN.
25fn estimate_friedrich_coefficients(x: &[f64], m: usize, r: usize) -> Option<Vec<f64>> {
26    if x.len() < 2 || m == 0 || r == 0 {
27        return None;
28    }
29    let signal = &x[..x.len() - 1];
30    let delta: Vec<f64> = x.windows(2).map(|w| w[1] - w[0]).collect();
31
32    // pd.qcut: quantile levels rounded up where not exact, then the edges
33    // as np.quantile of the non-NaN values
34    let levels = numpy_linspace(0.0, 1.0, r + 1)
35        .into_iter()
36        .enumerate()
37        .map(|(i, q)| {
38            if r as f64 * q != i as f64 {
39                q.next_up().min(1.0)
40            } else {
41                q
42            }
43        });
44    let mut sorted: Vec<f64> = signal.iter().copied().filter(|v| !v.is_nan()).collect();
45    if sorted.is_empty() {
46        return None;
47    }
48    sorted.sort_unstable_by(f64::total_cmp);
49    let edges: Vec<f64> = levels.map(|q| numpy_quantile(&sorted, q)).collect();
50    // "Bin edges must be unique" (pandas lets two equal edges pass)
51    let same = |a: f64, b: f64| a == b || (a.is_nan() && b.is_nan());
52    if edges.len() != 2
53        && (0..edges.len()).any(|i| (i + 1..edges.len()).any(|j| same(edges[i], edges[j])))
54    {
55        return None;
56    }
57
58    // pd.qcut labels each bin with an interval of rounded edges; building
59    // them can fail, and bins with equal labels form one group
60    let group_of_bin = interval_label_groups(&edges)?;
61    let groups = group_of_bin.last().map_or(0, |&g| g + 1);
62
63    // Bin of each value (right-closed, the lowest edge included), then the
64    // group-by mean of values and changes per group, with pandas' Kahan sums
65    let ids = numpy_searchsorted_left(&edges, signal);
66    let mut sums = vec![[Kahan::default(), Kahan::default()]; groups];
67    let mut observed = vec![false; groups];
68    for (k, (&v, &id)) in signal.iter().zip(&ids).enumerate() {
69        let id = if v == edges[0] { 1 } else { id };
70        if v.is_nan() || id == 0 || id == edges.len() {
71            continue;
72        }
73        let group = group_of_bin[id - 1];
74        observed[group] = true;
75        sums[group][0].add(v);
76        sums[group][1].add(delta[k]);
77    }
78    let (xs, ys): (Vec<f64>, Vec<f64>) = sums
79        .iter()
80        .zip(&observed)
81        .filter(|&(_, &observed)| observed)
82        .map(|(sums, _)| (sums[0].mean(), sums[1].mean()))
83        .filter(|(x, y)| !x.is_nan() && !y.is_nan())
84        .unzip();
85    if xs.is_empty() {
86        return None;
87    }
88    polyfit(&xs, &ys, m)
89}

91/// The group of each quantile bin, as `pd.qcut` labels the bins: intervals
92/// between the edges rounded by `_round_frac` (to the smallest precision
93/// from 3 that keeps the edges distinct, else 3), the lowest edge lowered by
94/// `10 ** -precision`. Bins with equal intervals share a group. `None` where
95/// pandas fails to build the intervals (an edge that rounds to NaN, or the
96/// intervals not ordered).
97fn interval_label_groups(edges: &[f64]) -> Option<Vec<usize>> {
98    let same = |a: f64, b: f64| a == b || (a.is_nan() && b.is_nan());
99    let distinct = |levels: &[f64]| {
100        (0..levels.len()).all(|i| (i + 1..levels.len()).all(|j| !same(levels[i], levels[j])))
101    };
102    let round = |precision: i32| -> Vec<f64> {
103        edges
104            .iter()
105            .map(|&edge| round_frac(edge, precision))
106            .collect()
107    };
108    let precision = (3..20).find(|&p| distinct(&round(p))).unwrap_or(3);
109    let mut breaks = round(precision);
110    breaks[0] -= 10_f64.powf(-f64::from(precision));
111    // IntervalIndex.from_breaks validation
112    for w in breaks.windows(2) {
113        if w[0].is_nan() != w[1].is_nan() || w[0] > w[1] {
114            return None;
115        }
116    }
117    let mut groups = Vec::with_capacity(breaks.len() - 1);
118    for i in 0..breaks.len() - 1 {
119        // Intervals are ordered, so equal ones are neighbours
120        let group = match groups.last() {
121            Some(&g) if same(breaks[i - 1], breaks[i]) && same(breaks[i], breaks[i + 1]) => g,
122            Some(&g) => g + 1,
123            None => 0,
124        };
125        groups.push(group);
126    }
127    Some(groups)
128}

130/// pandas' `_round_frac`: round to `precision` digits after the decimal
131/// point, or for $|x| < 1$ to `precision` significant digits.
132fn round_frac(x: f64, precision: i32) -> f64 {
133    if !x.is_finite() || x == 0.0 {
134        return x;
135    }
136    let digits = if x.trunc() == 0.0 {
137        -(x.fract().abs().log10().floor() as i32) - 1 + precision
138    } else {
139        precision
140    };
141    numpy_around(x, digits)
142}

144/// `np.around(x, decimals)`: scale by a power of ten (computed by repeated
145/// multiplication, as numpy), round half to even, scale back.
146fn numpy_around(x: f64, decimals: i32) -> f64 {
147    let power_of_ten = |n: i32| {
148        const P10: [f64; 9] = [1e0, 1e1, 1e2, 1e3, 1e4, 1e5, 1e6, 1e7, 1e8];
149        if n < 9 {
150            P10[n as usize]
151        } else {
152            (9..n).fold(1e9, |p, _| p * 10.0)
153        }
154    };
155    if decimals >= 0 {
156        let f = power_of_ten(decimals);
157        (x * f).round_ties_even() / f
158    } else {
159        let f = power_of_ten(-decimals);
160        (x / f).round_ties_even() * f
161    }
162}

164/// pandas' group-by sum: Kahan summation skipping NaN, the compensation
165/// reset when it turns NaN (an infinite value).
166#[derive(Clone, Copy, Default)]
167struct Kahan {
168    sum: f64,
169    compensation: f64,
170    count: usize,
171}
172
173impl Kahan {
174    fn add(&mut self, value: f64) {
175        if value.is_nan() {
176            return;
177        }
178        self.count += 1;
179        let y = value - self.compensation;
180        let t = self.sum + y;
181        self.compensation = t - self.sum - y;
182        if self.compensation.is_nan() {
183            self.compensation = 0.0;
184        }
185        self.sum = t;
186    }
187
188    fn mean(&self) -> f64 {
189        if self.count == 0 {
190            f64::NAN
191        } else {
192            self.sum / self.count as f64
193        }
194    }
195}

197/// Least-squares polynomial of degree `deg` through `(x, y)`, highest power
198/// first, as `np.polyfit`: Vandermonde columns scaled to unit norm, solved
199/// by SVD with singular values up to `len(x) * eps` times the largest
200/// treated as zero (the minimum-norm solution when underdetermined). `None`
201/// for non-finite input, where `np.linalg.lstsq` raises.
202fn polyfit(x: &[f64], y: &[f64], deg: usize) -> Option<Vec<f64>> {
203    let (rows, order) = (x.len(), deg + 1);
204    // np.vander: powers by repeated multiplication, highest first
205    let mut lhs = DMatrix::from_fn(rows, order, |_, _| 1.0);
206    for i in 0..rows {
207        for j in (0..deg).rev() {
208            lhs[(i, j)] = lhs[(i, j + 1)] * x[i];
209        }
210    }
211    let scale: Vec<f64> = (0..order)
212        .map(|j| {
213            (0..rows)
214                .fold(0.0, |s, i| s + lhs[(i, j)] * lhs[(i, j)])
215                .sqrt()
216        })
217        .collect();
218    for j in 0..order {
219        for i in 0..rows {
220            lhs[(i, j)] /= scale[j];
221        }
222    }
223    if lhs.iter().chain(y).any(|v| !v.is_finite()) {
224        return None;
225    }
226    let rhs = DMatrix::from_column_slice(rows, 1, y);
227    let svd = lhs.clone().svd(true, true);
228    let largest = svd.singular_values.max();
229    let cutoff = rows as f64 * f64::EPSILON * largest;
230    let full_rank = rows >= order && svd.singular_values.iter().all(|&s| s > cutoff);
231    let solution = if full_rank {
232        // Householder QR: nalgebra's SVD loses digits on small singular
233        // values that LAPACK's keeps, QR does not
234        let qr = lhs.qr();
235        let qtb = qr.q().transpose() * &rhs;
236        qr.r().solve_upper_triangular(&qtb)?
237    } else {
238        svd.solve(&rhs, cutoff).ok()?
239    };
240    Some((0..order).map(|j| solution[(j, 0)] / scale[j]).collect())
241}

243/// The largest real part of the roots of the polynomial `p` (highest power
244/// first), as `np.max(np.real(np.roots(p)))`: leading zeros dropped,
245/// trailing zeros giving roots at 0, the rest as the eigenvalues of the
246/// companion matrix. NaN if there are no roots or the companion matrix is
247/// not finite (where numpy raises).
248fn max_real_root(p: &[f64]) -> f64 {
249    let Some(first) = p.iter().position(|&c| c != 0.0) else {
250        return f64::NAN;
251    };
252    let last = p.iter().rposition(|&c| c != 0.0).unwrap_or(first);
253    let trailing_zeros = p.len() - last - 1;
254    let p = &p[first..=last];
255    let mut real_parts = Vec::new();
256    if p.len() > 1 {
257        let n = p.len() - 1;
258        let mut companion = DMatrix::zeros(n, n);
259        for j in 0..n {
260            companion[(0, j)] = -p[j + 1] / p[0];
261        }
262        for i in 1..n {
263            companion[(i, i - 1)] = 1.0;
264        }
265        if companion.iter().any(|v: &f64| !v.is_finite()) {
266            return f64::NAN;
267        }
268        // Balanced as LAPACK's dgeev does (nalgebra does not), then the
269        // eigenvalues polished on the polynomial
270        balance(&mut companion);
271        real_parts.extend(
272            companion
273                .complex_eigenvalues()
274                .iter()
275                .map(|&z| polish_root(p, z).re),
276        );
277    }
278    real_parts.extend(std::iter::repeat_n(0.0, trailing_zeros));
279    if real_parts.is_empty() {
280        return f64::NAN;
281    }
282    // np.max: NaN if any is NaN
283    real_parts
284        .into_iter()
285        .reduce(|a, b| {
286            if a.is_nan() || b.is_nan() {
287                f64::NAN
288            } else {
289                a.max(b)
290            }
291        })
292        .unwrap_or(f64::NAN)
293}

295/// Balance a square matrix in place by a diagonal similarity transform with
296/// powers of 2 (Parlett & Reinsch), so that each row and its column have
297/// similar norms. The eigenvalues are unchanged but computed far more
298/// accurately for badly scaled matrices such as companion matrices.
299fn balance(a: &mut DMatrix<f64>) {
300    let n = a.nrows();
301    let mut converged = false;
302    while !converged {
303        converged = true;
304        for i in 0..n {
305            let mut c: f64 = (0..n).filter(|&j| j != i).map(|j| a[(j, i)].abs()).sum();
306            let r: f64 = (0..n).filter(|&j| j != i).map(|j| a[(i, j)].abs()).sum();
307            if c == 0.0 || r == 0.0 {
308                continue;
309            }
310            let s = c + r;
311            let mut f = 1.0;
312            while c < r / 2.0 {
313                f *= 2.0;
314                c *= 4.0;
315            }
316            while c >= r * 2.0 {
317                f /= 2.0;
318                c /= 4.0;
319            }
320            if (c + r) / f < 0.95 * s {
321                converged = false;
322                for j in 0..n {
323                    a[(i, j)] /= f;
324                    a[(j, i)] *= f;
325                }
326            }
327        }
328    }
329}

331/// Refine an approximate root `z` of the polynomial `p` (highest power
332/// first) by Newton steps, keeping a step only while it lowers $|p(z)|$.
333fn polish_root(p: &[f64], mut z: Complex<f64>) -> Complex<f64> {
334    let eval = |z: Complex<f64>| {
335        p.iter().fold(
336            (Complex::new(0.0, 0.0), Complex::new(0.0, 0.0)),
337            |(value, slope), &c| (value * z + c, slope * z + value),
338        )
339    };
340    let (mut value, mut slope) = eval(z);
341    for _ in 0..50 {
342        if value.norm() == 0.0 || slope.norm() == 0.0 {
343            break;
344        }
345        let step = value / slope;
346        let next = z - step;
347        let (next_value, next_slope) = eval(next);
348        if next_value.norm().is_nan() || next_value.norm() >= value.norm() {
349            break;
350        }
351        (z, value, slope) = (next, next_value, next_slope);
352        if step.norm() <= f64::EPSILON * z.norm() {
353            break;
354        }
355    }
356    z
357}

359/// Friedrich coefficients feature.
360///
361/// The coefficients of the polynomial $h(x) = \sum_{k=0}^{m} c_k\,x^{m-k}$
362/// fitted to the deterministic dynamics of the Langevin model
363/// $$ \dot{x}(t) = h(x(t)) + \mathcal{N}(0, R) $$
364/// (Friedrich et al., 2000): the values (all but the last) are binned into
365/// `r` quantile bins, the values and their changes $x_{t+1} - x_t$ are
366/// averaged per bin, and $h$ is the least-squares polynomial of degree `m`
367/// through the bin averages. Coefficient `coeff` is that of $x^{m - \text{coeff}}$
368/// (`np.polyfit` order). For short series the result depends strongly on
369/// the parameters.
370///
371/// # Parameters
372/// - `coeff`: index of the returned coefficient (0 is the highest power).
373/// - `m`: degree of the polynomial.
374/// - `r`: number of quantile bins.
375///
376/// Config:
377/// `[friedrich_coefficients] parameters = [{ coeff = 0, m = 3, r = 30 }, ...]`;
378/// one column per entry. The fit is done once per distinct `(m, r)`.
379///
380/// # Output column
381/// `{name}__friedrich_coefficients__coeff_{coeff}__m_{m}__r_{r}`
382///
383/// # Edge cases
384/// - Nulls are dropped first; a group with no non-null values gives NaN.
385/// - `coeff > m` gives NaN.
386/// - Duplicate quantile bin edges (for example a series with many equal
387///   values, or fewer distinct values than bins) give NaN, as tsfresh.
388/// - Non-finite bin averages ($\pm\infty$ in the series) give NaN, as
389///   tsfresh. NaN values are left out of the bins and the averages, as
390///   pandas does.
391/// - Fewer than 2 values, every bin average NaN, `m = 0` or `r = 0` give NaN
392///   (tsfresh raises).
393///
394/// # tsfresh
395/// `feature_calculators.friedrich_coefficients` (v0.21.2). The least-squares
396/// solution comes from a different SVD implementation than numpy's LAPACK,
397/// so results differ from tsfresh's in the last bits (more for
398/// ill-conditioned fits).
399pub struct FriedrichCoefficients {
400    /// `(coeff, m, r)` per config entry.
401    pub params: Vec<(usize, usize, usize)>,
402}
403
404impl Kernel for FriedrichCoefficients {
405    fn names(&self) -> Vec<String> {
406        self.params
407            .iter()
408            .map(|(coeff, m, r)| format!("friedrich_coefficients__coeff_{coeff}__m_{m}__r_{r}"))
409            .collect()
410    }
411
412    fn compute(&self, ctx: &GroupContext, out: &mut Vec<f64>) {
413        let x = ctx.values();
414        // Coefficients per distinct (m, r), fitted once
415        let mut fits: Vec<(usize, usize, Option<Vec<f64>>)> = Vec::new();
416        for &(coeff, m, r) in &self.params {
417            let fit = match fits.iter().position(|f| (f.0, f.1) == (m, r)) {
418                Some(i) => &fits[i].2,
419                None => {
420                    fits.push((m, r, estimate_friedrich_coefficients(x, m, r)));
421                    &fits[fits.len() - 1].2
422                }
423            };
424            out.push(
425                fit.as_ref()
426                    .and_then(|c| c.get(coeff).copied())
427                    .unwrap_or(f64::NAN),
428            );
429        }
430    }
431}

433/// Maximum Langevin fixed point feature.
434///
435/// The largest fixed point $\max \{x : h(x) = 0\}$ (largest real part of
436/// the roots) of the polynomial $h$ fitted to the deterministic dynamics of
437/// the Langevin model, as in [`FriedrichCoefficients`].
438///
439/// # Parameters
440/// - `m`: degree of the polynomial.
441/// - `r`: number of quantile bins.
442///
443/// Config: `[max_langevin_fixed_point] parameters = [{ m = 3, r = 30 }, ...]`;
444/// one column per entry.
445///
446/// # Output column
447/// `{name}__max_langevin_fixed_point__m_{m}__r_{r}`
448///
449/// # Edge cases
450/// - NaN wherever [`FriedrichCoefficients`] gives NaN coefficients.
451/// - A polynomial without roots (constant) gives NaN, as tsfresh.
452///
453/// # tsfresh
454/// `feature_calculators.max_langevin_fixed_point` (v0.21.2). Roots are the
455/// eigenvalues of the companion matrix, as `np.roots`, from a different
456/// eigenvalue solver and refined by Newton steps on the polynomial, so they
457/// agree with numpy's to about the last bits. When the fitted leading
458/// coefficient is (numerically) zero the largest root is ill-conditioned and
459/// can differ substantially from tsfresh's.
460pub struct MaxLangevinFixedPoint {
461    /// `(m, r)` per config entry.
462    pub params: Vec<(usize, usize)>,
463}
464
465impl Kernel for MaxLangevinFixedPoint {
466    fn names(&self) -> Vec<String> {
467        self.params
468            .iter()
469            .map(|(m, r)| format!("max_langevin_fixed_point__m_{m}__r_{r}"))
470            .collect()
471    }
472
473    fn compute(&self, ctx: &GroupContext, out: &mut Vec<f64>) {
474        let x = ctx.values();
475        for &(m, r) in &self.params {
476            out.push(
477                estimate_friedrich_coefficients(x, m, r)
478                    .map(|c| max_real_root(&c))
479                    .unwrap_or(f64::NAN),
480            );
481        }
482    }
483}

489    #[test]
490    fn test_max_real_root() {
491        // Badly scaled: (t + 1)(t^2 + 1) with t = x / a, roots -a and ±ia
492        let a = 774366.0_f64;
493        let p = [1.0 / (a * a * a), 1.0 / (a * a), 1.0 / a, 1.0];
494        assert!(max_real_root(&p).abs() < 1e-6, "{}", max_real_root(&p));
495        // (x - 1)(x - 2)(x + 3) = x^3 - 7x + 6
496        let root = max_real_root(&[1.0, 0.0, -7.0, 6.0]);
497        assert!((root - 2.0).abs() < 1e-12);
498        // trailing zero: x^2 - x = x (x - 1); leading zero dropped
499        assert!((max_real_root(&[0.0, 1.0, -1.0, 0.0]) - 1.0).abs() < 1e-12);
500        assert_eq!(max_real_root(&[0.0, 0.0, 2.0, 0.0]), 0.0);
501        assert!(max_real_root(&[0.0, 0.0, 0.0, 2.0]).is_nan());
502        assert!(max_real_root(&[0.0; 4]).is_nan());
