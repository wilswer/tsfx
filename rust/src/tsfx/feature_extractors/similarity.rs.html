8/// Below this, stumpy treats `σ_Q σ_T m` as this value
9/// (`config.STUMPY_DENOM_THRESHOLD`).
10const DENOM_THRESHOLD: f64 = 1e-14;
11
12/// One query of [`QuerySimilarityCount`].
13#[derive(Clone, Debug, PartialEq)]
14pub struct Query {
15    /// The query subsequence $Q$.
16    pub query: Vec<f64>,
17    /// Largest distance that counts as a match.
18    pub threshold: f64,
19    /// z-normalise $Q$ and each subsequence before comparing.
20    pub normalize: bool,
21}
22
23/// Query similarity count feature.
24///
25/// The number of length-$m$ subsequences of the series whose distance to
26/// the query $Q$ (of length $m$) is at most `threshold`. With `normalize`,
27/// the distance is the z-normalised Euclidean distance computed as
28/// `stumpy.core.mass` does,
29/// $$ d = \sqrt{\left|2m\left(1 - \min\left(1, \frac{Q \cdot T_i - m\,\mu_Q\,\mu_{T_i}}{\max(m\,\sigma_Q\,\sigma_{T_i}, 10^{-14})}\right)\right)\right|}, $$
30/// with population standard deviations, $d = 0$ if both $Q$ and the
31/// subsequence $T_i$ are constant and $d = \sqrt{m}$ if exactly one is;
32/// without, it is the plain Euclidean distance $\lVert Q - T_i \rVert_2$.
33///
34/// # Parameters
35/// - `query`: the query subsequence, at least 3 values.
36/// - `threshold`: largest distance that counts (default 0, exact matches).
37/// - `normalize`: z-normalised distance (default `true`).
38///
39/// Config:
40/// `[query_similarity_count] parameters = [{ query = [1.0, 2.0, 3.0], threshold = 0.5, normalize = true }, ...]`;
41/// one column per entry. Off by default: the feature needs a query, and
42/// tsfresh's default (no query) only gives NaN.
43///
44/// # Output column
45/// `{name}__query_similarity_count__query_{i}`, where `i` is the 0-based
46/// position of the entry in the config's `parameters` list.
47///
48/// # Edge cases
49/// - Nulls are dropped first; a group with no non-null values gives NaN.
50/// - A query with fewer than 3 values gives NaN.
51/// - A series shorter than the query gives NaN (tsfresh raises an error).
52/// - A query with NaN or $\pm\infty$ matches nothing; a subsequence with NaN
53///   or $\pm\infty$ is at infinite distance.
54/// - The default threshold of 0 asks for exact matches; tsfresh computes
55///   the dot products with an FFT and `fastmath`, so its distance of an
56///   exact match is often a tiny positive number instead of 0. tsfx
57///   computes them directly, so a count at a threshold within rounding of
58///   a distance can differ from tsfresh's.
59///
60/// # Cost
61/// $O(n\,m)$ per query.
62///
63/// # tsfresh
64/// `feature_calculators.query_similarity_count` (v0.21.2), with stumpy's
65/// `core.mass` / `core.mass_absolute`.
66pub struct QuerySimilarityCount {
67    pub queries: Vec<Query>,
68}
69
70impl Kernel for QuerySimilarityCount {
71    fn names(&self) -> Vec<String> {
72        (0..self.queries.len())
73            .map(|i| format!("query_similarity_count__query_{i}"))
74            .collect()
75    }
76
77    fn compute(&self, ctx: &GroupContext, out: &mut Vec<f64>) {
78        let t = ctx.values();
79        for query in &self.queries {
80            let q = query.query.as_slice();
81            let m = q.len();
82            if m < 3 || t.len() < m {
83                out.push(f64::NAN);
84                continue;
85            }
86            let count = if q.iter().all(|v| v.is_finite()) {
87                let distance = if query.normalize {
88                    z_normalized_distances(q, t)
89                } else {
90                    euclidean_distances(q, t)
91                };
92                distance.iter().filter(|&&d| d <= query.threshold).count()
93            } else {
94                // stumpy: every distance to a non-finite query is infinite
95                if f64::INFINITY <= query.threshold {
96                    t.len() - m + 1
97                } else {
98                    0
99                }
100            };
101            out.push(count as f64);
102        }
103    }
104}

106/// z-normalised Euclidean distance of the finite query `q` to every
107/// length-`q.len()` subsequence of `t`, as `stumpy.core.mass`.
108fn z_normalized_distances(q: &[f64], t: &[f64]) -> Vec<f64> {
109    let m = q.len();
110    let mf = m as f64;
111    let mean_q = numpy_mean(q);
112    let std_q = population_var(&ArrayView1::from(q)).sqrt();
113    let (min_q, max_q) = extremes(q.iter().copied());
114    let q_constant = max_q - min_q == 0.0;
115    t.windows(m)
116        .map(|w| {
117            if !w.iter().all(|v| v.is_finite()) {
118                return f64::INFINITY;
119            }
120            let (min_t, max_t) = extremes(w.iter().copied());
121            let t_constant = max_t - min_t == 0.0;
122            let d_squared = if q_constant && t_constant {
123                0.0
124            } else if q_constant || t_constant {
125                mf
126            } else {
127                let mean_t = numpy_mean(w);
128                let std_t = population_var(&ArrayView1::from(w)).sqrt();
129                let qt = q.iter().zip(w).map(|(a, b)| a * b).sum::<f64>();
130                let denom = (std_q * std_t * mf).max(DENOM_THRESHOLD);
131                let rho = ((qt - mean_q * mean_t * mf) / denom).min(1.0);
132                (2.0 * mf * (1.0 - rho)).abs()
133            };
134            d_squared.sqrt()
135        })
136        .collect()
137}
138
139/// Euclidean distance of the finite query `q` to every length-`q.len()`
140/// subsequence of `t`, as `stumpy.core.mass_absolute`.
141fn euclidean_distances(q: &[f64], t: &[f64]) -> Vec<f64> {
142    t.windows(q.len())
143        .map(|w| {
144            if !w.iter().all(|v| v.is_finite()) {
145                return f64::INFINITY;
146            }
147            q.iter()
148                .zip(w)
149                .map(|(a, b)| (a - b) * (a - b))
150                .sum::<f64>()
151                .sqrt()
152        })
153        .collect()
154}

156#[cfg(test)]
157mod tests {
158    use super::*;
159
160    #[test]
161    fn test_exact_match_is_zero() {
162        let q = [1.0, 3.0, 2.0];
163        let t = [0.0, 1.0, 3.0, 2.0, 5.0, 2.0, 6.0, 4.0];
164        let d = z_normalized_distances(&q, &t);
165        assert!(d[1] < 1e-7);
166        // [2, 6, 4] is [1, 3, 2] scaled by 2: z-normalised, identical
167        assert!(d[5] < 1e-7);
168        assert_eq!(euclidean_distances(&q, &t)[1], 0.0);
169    }
170
171    #[test]
172    fn test_constant_subsequences() {
173        let q = [1.0, 1.0, 1.0];
174        let t = [2.0, 2.0, 2.0, 3.0];
175        let d = z_normalized_distances(&q, &t);
176        assert_eq!(d, vec![0.0, 3.0_f64.sqrt()]);
177    }
