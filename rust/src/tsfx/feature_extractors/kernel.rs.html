1//! All features of a value column computed in one pass over each group.
2//!
3//! Polars collects a group's values into a list once per `apply`, and with
4//! dozens of features that dominated extraction time. Instead, one `apply`
5//! per value column builds a [`GroupContext`] (the group's values plus
6//! statistics that several features share, each computed at most once) and
7//! runs every [`Kernel`] on it, returning all of the group's features as one
8//! list. After the group-by, [`KernelPlan::unpack`] turns list position `i`
9//! into the named feature column `i`.
10

20/// One group's values (nulls dropped) and the statistics features share.
21///
22/// Shared statistics are computed at most once, on first use; mean and
23/// variance exactly as numpy computes them.
24pub struct GroupContext<'a> {
25    values: &'a [f64],
26    has_nulls: bool,
27    x: ArrayView1<'a, f64>,
28    mean: OnceCell<f64>,
29    var: OnceCell<f64>,
30    extremes: OnceCell<(f64, f64)>,
31    sorted: OnceCell<Vec<f64>>,
32}
33
34impl<'a> GroupContext<'a> {
35    pub fn new(values: &'a [f64], has_nulls: bool) -> Self {
36        Self {
37            values,
38            has_nulls,
39            x: ArrayView1::from(values),
40            mean: OnceCell::new(),
41            var: OnceCell::new(),
42            extremes: OnceCell::new(),
43            sorted: OnceCell::new(),
44        }
45    }
46
47    /// The group's values, nulls dropped. Never empty.
48    pub fn x(&self) -> &ArrayView1<'a, f64> {
49        &self.x
50    }
51
52    /// The same values as a slice.
53    pub fn values(&self) -> &'a [f64] {
54        self.values
55    }
56
57    /// Whether the group had nulls before they were dropped.
58    pub fn has_nulls(&self) -> bool {
59        self.has_nulls
60    }
61
62    pub fn len(&self) -> usize {
63        self.x.len()
64    }
65
66    pub fn is_empty(&self) -> bool {
67        self.x.is_empty()
68    }
69
70    /// Arithmetic mean, exactly as `np.mean` ([`numpy_mean`]); NaN if any
71    /// value is NaN.
72    pub fn mean(&self) -> f64 {
73        *self.mean.get_or_init(|| numpy_mean(self.values))
74    }
75
76    /// Population variance (`ddof = 0`), see [`population_var`].
77    pub fn var(&self) -> f64 {
78        *self.var.get_or_init(|| population_var(&self.x))
79    }
80
81    /// Population standard deviation (`ddof = 0`): the square root of
82    /// [`Self::var`], exactly as `np.std` computes it.
83    pub fn std(&self) -> f64 {
84        self.var().sqrt()
85    }
86
87    /// Smallest value; NaN if any value is NaN (like `np.min`).
88    pub fn min(&self) -> f64 {
89        self.extremes().0
90    }
91
92    /// Largest value; NaN if any value is NaN (like `np.max`).
93    pub fn max(&self) -> f64 {
94        self.extremes().1
95    }
96
97    /// Minimum and maximum in one pass, with exactly the semantics of
98    /// `ndarray_stats`' `min`/`max` (both NaN if any value is NaN; on ties
99    /// the first value wins, which matters only for `-0.0` versus `0.0`).
100    fn extremes(&self) -> (f64, f64) {
101        *self
102            .extremes
103            .get_or_init(|| extremes(self.values.iter().copied()))
104    }
105
106    /// The values sorted ascending (`total_cmp`, so NaN sorts last).
107    pub fn sorted(&self) -> &[f64] {
108        self.sorted.get_or_init(|| {
109            let mut sorted = self.values.to_vec();
110            sorted.sort_unstable_by(f64::total_cmp);
111            sorted
112        })
113    }
114}

116/// Minimum and maximum of a non-empty sequence in one pass: both NaN if any
117/// value is NaN; on ties the first value wins.
118pub(crate) fn extremes(values: impl IntoIterator<Item = f64>) -> (f64, f64) {
119    let mut values = values.into_iter();
120    let Some(first) = values.next() else {
121        return (f64::NAN, f64::NAN);
122    };
123    if first.is_nan() {
124        return (f64::NAN, f64::NAN);
125    }
126    let (mut min, mut max) = (first, first);
127    for v in values {
128        if v.is_nan() {
129            return (f64::NAN, f64::NAN);
130        }
131        if v < min {
132            min = v;
133        } else if v > max {
134            max = v;
135        }
136    }
137    (min, max)
138}

140/// A feature (or a family of features) computed from a [`GroupContext`].
141pub trait Kernel: Send + Sync {
142    /// Output names without the `{column}__` prefix, in the order
143    /// [`Kernel::compute`] pushes the values.
144    fn names(&self) -> Vec<String>;
145
146    /// Push one value per name onto `out`.
147    fn compute(&self, ctx: &GroupContext, out: &mut Vec<f64>);
148}
149
150/// Define a [`Kernel`] with a single output: a unit struct (carrying the
151/// doc comment), its output name and how to compute it from the context.
152macro_rules! scalar_kernel {
153    ($(#[$meta:meta])* $kernel:ident, $name:literal, |$ctx:ident| $body:expr) => {
154        $(#[$meta])*
155        pub struct $kernel;
156
157        impl $crate::feature_extractors::kernel::Kernel for $kernel {
158            fn names(&self) -> Vec<String> {
159                vec![$name.to_string()]
160            }
161
162            fn compute(
163                &self,
164                $ctx: &$crate::feature_extractors::kernel::GroupContext,
165                out: &mut Vec<f64>,
166            ) {
167                out.push($body);
168            }
169        }
170    };
171}
172pub(crate) use scalar_kernel;

174/// The kernels of one value column, run by a single `apply` per group.
175#[derive(Default)]
176pub struct KernelPlan {
177    kernels: Vec<Box<dyn Kernel>>,
178    names: Vec<String>,
179}
180
181impl KernelPlan {
182    pub fn push(&mut self, kernel: impl Kernel + 'static) {
183        self.names.extend(kernel.names());
184        self.kernels.push(Box::new(kernel));
185    }
186
187    pub fn is_empty(&self) -> bool {
188        self.kernels.is_empty()
189    }
190
191    /// Every feature of one group; NaN for all of them if the group has no
192    /// non-null values.
193    fn run(&self, s: Column) -> PolarsResult<Column> {
194        let has_nulls = s.null_count() > 0;
195        let s = s.drop_nulls();
196        let mut out = Vec::with_capacity(self.names.len());
197        if s.is_empty() {
198            out.resize(self.names.len(), f64::NAN);
199        } else {
200            let ca = f64_values(&s)?;
201            let ctx = GroupContext::new(ca.cont_slice()?, has_nulls);
202            for kernel in &self.kernels {
203                kernel.compute(&ctx, &mut out);
204            }
205        }
206        polars_ensure!(
207            out.len() == self.names.len(),
208            ComputeError: "feature kernels produced {} values for {} names",
209            out.len(),
210            self.names.len()
211        );
212        Ok(Column::new(PlSmallStr::EMPTY, out))
213    }
214
215    /// The name of the list column that holds `column`'s features until
216    /// [`Self::unpack`].
217    pub fn packed_name(column: &str) -> String {
218        format!("__tsfx_features__{column}")
219    }
220
221    /// The aggregation: one list of all features per group.
222    pub fn aggregation(self: &Arc<Self>, column: &str) -> Expr {
223        let plan = Arc::clone(self);
224        col(column)
225            .apply(
226                move |s| plan.run(s),
227                |_, _| Ok(Field::new(PlSmallStr::EMPTY, DataType::Float64)),
228            )
229            .alias(Self::packed_name(column))
230    }
231
232    /// Replace `column`'s packed list column in the aggregated `df` with one
233    /// named column per feature (`{column}__{name}`). Done eagerly on the
234    /// small aggregated frame: a lazy `list.get` per feature would plan a
235    /// second query on every call.
236    pub fn unpack(&self, df: &mut DataFrame, column: &str) -> PolarsResult<()> {
237        let packed = df.drop_in_place(&Self::packed_name(column))?;
238        let list = packed.list()?.rechunk();
239        let inner = list.get_inner();
240        let inner = inner.f64()?.rechunk();
241        let flat = inner.cont_slice()?;
242        let n = self.names.len();
243        let mut columns: Vec<Vec<f64>> = vec![Vec::with_capacity(list.len()); n];
244        for arr in list.downcast_iter() {
245            for row in arr.offsets().buffer().windows(2) {
246                let values = &flat[row[0] as usize..row[1] as usize];
247                polars_ensure!(
248                    values.len() == n,
249                    ComputeError: "a group produced {} feature values for {} names",
250                    values.len(),
251                    n
252                );
253                for (column, &value) in columns.iter_mut().zip(values) {
254                    column.push(value);
255                }
256            }
257        }
258        let columns = self
259            .names
260            .iter()
261            .zip(columns)
262            .map(|(name, values)| Column::new(format!("{column}__{name}").into(), values))
263            .collect::<Vec<_>>();
264        df.hstack_mut(&columns)?;
265        Ok(())
266    }
267}
