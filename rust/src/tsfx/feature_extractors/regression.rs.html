6/// An ordinary least squares fit, as statsmodels' `OLS(y, X).fit()` (the
7/// default `method="pinv"`).
8struct Ols {
9    params: Vec<f64>,
10    /// Sum of squared residuals.
11    ssr: f64,
12    rank: usize,
13    nobs: usize,
14    /// Diagonal of $(X^T X)^+$, statsmodels' `normalized_cov_params`.
15    normalized_cov_diag: Vec<f64>,
16}
17
18/// Residuals at most this fraction of the scale of the fitted terms count as
19/// an exact fit (see [`Ols::fit`]).
20const PERFECT_FIT_TOL: f64 = 1e-10;
21
22/// Fits whose column-scaled design has a condition number bound below this
23/// are solved from the Gram matrix (see [`Ols::fit_prefixes_gram`]).
24const MAX_GRAM_CONDITION: f64 = 1e3;
25
26/// The largest estimated relative error of a residual sum of squares the
27/// Gram matrix path accepts. That residual is computed as a difference,
28/// $y^T y - \lVert c \rVert^2$, whose rounding error is about
29/// $\varepsilon\,\kappa^2\,y^T y$ for a (scaled) condition number
30/// $\kappa$; t-values and information criteria magnify it when few degrees
31/// of freedom remain.
32const MAX_GRAM_RESIDUAL_ERROR: f64 = 1e-12;
33
34/// Rows per block when accumulating the Gram matrix, so that a block of every
35/// column stays in cache.
36const GRAM_BLOCK_ROWS: usize = 512;
37
38/// Information criteria closer than this (relative to their magnitude, at
39/// least 1) count as equal when choosing the ADF lag.
40const IC_TIE_TOL: f64 = 1e-9;
41
42impl Ols {
43    /// Fit `y` against the columns of `x`, see [`Ols::fit_prefixes`].
44    fn fit(x: &[Vec<f64>], y: &[f64]) -> Option<Self> {
45        Self::fit_prefixes(x, y, x.len()).and_then(|mut fits| fits.pop())
46    }

48    /// Fit `y` against the first `p` columns of `x`, for every `p` from
49    /// `min_cols` to all of them (sharing one QR decomposition, as the first
50    /// `p` columns of a QR decomposition are one of the first `p` columns).
51    ///
52    /// Each fit is statsmodels' pseudoinverse solution: singular values
53    /// below $10^{-15}$ times the largest are dropped, and the rank counts
54    /// singular values above $\max(s) \cdot p \cdot \varepsilon$
55    /// (`np.linalg.matrix_rank` of the diagonal matrix of singular values).
56    /// Like LAPACK's SVD (which reduces a tall matrix by QR first), the
57    /// Householder QR followed by a Jacobi SVD of the triangular factor
58    /// resolves exactly collinear columns (a constant stretch of the series)
59    /// to negligible singular values, so the minimum-norm solution is found.
60    ///
61    /// A fit whose root mean squared residual is at most
62    /// [`PERFECT_FIT_TOL`] times the scale of the terms (largest
63    /// $|y_i| + \sum_j |X_{ij}\,\beta_j|$) is exact up to rounding, and its
64    /// residual sum of squares is set to 0: the rounding noise would
65    /// otherwise decide t-values and information criteria.
66    ///
67    /// Well-conditioned fits that are far from exact take a faster path, see
68    /// [`Ols::fit_prefixes_gram`]; its results agree with this one's to
69    /// within rounding, not bit for bit.
70    ///
71    /// `None` if the design is not finite (numpy's SVD raises
72    /// `LinAlgError`); a non-finite `y` propagates into the results.
73    fn fit_prefixes(x: &[Vec<f64>], y: &[f64], min_cols: usize) -> Option<Vec<Self>> {
74        if x.iter().flatten().any(|v| !v.is_finite()) {
75            return None;
76        }
77        if let Some(fits) = Self::fit_prefixes_gram(x, y, min_cols) {
78            return Some(fits);
79        }
80        let (nobs, k) = (y.len(), x.len());
81        let (r, c) = householder_qr(x, y);
82        // tails[p] = |c[p..]|^2, the part of y no combination of the first
83        // p columns can reach
84        let mut tails = vec![0.0; k + 1];
85        tails[k] = c[k.min(nobs)..].iter().map(|v| v * v).sum();
86        for p in (0..k).rev() {
87            tails[p] = tails[p + 1] + if p < nobs { c[p] * c[p] } else { 0.0 };
88        }
89        // For a cheap upper bound of each fit's scale (see below)
90        let abs_max = |v: &[f64]| v.iter().fold(0.0, |m: f64, e| m.max(e.abs()));
91        let y_max = abs_max(y);
92        let col_max: Vec<f64> = x.iter().map(|col| abs_max(col)).collect();
93        let fits = (min_cols..=k)
94            .map(|p| {
95                // The p x p triangular factor of the first p columns
96                let r_p: Vec<Vec<f64>> = r[..p].iter().map(|col| col[..p].to_vec()).collect();
97                let (u, s, v) = jacobi_svd(r_p.clone());
98                let s_max = s.iter().copied().fold(0.0, f64::max);
99                let cutoff = 1e-15 * s_max;
100                // pinv = V diag(1/s) U^T with unit columns u_j = U_j / s_j,
101                // so beta = sum_j v_j (U_j . c) / s_j^2
102                let inv_sq: Vec<f64> = s
103                    .iter()
104                    .map(|&sj| if sj > cutoff { 1.0 / (sj * sj) } else { 0.0 })
105                    .collect();
106                let mut params = vec![0.0; p];
107                for j in 0..p {
108                    if inv_sq[j] == 0.0 {
109                        continue;
110                    }
111                    let uc = u[j].iter().zip(&c).map(|(a, b)| a * b).sum::<f64>() * inv_sq[j];
112                    for (b, vij) in params.iter_mut().zip(&v[j]) {
113                        *b += vij * uc;
114                    }
115                }
116                let normalized_cov_diag = (0..p)
117                    .map(|i| (0..p).map(|j| v[j][i] * v[j][i] * inv_sq[j]).sum())
118                    .collect();
119                // Residual: the unreachable tail plus what the (possibly
120                // rank-deficient) solution misses of the reachable part
121                let mut ssr = tails[p];
122                for i in 0..p.min(nobs) {
123                    let fitted: f64 = (i..p).map(|j| r_p[j][i] * params[j]).sum();
124                    ssr += (c[i] - fitted) * (c[i] - fitted);
125                }
126                let rms = (ssr / nobs as f64).sqrt();
127                // The scale is at most max |y| + sum_j |b_j| max |X_j| (twice
128                // that covers rounding), so the O(nobs p) scale is only
129                // needed when the residual is below the tolerance of that
130                let bound = y_max
131                    + params
132                        .iter()
133                        .zip(&col_max)
134                        .map(|(b, m)| b.abs() * m)
135                        .sum::<f64>();
136                if !bound.is_finite() || rms <= 2.0 * PERFECT_FIT_TOL * bound {
137                    let scale = (0..nobs)
138                        .map(|row| {
139                            y[row].abs()
140                                + x[..p]
141                                    .iter()
142                                    .zip(&params)
143                                    .map(|(col, b)| (col[row] * b).abs())
144                                    .sum::<f64>()
145                        })
146                        .fold(0.0, f64::max);
147                    if rms <= PERFECT_FIT_TOL * scale {
148                        ssr = 0.0;
149                    }
150                }
151                let tol = s_max * p as f64 * f64::EPSILON;
152                Self {
153                    params,
154                    ssr,
155                    rank: s.iter().filter(|&&v| v > tol).count(),
156                    nobs,
157                    normalized_cov_diag,
158                }
159            })
160            .collect();
161        Some(fits)
162    }

164    /// [`Ols::fit_prefixes`] from the Gram matrix $X^T X$ and $X^T y$, built
165    /// in one pass over the rows, and its Cholesky factor $R$ ($R^T R = X^T
166    /// X$, the same $R$ as a QR decomposition up to signs). A Householder QR
167    /// sweeps the whole design once per column, which on long series is
168    /// limited by memory bandwidth; this reads it once.
169    ///
170    /// The columns are scaled to unit norm first. Squaring the design squares
171    /// its condition number, so this path is only taken when every fit is
172    /// well conditioned (scaled condition number bound below
173    /// [`MAX_GRAM_CONDITION`]: results within about $10^{6}\,\varepsilon$
174    /// relative of the QR path's), leaves a residual large enough that its
175    /// estimated relative error stays below [`MAX_GRAM_RESIDUAL_ERROR`] and is
176    /// clearly not an exact fit. `None` otherwise, for the QR path.
177    fn fit_prefixes_gram(x: &[Vec<f64>], y: &[f64], min_cols: usize) -> Option<Vec<Self>> {
178        let (nobs, k) = (y.len(), x.len());
179        if k == 0 {
180            return None;
181        }
182        let mut gram = vec![vec![0.0; k]; k];
183        let mut xty = vec![0.0; k];
184        let mut yty = 0.0;
185        for start in (0..nobs).step_by(GRAM_BLOCK_ROWS) {
186            let rows = start..(start + GRAM_BLOCK_ROWS).min(nobs);
187            let y_block = &y[rows.clone()];
188            for a in 0..k {
189                let xa = &x[a][rows.clone()];
190                for b in a..k {
191                    gram[a][b] += dot(xa, &x[b][rows.clone()]);
192                }
193                xty[a] += dot(xa, y_block);
194            }
195            yty += dot(y_block, y_block);
196        }
197        if !yty.is_finite() || xty.iter().any(|v| !v.is_finite()) {
198            return None;
199        }
200        // Unit columns: gram[a][b] / (norm_a norm_b), so the diagonal is 1
201        let norms: Vec<f64> = (0..k).map(|a| gram[a][a].sqrt()).collect();
202        if norms.contains(&0.0) {
203            return None;
204        }
205        // Upper triangular Cholesky factor, by rows: r[i][j] for j >= i
206        let mut r = vec![vec![0.0; k]; k];
207        for j in 0..k {
208            for i in 0..=j {
209                let g = gram[i][j] / (norms[i] * norms[j]);
210                let s: f64 = (0..i).map(|l| r[l][i] * r[l][j]).sum();
211                if i == j {
212                    let pivot = g - s;
213                    if pivot.is_nan() || pivot <= 0.0 {
214                        return None;
215                    }
216                    r[j][j] = pivot.sqrt();
217                } else {
218                    r[i][j] = (g - s) / r[i][i];
219                }
220            }
221        }
222        // R^{-1} (upper triangular) by back substitution, column by column;
223        // |R|_F^2 = k for unit columns, so the condition bound is
224        // sqrt(k) |R^{-1}|_F
225        let mut inv = vec![vec![0.0; k]; k];
226        let mut inv_sq = 0.0;
227        for j in 0..k {
228            inv[j][j] = 1.0 / r[j][j];
229            for i in (0..j).rev() {
230                let s: f64 = (i + 1..=j).map(|l| r[i][l] * inv[l][j]).sum();
231                inv[i][j] = -s / r[i][i];
232            }
233            inv_sq += (0..=j).map(|i| inv[i][j] * inv[i][j]).sum::<f64>();
234        }
235        let bound_sq = k as f64 * inv_sq;
236        if bound_sq.is_nan() || bound_sq >= MAX_GRAM_CONDITION * MAX_GRAM_CONDITION {
237            return None;
238        }
239        // c = R^{-T} (scaled X^T y): the first p entries are Q^T y for the
240        // first p columns
241        let mut c = vec![0.0; k];
242        for j in 0..k {
243            let s: f64 = (0..j).map(|i| r[i][j] * c[i]).sum();
244            c[j] = (xty[j] / norms[j] - s) / r[j][j];
245        }
246        let y_max = y.iter().fold(0.0, |m: f64, e| m.max(e.abs()));
247        let col_max: Vec<f64> = x
248            .iter()
249            .map(|col| col.iter().fold(0.0, |m: f64, e| m.max(e.abs())))
250            .collect();
251        let mut reached = (0..min_cols).map(|i| c[i] * c[i]).sum::<f64>();
252        let mut fits = Vec::with_capacity(k + 1 - min_cols);
253        for p in min_cols..=k {
254            if p > min_cols {
255                reached += c[p - 1] * c[p - 1];
256            }
257            let ssr = yty - reached;
258            // bound_sq bounds the squared condition number
259            if ssr.is_nan() || ssr * MAX_GRAM_RESIDUAL_ERROR < f64::EPSILON * bound_sq * yty {
260                return None;
261            }
262            // beta = R_p^{-1} c, diag((X^T X)^{-1}) = squared rows of R_p^{-1},
263            // both scaled back by the column norms
264            let params: Vec<f64> = (0..p)
265                .map(|i| (i..p).map(|j| inv[i][j] * c[j]).sum::<f64>() / norms[i])
266                .collect();
267            let normalized_cov_diag = (0..p)
268                .map(|i| (i..p).map(|j| inv[i][j] * inv[i][j]).sum::<f64>() / (norms[i] * norms[i]))
269                .collect();
270            // Could this be an exact fit (see fit_prefixes)? Then the QR path
271            // decides
272            let bound = y_max
273                + params
274                    .iter()
275                    .zip(&col_max)
276                    .map(|(b, m)| b.abs() * m)
277                    .sum::<f64>();
278            if (ssr / nobs as f64).sqrt() <= 2.0 * PERFECT_FIT_TOL * bound || !bound.is_finite() {
279                return None;
280            }
281            fits.push(Self {
282                params,
283                ssr,
284                rank: p,
285                nobs,
286                normalized_cov_diag,
287            });
288        }
289        Some(fits)
290    }

292    /// The t-value of parameter `i`: the estimate over its standard error,
293    /// with the error variance estimated as `ssr / (nobs - rank)`.
294    fn tvalue(&self, i: usize) -> f64 {
295        let scale = self.ssr / (self.nobs - self.rank) as f64;
296        self.params[i] / (self.normalized_cov_diag[i] * scale).sqrt()
297    }
298
299    /// The Gaussian log-likelihood at the estimate (statsmodels' `llf`).
300    fn llf(&self) -> f64 {
301        let nobs2 = self.nobs as f64 / 2.0;
302        -nobs2 * (2.0 * std::f64::consts::PI).ln()
303            - nobs2 * (self.ssr / self.nobs as f64).ln()
304            - nobs2
305    }
306
307    /// Akaike's information criterion, counting `rank` parameters.
308    fn aic(&self) -> f64 {
309        -2.0 * self.llf() + 2.0 * self.rank as f64
310    }
311
312    /// Bayes' information criterion, counting `rank` parameters.
313    fn bic(&self) -> f64 {
314        -2.0 * self.llf() + (self.nobs as f64).ln() * self.rank as f64
315    }
316}

318/// Dot product with eight independent accumulators, so it vectorises (the
319/// sum is rounded differently from a sequential one).
320fn dot(a: &[f64], b: &[f64]) -> f64 {
321    let (a_lanes, a_rest) = a.as_chunks::<8>();
322    let (b_lanes, b_rest) = b.as_chunks::<8>();
323    let mut acc = [0.0; 8];
324    for (u, v) in a_lanes.iter().zip(b_lanes) {
325        for l in 0..8 {
326            acc[l] += u[l] * v[l];
327        }
328    }
329    let rest: f64 = a_rest.iter().zip(b_rest).map(|(u, v)| u * v).sum();
330    acc.iter().sum::<f64>() + rest
331}

333/// Householder QR decomposition $X = QR$ of the matrix with columns `x`
334/// (at least as many rows as columns). Returns the columns of the upper
335/// triangular $R$ (each of length $k$) and $Q^T y$.
336fn householder_qr(x: &[Vec<f64>], y: &[f64]) -> (Vec<Vec<f64>>, Vec<f64>) {
337    let (n, k) = (y.len(), x.len());
338    debug_assert!(n >= k);
339    let mut a: Vec<Vec<f64>> = x.to_vec();
340    let mut c = y.to_vec();
341    for j in 0..k {
342        let norm = a[j][j..].iter().map(|v| v * v).sum::<f64>().sqrt();
343        if norm == 0.0 {
344            continue;
345        }
346        // Reflect a[j][j..] onto alpha e_1, alpha of opposite sign to the
347        // leading entry to avoid cancellation
348        let alpha = -norm.copysign(a[j][j]);
349        let mut v = a[j][j..].to_vec();
350        v[0] -= alpha;
351        let v_norm_sq: f64 = v.iter().map(|e| e * e).sum();
352        let reflect = |col: &mut [f64]| {
353            let s = 2.0 * v.iter().zip(col.iter()).map(|(a, b)| a * b).sum::<f64>() / v_norm_sq;
354            for (e, vi) in col.iter_mut().zip(&v) {
355                *e -= s * vi;
356            }
357        };
358        for col in a[j + 1..].iter_mut() {
359            reflect(&mut col[j..]);
360        }
361        reflect(&mut c[j..]);
362        a[j][j] = alpha;
363        a[j][j + 1..].iter_mut().for_each(|e| *e = 0.0);
364    }
365    for col in a.iter_mut() {
366        col.truncate(k);
367    }
368    (a, c)
369}

371/// One-sided Jacobi (Hestenes) SVD of the matrix with columns `a`: rotate
372/// pairs of columns until all are orthogonal, applying the same rotations
373/// to the identity. Returns the rotated columns $U_j = s_j u_j$ (not
374/// normalised), the singular values $s_j = \lVert U_j \rVert$ (unsorted) and
375/// the right singular vectors $v_j$ as columns, so that $A = \sum_j U_j v_j^T$.
376fn jacobi_svd(mut a: Vec<Vec<f64>>) -> (Vec<Vec<f64>>, Vec<f64>, Vec<Vec<f64>>) {
377    let k = a.len();
378    let mut v: Vec<Vec<f64>> = (0..k)
379        .map(|j| (0..k).map(|i| if i == j { 1.0 } else { 0.0 }).collect())
380        .collect();
381    let dot = |x: &[f64], y: &[f64]| x.iter().zip(y).map(|(a, b)| a * b).sum::<f64>();
382    let rotate = |cols: &mut [Vec<f64>], p: usize, q: usize, c: f64, s: f64| {
383        let (left, right) = cols.split_at_mut(q);
384        for (xp, xq) in left[p].iter_mut().zip(right[0].iter_mut()) {
385            let (vp, vq) = (*xp, *xq);
386            *xp = c * vp - s * vq;
387            *xq = s * vp + c * vq;
388        }
389    };
390    for _sweep in 0..100 {
391        let mut rotated = false;
392        for p in 0..k {
393            for q in p + 1..k {
394                // The three dot products in one pass, each summed in the
395                // same order as `dot`
396                let (mut alpha, mut beta, mut gamma) = (-0.0f64, -0.0f64, -0.0f64);
397                for (x, y) in a[p].iter().zip(&a[q]) {
398                    alpha += x * x;
399                    beta += y * y;
400                    gamma += x * y;
401                }
402                if gamma == 0.0 || gamma.abs() <= f64::EPSILON * (alpha * beta).sqrt() {
403                    continue;
404                }
405                rotated = true;
406                let zeta = (beta - alpha) / (2.0 * gamma);
407                let t = zeta.signum() / (zeta.abs() + zeta.hypot(1.0));
408                let c = 1.0 / t.hypot(1.0);
409                let s = c * t;
410                rotate(&mut a, p, q, c, s);
411                rotate(&mut v, p, q, c, s);
412            }
413        }
414        if !rotated {
415            break;
416        }
417    }
418    let s = a.iter().map(|col| dot(col, col).sqrt()).collect();
419    (a, s, v)
420}

422/// Autoregressive coefficient feature.
423///
424/// Fits an AR(k) process with an intercept by conditional maximum likelihood
425/// (ordinary least squares on the lagged values, as statsmodels'
426/// `AutoReg(x, lags=k, trend="c").fit()`),
427/// $$ x_t = \varphi_0 + \sum_{i=1}^{k} \varphi_i\, x_{t-i} + \varepsilon_t, $$
428/// and returns the coefficient $\varphi_{\text{coeff}}$ ($\varphi_0$ is the
429/// intercept).
430///
431/// # Parameters
432/// - `coeff`: which coefficient $i$ to return.
433/// - `k`: the order of the process.
434///
435/// Config: `[ar_coefficient] parameters = [{ coeff = 0, k = 10 }, ...]`; one
436/// column per entry. One fit per distinct `k`.
437///
438/// # Output column
439/// `{name}__ar_coefficient__coeff_{coeff}__k_{k}`
440///
441/// # Edge cases
442/// - Nulls are dropped first; a group with no non-null values gives NaN.
443/// - `coeff` $> k$ gives NaN.
444/// - Fewer than $2k + 1$ values (statsmodels needs at least as many
445///   observations as regressors and raises otherwise): NaN for `coeff`
446///   $< k$ and 0 for `coeff` $= k$, as tsfresh (whose fallback has only $k$
447///   entries).
448/// - Exactly $2k + 1$ values: the exact fit (no residual degrees of
449///   freedom), as statsmodels 0.15; for example the value itself for a
450///   single value and $k = 0$.
451/// - A constant or otherwise collinear series gives the minimum-norm least
452///   squares solution, as statsmodels' pseudoinverse. For near-collinear
453///   (ill-conditioned) lag matrices the result is rounding-dependent, in
454///   statsmodels too.
455/// - Well-conditioned fits far from exact are solved from the Gram matrix
456///   rather than a QR decomposition (faster on long series), agreeing to
457///   within about $10^{-10}$ of the largest coefficient, not bit for bit.
458/// - **Deliberate deviation from tsfresh:** a series containing NaN or
459///   $\pm\infty$ gives NaN for every coefficient. tsfresh raises
460///   (statsmodels' `MissingDataError` is not caught), failing the whole
461///   extraction.
462///
463/// # tsfresh
464/// `feature_calculators.ar_coefficient` (v0.21.2).
465pub struct ArCoefficient {
466    /// `(coeff, k)` per config entry.
467    pub params: Vec<(usize, usize)>,
468}
469
470/// The AR(k) coefficients $\varphi_0, \dots, \varphi_k$, or `None` if the
471/// model cannot be fitted.
472fn ar_fit(x: &[f64], k: usize) -> Option<Vec<f64>> {
473    let n = x.len();
474    // statsmodels needs at least as many observations (n - k) as regressors
475    // (k + 1) and raises ValueError otherwise; with exactly as many, 0.15
476    // returns the exact fit (0.14 raised ZeroDivisionError)
477    if n < 2 * k + 1 {
478        return None;
479    }
480    let nobs = n - k;
481    // Columns: intercept, then x lagged by 1, ..., k
482    let design: Vec<Vec<f64>> = (0..=k)
483        .map(|c| {
484            if c == 0 {
485                vec![1.0; nobs]
486            } else {
487                x[k - c..n - c].to_vec()
488            }
489        })
490        .collect();
491    Ols::fit(&design, &x[k..]).map(|fit| fit.params)
492}

494impl Kernel for ArCoefficient {
495    fn names(&self) -> Vec<String> {
496        self.params
497            .iter()
498            .map(|(coeff, k)| format!("ar_coefficient__coeff_{coeff}__k_{k}"))
499            .collect()
500    }
501
502    fn compute(&self, ctx: &GroupContext, out: &mut Vec<f64>) {
503        let x = ctx.values();
504        if x.iter().any(|v| !v.is_finite()) {
505            out.extend(self.params.iter().map(|_| f64::NAN));
506            return;
507        }
508        let mut fits: Vec<(usize, Option<Vec<f64>>)> = Vec::new();
509        for &(coeff, k) in &self.params {
510            let fit = match fits.iter().find(|(fk, _)| *fk == k) {
511                Some((_, fit)) => fit,
512                None => {
513                    fits.push((k, ar_fit(x, k)));
514                    &fits.last().unwrap().1
515                }
516            };
517            out.push(match fit {
518                _ if coeff > k => f64::NAN,
519                Some(params) => params[coeff],
520                // tsfresh's fallback is [nan] * k, and indexing past it gives 0
521                None if coeff == k => 0.0,
522                None => f64::NAN,
523            });
524        }
525    }
526}

528/// MacKinnon's (1994, 2010 tables) approximate p-value of an augmented
529/// Dickey-Fuller test statistic for a regression with a constant and one
530/// integrated series, as statsmodels' `mackinnonp(teststat, "c", N=1)`.
531fn mackinnon_p(teststat: f64) -> f64 {
532    // statsmodels' tau_max_c, tau_min_c, tau_star_c and the scaled
533    // tau_c_smallp / tau_c_largep coefficients for N = 1
534    const TAU_MAX: f64 = 2.74;
535    const TAU_MIN: f64 = -18.83;
536    const TAU_STAR: f64 = -1.61;
537    const SMALL_P: [f64; 3] = [2.1659, 1.4412, 0.038269000000000004];
538    const LARGE_P: [f64; 4] = [1.7339, 0.9320200000000001, -0.12745, -0.010368];
539    if teststat > TAU_MAX {
540        return 1.0;
541    } else if teststat < TAU_MIN {
542        return 0.0;
543    }
544    let coef: &[f64] = if teststat <= TAU_STAR {
545        &SMALL_P
546    } else {
547        &LARGE_P
548    };
549    // np.polyval on the reversed (ascending) coefficients: Horner's scheme
550    let z = coef.iter().rev().fold(0.0, |acc, c| acc * teststat + c);
551    normal_cdf(z)
552}
553
554/// The standard normal CDF, as scipy's `ndtr` computes it from erf / erfc.
555fn normal_cdf(a: f64) -> f64 {
556    let x = a * std::f64::consts::FRAC_1_SQRT_2;
557    let z = x.abs();
558    if z < std::f64::consts::FRAC_1_SQRT_2 {
559        0.5 + 0.5 * libm::erf(x)
560    } else {
561        let y = 0.5 * libm::erfc(z);
562        if x > 0.0 { 1.0 - y } else { y }
563    }
564}

566/// The ADF regression of the last `nobs` differences (the response, second)
567/// on the lagged level, `lags` lagged differences and a constant (the
568/// design columns, first): `[constant, level, lag 1, ..., lag lags]` if
569/// `const_first`, else `[level, lag 1, ..., lag lags, constant]`.
570fn adf_regression(
571    x: &[f64],
572    lags: usize,
573    nobs: usize,
574    const_first: bool,
575) -> (Vec<Vec<f64>>, Vec<f64>) {
576    let n = x.len();
577    // Row r explains the difference x[t + 1] - x[t] with t = n - 1 - nobs + r
578    let start = n - 1 - nobs;
579    let diffs = |from: usize| (from..from + nobs).map(|i| x[i + 1] - x[i]).collect();
580    let mut design: Vec<Vec<f64>> = Vec::with_capacity(lags + 2);
581    design.push(x[start..start + nobs].to_vec());
582    design.extend((1..=lags).map(|lag| diffs(start - lag)));
583    if const_first {
584        design.insert(0, vec![1.0; nobs]);
585    } else {
586        design.push(vec![1.0; nobs]);
587    }
588    (design, diffs(start))
589}

591/// Test statistic, p-value and used lag of the augmented Dickey-Fuller test
592/// with a constant, as statsmodels' `adfuller(x, autolag=...)`; NaN for all
593/// three where statsmodels raises (constant or too short series, a
594/// non-finite value in a design matrix).
595fn adfuller(x: &[f64], autolag: AdfAutolag) -> [f64; 3] {
596    const FAILED: [f64; 3] = [f64::NAN; 3];
597    let n = x.len();
598    // x.max() == x.min(): never true with a NaN
599    let (min, max) = super::kernel::extremes(x.iter().copied());
600    if min == max {
601        return FAILED;
602    }
603    // Schwert's rule, capped at nobs // 2 - ntrend - 1 (ntrend = 1)
604    let schwert = (12.0 * (n as f64 / 100.0).powf(0.25)).ceil() as i64;
605    let maxlag = schwert.min(n as i64 / 2 - 2);
606    if maxlag < 0 {
607        return FAILED;
608    }
609    let maxlag = maxlag as usize;
610
611    let usedlag = if autolag == AdfAutolag::None {
612        maxlag
613    } else {
614        // Every candidate is fitted on the same observations, so their
615        // information criteria are comparable
616        let (full, y) = adf_regression(x, maxlag, n - 1 - maxlag, true);
617        // fits[lags]: constant, level and the first `lags` lags
618        let Some(fits) = Ols::fit_prefixes(&full, &y, 2) else {
619            return FAILED;
620        };
621        match autolag {
622            AdfAutolag::Aic | AdfAutolag::Bic => {
623                let ic = |fit: &Ols| {
624                    if autolag == AdfAutolag::Aic {
625                        fit.aic()
626                    } else {
627                        fit.bic()
628                    }
629                };
630                // Python's min over (ic, lag): the first strict minimum.
631                // Criteria within rounding of each other tie (an extra lag
632                // collinear with the others adds neither rank nor fit), and
633                // ties go to the smaller lag
634                let mut best = 0;
635                for lags in 1..=maxlag {
636                    let best_ic = ic(&fits[best]);
637                    if ic(&fits[lags]) < best_ic - IC_TIE_TOL * best_ic.abs().max(1.0) {
638                        best = lags;
639                    }
640                }
641                best
642            }
643            _ => {
644                // t-stat: drop lags until the last one is significant at 5%
645                const STOP: f64 = 1.6448536269514722;
646                let mut best = 0;
647                for lags in (0..=maxlag).rev() {
648                    best = lags;
649                    if fits[lags].tvalue(lags + 1).abs() >= STOP {
650                        break;
651                    }
652                }
653                best
654            }
655        }
656    };
657    let (design, y) = adf_regression(x, usedlag, n - 1 - usedlag, false);
658    let Some(fit) = Ols::fit(&design, &y) else {
659        return FAILED;
660    };
661    let teststat = fit.tvalue(0);
662    [teststat, mackinnon_p(teststat), usedlag as f64]
663}

665/// Augmented Dickey-Fuller feature.
666///
667/// The augmented Dickey-Fuller test for a unit root, with a constant, as
668/// statsmodels' `adfuller(x, autolag=...)`: regress the differences on the
669/// lagged level, $p$ lagged differences and a constant,
670/// $$ \Delta x_t = \alpha + \gamma\, x_{t-1} + \sum_{i=1}^{p} \beta_i\, \Delta x_{t-i} + \varepsilon_t, $$
671/// where $p$ is chosen among $0, \dots, \lceil 12 (n/100)^{1/4} \rceil$
672/// (capped at $\lfloor n/2 \rfloor - 2$) by `autolag`. Returns the test
673/// statistic (the t-value of $\gamma$), its MacKinnon approximate p-value or
674/// the used lag $p$. A small p-value rejects the unit root.
675///
676/// # Parameters
677/// - `attr`: `teststat`, `pvalue` or `usedlag`.
678/// - `autolag` (optional, default `"AIC"`): how $p$ is chosen: `"AIC"` or
679///   `"BIC"` (smallest criterion, all candidates fitted on the same
680///   observations), `"t-stat"` (from the largest $p$ down until the last lag
681///   is significant at 5%) or `"None"` (the largest $p$).
682///
683/// Config: `[augmented_dickey_fuller] parameters = [{ attr = "teststat" }, { attr = "pvalue", autolag = "BIC" }, ...]`;
684/// one column per entry. One test per distinct `autolag`.
685///
686/// # Output column
687/// `{name}__augmented_dickey_fuller__attr_{attr}__autolag_{autolag}`
688///
689/// # Edge cases
690/// - Nulls are dropped first; a group with no non-null values gives NaN.
691/// - Fewer than 4 values or a constant series give NaN (statsmodels raises,
692///   tsfresh returns NaN).
693/// - NaN or $\pm\infty$ values give NaN. Only the last value enters no
694///   design matrix, so if it is the only non-finite one, statsmodels gets
695///   as far as choosing a lag: the test statistic and p-value are NaN, the
696///   used lag is 0 (`"AIC"`, `"BIC"`, `"t-stat"`, as every criterion is NaN)
697///   or the largest lag (`"None"`).
698/// - **Deliberate deviation from tsfresh:** when the regression fits exactly
699///   (root mean squared residual at most $10^{-10}$ of the scale of the
700///   fitted terms, e.g. a periodic or linear series), the residual variance
701///   is taken as 0: the test statistic is $\pm\infty$ (p-value 0 or 1) and
702///   AIC / BIC pick the smallest exactly fitting lag. statsmodels divides by
703///   the rounding noise of the residuals instead, so its statistic (around
704///   $\pm 10^{15}$) and chosen lag are arbitrary.
705/// - **Deliberate deviation from tsfresh:** when two candidate lags fit
706///   equally well because the extra lagged difference is collinear with the
707///   other regressors (e.g. after a constant stretch), their AIC / BIC tie
708///   (to within a relative $10^{-9}$) and the smaller lag is used, as
709///   Python's `min` does for exact ties. statsmodels breaks the tie by
710///   rounding noise.
711/// - Nonzero values so small that their squares underflow (below about
712///   $10^{-150}$) leave the regression numerically rank-deficient; the
713///   result then depends on rounding, in statsmodels as here.
714/// - Well-conditioned regressions that are far from an exact fit (most
715///   series) are solved from the Gram matrix instead of a QR decomposition,
716///   which is much faster on long series. The two agree to within about
717///   $10^{-10}$ relative, not bit for bit; ill-conditioned, near-exact and
718///   exactly fitting regressions always use the QR decomposition.
719///
720/// # Cost
721/// $O(n\,p^2)$ per distinct `autolag`, with $p$ the largest lag (68 for
722/// $n = 10^5$): the $p + 1$ candidate fits share one Gram matrix (or QR
723/// decomposition), which needs a single pass over the series.
724///
725/// # tsfresh
726/// `feature_calculators.augmented_dickey_fuller` (v0.21.2).
727pub struct AugmentedDickeyFuller {
728    /// `(attr, autolag)` per config entry.
729    pub params: Vec<(AdfAttr, AdfAutolag)>,
730}
731
732impl Kernel for AugmentedDickeyFuller {
733    fn names(&self) -> Vec<String> {
734        self.params
735            .iter()
736            .map(|(attr, autolag)| {
737                format!("augmented_dickey_fuller__attr_{attr}__autolag_{autolag}")
738            })
739            .collect()
740    }
741
742    fn compute(&self, ctx: &GroupContext, out: &mut Vec<f64>) {
743        let x = ctx.values();
744        let mut tests: Vec<(AdfAutolag, [f64; 3])> = Vec::new();
745        for &(attr, autolag) in &self.params {
746            let result = match tests.iter().find(|(a, _)| *a == autolag) {
747                Some((_, result)) => *result,
748                None => {
749                    let result = adfuller(x, autolag);
750                    tests.push((autolag, result));
751                    result
752                }
753            };
754            out.push(match attr {
755                AdfAttr::Teststat => result[0],
756                AdfAttr::Pvalue => result[1],
757                AdfAttr::Usedlag => result[2],
758            });
759        }
760    }
761}
