10/// Mean absolute change feature.
11///
12/// The mean of the absolute differences between consecutive values:
13/// $$ \frac{1}{n-1} \sum_{i=1}^{n-1} |x_{i+1} - x_i|, $$
14/// where $n$ is the number of values in the time series.
15///
16/// # Output column
17/// `{name}__mean_absolute_change`
18///
19/// # Edge cases
20/// - Nulls are dropped first; a group with no non-null values gives NaN.
21/// - A single value has no differences and gives NaN.
22/// - NaN values are kept and make the result NaN.
23///
24/// # tsfresh
25/// `feature_calculators.mean_abs_change` (v0.21.2).
26    MeanAbsoluteChange, "mean_absolute_change", |ctx| {
27        // np.mean(np.abs(np.diff(x))); NaN for a single value
28        let x = ctx.values();
29        numpy_sum(x.len() - 1, |i| (x[i + 1] - x[i]).abs()) / (x.len() - 1) as f64
30    }
31}

56scalar_kernel! {
57/// Mean change feature.
58///
59/// The mean of the differences between consecutive values. The sum
60/// telescopes, so only the first and last values matter:
61/// $$ \frac{1}{n-1} \sum_{i=1}^{n-1} (x_{i+1} - x_i) = \frac{x_n - x_1}{n-1}. $$
62///
63/// # Output column
64/// `{name}__mean_change`
65///
66/// # Edge cases
67/// - Nulls are dropped first; a group with no non-null values gives NaN.
68/// - A single value gives NaN.
69/// - Only the endpoints are used, so NaN elsewhere does not affect the
70///   result; NaN at either end gives NaN.
71///
72/// # tsfresh
73/// `feature_calculators.mean_change` (v0.21.2).
74    MeanChange, "mean_change", |ctx| {
75        let x = ctx.values();
76        if x.len() < 2 {
77            f64::NAN
78        } else {
79            (x[x.len() - 1] - x[0]) / ((x.len() - 1) as f64)
80        }
81    }
82}

84/// Complexity-invariant distance (CID CE) feature.
85///
86/// An estimate of the time series' complexity: the length of the line
87/// connecting consecutive values,
88/// $$ \sqrt{\sum_{i=1}^{n-1} (z_{i+1} - z_i)^2}, $$
89/// where $z = x$, or with normalisation $z_i = (x_i - \mu)/\sigma$ using the
90/// mean $\mu$ and population standard deviation $\sigma$. A more complex
91/// series (more peaks and valleys) scores higher.
92///
93/// # Parameters
94/// - `normalize`: z-normalise the series first. Config:
95///   `[cid_ce] parameters = [{ normalize = true }, { normalize = false }]`; one
96///   column per entry.
97///
98/// # Output column
99/// `{name}__cid_ce__normalize_t` or `{name}__cid_ce__normalize_f`
100///
101/// # Edge cases
102/// - Nulls are dropped first; a group with no non-null values gives NaN.
103/// - A single value gives 0.
104/// - With `normalize`, a constant series has $\sigma = 0$ and gives 0.
105/// - NaN values are kept and make the result NaN.
106///
107/// # tsfresh
108/// `feature_calculators.cid_ce` (v0.21.2).
109pub struct CidCe {
110    pub normalize: Vec<bool>,
111}
112
113impl Kernel for CidCe {
114    fn names(&self) -> Vec<String> {
115        // `{:.1}` truncates the bool to "t" / "f"; kept for stable names
116        self.normalize
117            .iter()
118            .map(|normalize| format!("cid_ce__normalize_{normalize:.1}"))
119            .collect()
120    }
121
122    fn compute(&self, ctx: &GroupContext, out: &mut Vec<f64>) {
123        for &normalize in &self.normalize {
124            let x = if normalize {
125                // tsfresh: a constant series has no complexity
126                if ctx.std() == 0.0 {
127                    out.push(0.0);
128                    continue;
129                }
130                CowArray::from((ctx.x() - ctx.mean()) / ctx.std())
131            } else {
132                CowArray::from(ctx.x().view())
133            };
134            let diffs = &x.slice(s![1..]) - &x.slice(s![..-1]);
135            out.push(diffs.mapv(|d| d.powi(2)).sum().sqrt());
136        }
137    }
138}

140scalar_kernel! {
141/// Mean second derivative central feature.
142///
143/// The mean of the central approximation of the second derivative,
144/// $\frac{1}{2}(x_{i+2} - 2\,x_{i+1} + x_i)$. The sum telescopes, so only
145/// the first two and last two values matter:
146/// $$ \frac{1}{n-2} \sum_{i=1}^{n-2} \frac{1}{2}(x_{i+2} - 2\,x_{i+1} + x_i)
147///    = \frac{x_n - x_{n-1} - x_2 + x_1}{2\,(n-2)}. $$
148///
149/// # Output column
150/// `{name}__mean_second_derivative_central`
151///
152/// # Edge cases
153/// - Nulls are dropped first; a group with no non-null values gives NaN.
154/// - Fewer than 3 values give NaN.
155/// - Only the first two and last two values are used, so NaN elsewhere does
156///   not affect the result.
157///
158/// # tsfresh
159/// `feature_calculators.mean_second_derivative_central` (v0.21.2).
160    MeanSecondDerivativeCentral, "mean_second_derivative_central", |ctx| {
161        let x = ctx.values();
162        let n = x.len();
163        if n > 2 {
164            // (x[-1] - x[-2] - x[1] + x[0]) / (2 * (len(x) - 2)), in this order
165            (x[n - 1] - x[n - 2] - x[1] + x[0]) / (2 * (n - 2)) as f64
166        } else {
167            f64::NAN
168        }
169    }
170}

172/// Change quantiles feature.
173///
174/// Fix a corridor between the `ql`- and `qh`-quantiles of the values, then
175/// aggregate the consecutive changes $x_{i+1} - x_i$ (or their absolute
176/// values) that start and end inside the corridor:
177/// $$ f_{agg}\left(\{\, x_{i+1} - x_i : c_i \wedge c_{i+1} \,\}\right), $$
178/// where $c_i$ says whether $x_i$ lies in the corridor
179/// $(Q(q_l), Q(q_h)]$, the lower edge included.
180///
181/// The corridor is `pd.qcut(x, [ql, qh])`'s first bin: the edges are the
182/// quantiles of the non-NaN values with linear interpolation (as pandas
183/// computes them, through `np.percentile` with `100 q`), and values are
184/// binned with numpy's `searchsorted` exactly as pandas does.
185///
186/// # Parameters
187/// - `ql`, `qh`: lower and upper quantile of the corridor.
188/// - `isabs`: aggregate absolute changes.
189/// - `f_agg`: `mean`, `var`, `std` or `median` of the changes (population
190///   variance and standard deviation, `ddof = 0`; NaN propagates as in
191///   numpy). Any other name is rejected when the config is loaded.
192///
193/// Config:
194/// `[change_quantiles] parameters = [{ ql = 0.0, qh = 0.2, isabs = false, f_agg = "mean" }, ...]`;
195/// one column per entry.
196///
197/// # Output column
198/// `{name}__change_quantiles__ql_{ql}__qh_{qh}__isabs_{isabs}__f_agg_{f_agg}`,
199/// with `ql` and `qh` written as Python's `repr` would (e.g. `0.2`, `1.0`)
200/// and `isabs` as `true` / `false`.
201///
202/// # Edge cases
203/// - Nulls are dropped first; a group with no non-null values gives NaN.
204/// - $q_l \geq q_h$ gives 0.
205/// - No change inside the corridor (e.g. a single value) gives 0.
206/// - NaN values never lie in the corridor; a change involving $\pm\infty$
207///   can be NaN and then makes the aggregate NaN, as in numpy.
208/// - If both edges coincide, pandas (3.x) keeps the duplicate edge for two
209///   quantiles, so the corridor holds exactly the values equal to it.
210///
211/// # tsfresh
212/// `feature_calculators.change_quantiles` (v0.21.2).
213pub struct ChangeQuantiles {
214    /// `(ql, qh, isabs, f_agg)` per config entry.
215    pub params: Vec<(f64, f64, bool, ChangeAggregator)>,
216}
217
218impl Kernel for ChangeQuantiles {
219    fn names(&self) -> Vec<String> {
220        self.params
221            .iter()
222            .map(|(ql, qh, isabs, f_agg)| {
223                format!("change_quantiles__ql_{ql:?}__qh_{qh:?}__isabs_{isabs}__f_agg_{f_agg}")
224            })
225            .collect()
226    }
227
228    fn compute(&self, ctx: &GroupContext, out: &mut Vec<f64>) {
229        let x = ctx.values();
230        // The quantiles of the non-NaN values (pandas drops NaN first)
231        let finite_sorted = ctx
232            .sorted()
233            .iter()
234            .copied()
235            .filter(|v| !v.is_nan())
236            .collect::<Vec<f64>>();
237        // The corridor's changes, shared by the aggregations of the same
238        // (ql, qh, isabs); the default grid has 3 per corridor and side
239        let mut corridors: Vec<((f64, f64, bool), Vec<f64>)> = Vec::new();
240        let mut scratch = Vec::new();
241        for &(ql, qh, isabs, f_agg) in &self.params {
242            if ql >= qh {
243                out.push(0.0);
244                continue;
245            }
246            let key = (ql, qh, isabs);
247            let changes = match corridors.iter().position(|(k, _)| *k == key) {
248                Some(i) => &corridors[i].1,
249                None => {
250                    let bins = [ql, qh].map(|q| pandas_quantile(&finite_sorted, q));
251                    let inside = qcut_first_bin(x, bins);
252                    // Only changes that start and end inside the corridor
253                    let changes = (0..x.len().saturating_sub(1))
254                        .filter(|&i| inside[i] && inside[i + 1])
255                        .map(|i| {
256                            let d = x[i + 1] - x[i];
257                            if isabs { d.abs() } else { d }
258                        })
259                        .collect();
260                    corridors.push((key, changes));
261                    &corridors.last().unwrap().1
262                }
263            };
264            out.push(if changes.is_empty() {
265                0.0
266            } else {
267                // A fresh copy: the median reorders it
268                scratch.clear();
269                scratch.extend_from_slice(changes);
270                f_agg.aggregate(&mut scratch)
271            });
272        }
273    }
274}

276impl ChangeAggregator {
277    /// The numpy function of the same name on `values` (not empty; may be
278    /// reordered).
279    fn aggregate(&self, values: &mut [f64]) -> f64 {
280        match self {
281            ChangeAggregator::Mean => numpy_mean(values),
282            ChangeAggregator::Var => population_var(&ArrayView1::from(&*values)),
283            ChangeAggregator::Std => population_var(&ArrayView1::from(&*values)).sqrt(),
284            ChangeAggregator::Median => {
285                if values.iter().any(|v| v.is_nan()) {
286                    f64::NAN
287                } else {
288                    numpy_median(values)
289                }
290            }
291        }
292    }
293}

295/// `Series.quantile(q)` of non-NaN `sorted` values as pandas computes it:
296/// `np.percentile(x, q * 100)`, which divides by 100 again (so `q = 0.2`
297/// becomes `0.20000000000000004`). NaN for no values.
298fn pandas_quantile(sorted: &[f64], q: f64) -> f64 {
299    if sorted.is_empty() {
300        return f64::NAN;
301    }
302    numpy_quantile(sorted, (q * 100.0) / 100.0)
303}

305/// Which values fall in the first bin of `pd.qcut(x, bins, labels=False)`
306/// with two edges: `ids = np.searchsorted(bins, x, side="left")`, values
307/// equal to the lower edge moved into the bin, NaN never in it.
308fn qcut_first_bin(x: &[f64], bins: [f64; 2]) -> Vec<bool> {
309    let mut ids = numpy_searchsorted_left(&bins, x);
310    for (id, &v) in ids.iter_mut().zip(x) {
311        if v == bins[0] {
312            *id = 1;
313        }
314    }
315    ids.iter()
316        .zip(x)
317        .map(|(&id, v)| id == 1 && !v.is_nan())
318        .collect()
319}

321/// `np.searchsorted(arr, keys, side="left")` for float64, including numpy's
322/// use of the previous key to narrow the search (which matters only when
323/// `arr` is not sorted, e.g. a NaN edge before a number) and its NaN-last
324/// ordering.
325fn numpy_searchsorted_left(arr: &[f64], keys: &[f64]) -> Vec<usize> {
326    let less = |a: f64, b: f64| a < b || (b.is_nan() && !a.is_nan());
327    let n = arr.len();
328    let (mut min_idx, mut max_idx) = (0, n);
329    let Some(&first) = keys.first() else {
330        return Vec::new();
331    };
332    let mut last_key = first;
333    keys.iter()
334        .map(|&key| {
335            if less(last_key, key) {
336                max_idx = n;
337            } else {
338                min_idx = 0;
339                max_idx = if max_idx < n { max_idx + 1 } else { n };
340            }
341            last_key = key;
342            while min_idx < max_idx {
343                let mid = min_idx + ((max_idx - min_idx) >> 1);
344                if less(arr[mid], key) {
345                    min_idx = mid + 1;
346                } else {
347                    max_idx = mid;
348                }
349            }
350            min_idx
351        })
352        .collect()
