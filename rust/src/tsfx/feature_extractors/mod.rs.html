61/// Feature expressions for the configured features of the selected tier.
62///
63/// Columns come out as `length` first, then, for each value column in turn,
64/// all of its features in tier order (Minimal, then Efficient, then
65/// Comprehensive).
66pub fn aggregators(opts: &ExtractionSettings) -> Result<FeatureExprs, ConfigError> {
67    let config = load_config(opts.config_path.as_deref())?;
68    let tier = &opts.feature_setting;
69    let mut aggregators = Vec::new();
70    // Features computed by the single `apply` of each value column
71    let mut plans: Vec<KernelPlan> = opts
72        .value_cols
73        .iter()
74        .map(|_| KernelPlan::default())
75        .collect();
76
77    // Minimal tier
78    if config.length.is_some() {
79        aggregators.push(count(&opts.value_cols[0]));
80    }
81    for plan in plans.iter_mut() {
82        if config.sum_values.is_some() {
83            plan.push(SumValues);
84        }
85        if config.mean.is_some() {
86            plan.push(Mean);
87        }
88        if config.median.is_some() {
89            plan.push(Median);
90        }
91        if config.minimum.is_some() {
92            plan.push(Minimum);
93        }
94        if config.maximum.is_some() {
95            plan.push(Maximum);
96        }
97        if config.standard_deviation.is_some() {
98            plan.push(StandardDeviation);
99        }
100        if config.variance.is_some() {
101            plan.push(Variance);
102        }
103        if config.skewness.is_some() {
104            plan.push(Skewness);
105        }
106        if config.root_mean_square.is_some() {
107            plan.push(RootMeanSquare);
108        }
109    }
110
111    // Efficient tier
112    if *tier >= FeatureSetting::Efficient {
113        for plan in plans.iter_mut() {
114            if config.kurtosis.is_some() {
115                plan.push(Kurtosis);
116            }
117            if config.absolute_energy.is_some() {
118                plan.push(AbsoluteEnergy);
119            }
120            if config.mean_absolute_change.is_some() {
121                plan.push(MeanAbsoluteChange);
122            }
123            if config.linear_trend.is_some() {
124                plan.push(LinearTrend);
125            }
126            if config.variance_larger_than_standard_deviation.is_some() {
127                plan.push(VarianceLargerThanStandardDeviation);
128            }
129            if let Some(feature) = &config.ratio_beyond_r_sigma {
130                let params = &feature.parameters;
131                let mut rs = Vec::new();
132                for p in params {
133                    rs.push(p.r);
134                }
135                plan.push(RatioBeyondRSigma { rs });
136            }
137
138            if let Some(feature) = &config.large_standard_deviation {
139                let params = &feature.parameters;
140                let mut rs = Vec::new();
141                for p in params {
142                    rs.push(p.r);
143                }
144                plan.push(LargeStandardDeviation { rs });
145            }
146            if let Some(feature) = &config.symmetry_looking {
147                let params = &feature.parameters;
148                let mut rs = Vec::new();
149                for p in params {
150                    rs.push(p.r);
151                }
152                plan.push(SymmetryLooking { rs });
153            }
154            if config.has_duplicate_max.is_some() {
155                plan.push(HasDuplicateMax);
156            }
157            if config.has_duplicate_min.is_some() {
158                plan.push(HasDuplicateMin);
159            }
160            if let Some(feature) = &config.cid_ce {
161                let normalize = feature.parameters.iter().map(|p| p.normalize).collect();
162                plan.push(CidCe { normalize });
163            }
164            if config.absolute_maximum.is_some() {
165                plan.push(AbsoluteMaximum);
166            }
167            if config.absolute_sum_of_changes.is_some() {
168                plan.push(AbsoluteSumOfChanges);
169            }
170            if config.count_above_mean.is_some() {
171                plan.push(CountAboveMean);
172            }
173            if config.count_below_mean.is_some() {
174                plan.push(CountBelowMean);
175            }
176            if let Some(feature) = &config.count_above {
177                let ts = feature.parameters.iter().map(|p| p.t).collect();
178                plan.push(CountAbove { ts });
179            }
180            if let Some(feature) = &config.count_below {
181                let ts = feature.parameters.iter().map(|p| p.t).collect();
182                plan.push(CountBelow { ts });
183            }
184            if config.first_location_of_maximum.is_some() {
185                plan.push(FirstLocationOfMaximum);
186            }
187            if config.first_location_of_minimum.is_some() {
188                plan.push(FirstLocationOfMinimum);
189            }
190            if config.last_location_of_maximum.is_some() {
191                plan.push(LastLocationOfMaximum);
192            }
193            if config.last_location_of_minimum.is_some() {
194                plan.push(LastLocationOfMinimum);
195            }
196            if config.longest_strike_above_mean.is_some() {
197                plan.push(LongestStrikeAboveMean);
198            }
199            if config.longest_strike_below_mean.is_some() {
200                plan.push(LongestStrikeBelowMean);
201            }
202            // The counting-based reoccurrence features share one sort per group
203            let reoccurring = ReoccurrenceFeatures {
204                has_duplicate: config.has_duplicate.is_some(),
205                ratio_value_number_to_time_series_length: config
206                    .ratio_value_number_to_time_series_length
207                    .is_some(),
208                sum_of_reoccurring_values: config.sum_of_reoccurring_values.is_some(),
209                sum_of_reoccurring_data_points: config.sum_of_reoccurring_data_points.is_some(),
210                percentage_of_reoccurring_values_to_all_values: config
211                    .percentage_of_reoccurring_values_to_all_values
212                    .is_some(),
213                percentage_of_reoccurring_values_to_all_datapoints: config
214                    .percentage_of_reoccurring_values_to_all_datapoints
215                    .is_some(),
216            };
217            if reoccurring != ReoccurrenceFeatures::default() {
218                plan.push(reoccurring);
219            }
220            if config.variation_coefficient.is_some() {
221                plan.push(VariationCoefficient);
222            }
223            if config.mean_change.is_some() {
224                plan.push(MeanChange);
225            }
226            if let Some(feature) = &config.agg_linear_trend {
227                let params = feature
228                    .parameters
229                    .iter()
230                    .map(|p| (p.chunk_size, p.aggregator.clone()))
231                    .collect();
232                plan.push(AggLinearTrend { params });
233            }
234            if let Some(feature) = &config.mean_n_absolute_max {
235                let params = &feature.parameters;
236                let mut ns = Vec::new();
237                for p in params {
238                    ns.push(p.n);
239                }
240                plan.push(MeanNAbsoluteMax { ns });
241            }
242            if let Some(feature) = &config.autocorrelation {
243                let lags = feature.parameters.iter().map(|p| p.lag).collect();
244                plan.push(Autocorrelation { lags });
245            }
246            if let Some(feature) = &config.quantile {
247                let qs = feature.parameters.iter().map(|p| p.q).collect();
248                plan.push(Quantile { qs });
249            }
250            if let Some(feature) = &config.number_crossing_m {
251                let ms = feature.parameters.iter().map(|p| p.m).collect();
252                plan.push(NumberCrossingM { ms });
253            }
254            if let Some(feature) = &config.range_count {
255                let ranges = feature.parameters.iter().map(|p| (p.min, p.max)).collect();
256                plan.push(RangeCount { ranges });
257            }
258            if let Some(feature) = &config.index_mass_quantile {
259                let qs = feature.parameters.iter().map(|p| p.q).collect();
260                plan.push(IndexMassQuantile { qs });
261            }
262            if let Some(feature) = &config.c3 {
263                let lags = feature.parameters.iter().map(|p| p.lag).collect();
264                plan.push(C3 { lags });
265            }
266            if let Some(feature) = &config.time_reversal_asymmetry_statistic {
267                let lags = feature.parameters.iter().map(|p| p.lag).collect();
268                plan.push(TimeReversalAsymmetryStatistic { lags });
269            }
270            if let Some(feature) = &config.number_peaks {
271                let ns = feature.parameters.iter().map(|p| p.n).collect();
272                plan.push(NumberPeaks { ns });
273            }
274            if config.mean_second_derivative_central.is_some() {
275                plan.push(MeanSecondDerivativeCentral);
276            }
277            if let Some(feature) = &config.value_count {
278                let values = feature.parameters.iter().map(|p| p.value).collect();
279                plan.push(ValueCount { values });
280            }
281            if let Some(feature) = &config.energy_ratio_by_chunks {
282                let params = feature
283                    .parameters
284                    .iter()
285                    .map(|p| (p.num_segments, p.segment_focus))
286                    .collect();
287                plan.push(EnergyRatioByChunks { params });
288            }
289            if let Some(feature) = &config.change_quantiles {
290                let params = feature
291                    .parameters
292                    .iter()
293                    .map(|p| (p.ql, p.qh, p.isabs, p.f_agg))
294                    .collect();
295                plan.push(ChangeQuantiles { params });
296            }
297            if config.benford_correlation.is_some() {
298                plan.push(BenfordCorrelation);
299            }
300            if let Some(feature) = &config.query_similarity_count {
301                let queries = feature
302                    .parameters
303                    .iter()
304                    .map(|p| Query {
305                        query: p.query.clone(),
306                        threshold: p.threshold,
307                        normalize: p.normalize,
308                    })
309                    .collect();
310                plan.push(QuerySimilarityCount { queries });
311            }
312            if let Some(feature) = &config.fft_coefficient {
313                let params = feature
314                    .parameters
315                    .iter()
316                    .map(|p| (p.attr.clone(), p.coeff))
317                    .collect();
318                plan.push(FftCoefficient { params });
319            }
320            if let Some(feature) = &config.fft_aggregated {
321                let aggtypes = feature
322                    .parameters
323                    .iter()
324                    .map(|p| p.aggtype.clone())
325                    .collect();
326                plan.push(FftAggregated { aggtypes });
327            }
328            if let Some(feature) = &config.spkt_welch_density {
329                let coeffs = feature.parameters.iter().map(|p| p.coeff).collect();
330                plan.push(SpktWelchDensity { coeffs });
331            }
332            if let Some(feature) = &config.fourier_entropy {
333                let bins = feature.parameters.iter().map(|p| p.bins).collect();
334                plan.push(FourierEntropy { bins });
335            }
336            if let Some(feature) = &config.cwt_coefficients {
337                let params = feature
338                    .parameters
339                    .iter()
340                    .map(|p| (p.widths.clone(), p.coeff, p.w))
341                    .collect();
342                plan.push(CwtCoefficients { params });
343            }
344            if let Some(feature) = &config.number_cwt_peaks {
345                let ns = feature.parameters.iter().map(|p| p.n).collect();
346                plan.push(NumberCwtPeaks { ns });
347            }
348            if let Some(feature) = &config.agg_autocorrelation {
349                let params = feature
350                    .parameters
351                    .iter()
352                    .map(|p| (p.f_agg, p.maxlag))
353                    .collect();
354                plan.push(AggAutocorrelation { params });
355            }
356            if let Some(feature) = &config.partial_autocorrelation {
357                let lags = feature.parameters.iter().map(|p| p.lag).collect();
358                plan.push(PartialAutocorrelation { lags });
359            }
360            if let Some(feature) = &config.ar_coefficient {
361                let params = feature.parameters.iter().map(|p| (p.coeff, p.k)).collect();
362                plan.push(ArCoefficient { params });
363            }
364            if let Some(feature) = &config.augmented_dickey_fuller {
365                let params = feature
366                    .parameters
367                    .iter()
368                    .map(|p| (p.attr, p.autolag))
369                    .collect();
370                plan.push(AugmentedDickeyFuller { params });
371            }
372            if let Some(feature) = &config.binned_entropy {
373                let max_bins = feature.parameters.iter().map(|p| p.max_bins).collect();
374                plan.push(BinnedEntropy { max_bins });
375            }
376            if let Some(feature) = &config.lempel_ziv_complexity {
377                let bins = feature.parameters.iter().map(|p| p.bins).collect();
378                plan.push(LempelZivComplexity { bins });
379            }
380            if let Some(feature) = &config.permutation_entropy {
381                let params = feature
382                    .parameters
383                    .iter()
384                    .map(|p| (p.tau, p.dimension))
385                    .collect();
386                plan.push(PermutationEntropy { params });
387            }
388            if let Some(feature) = &config.friedrich_coefficients {
389                let params = feature
390                    .parameters
391                    .iter()
392                    .map(|p| (p.coeff, p.m, p.r))
393                    .collect();
394                plan.push(FriedrichCoefficients { params });
395            }
396            if let Some(feature) = &config.max_langevin_fixed_point {
397                let params = feature.parameters.iter().map(|p| (p.m, p.r)).collect();
398                plan.push(MaxLangevinFixedPoint { params });
399            }
400        }
401    }
402
403    if *tier >= FeatureSetting::Comprehensive {
404        for plan in plans.iter_mut() {
405            if config.sample_entropy.is_some() {
406                plan.push(SampleEntropy);
407            }
408            if let Some(feature) = &config.approximate_entropy {
409                let params = feature.parameters.iter().map(|p| (p.m, p.r)).collect();
410                plan.push(ApproximateEntropy { params });
411            }
412        }
413    }
414
415    let mut kernels = Vec::new();
416    for (col, plan) in opts.value_cols.iter().zip(plans) {
417        if plan.is_empty() {
418            continue;
419        }
420        let plan = Arc::new(plan);
421        aggregators.push(plan.aggregation(col));
422        kernels.push((col.clone(), plan));
423    }
424    Ok(FeatureExprs {
425        aggregations: aggregators,
426        kernels,
427    })
428}

430/// What [`aggregators`] builds for a group-by.
431pub struct FeatureExprs {
432    /// Expressions for the group-by aggregation.
433    pub aggregations: Vec<Expr>,
434    /// Each value column's kernel plan, whose packed feature list must be
435    /// unpacked ([`KernelPlan::unpack`]) in the aggregated frame.
436    pub kernels: Vec<(String, Arc<KernelPlan>)>,
437}
438
439/// The group's values as one contiguous `Float64` chunk, ready for
440/// `ArrayView1::from(ca.cont_slice()?)`.
441///
442/// Unlike `to_ndarray`, this does not copy a `Float64` group that is already
443/// a single chunk; other types are cast (one copy, as before). The column
444/// must not contain nulls, or `cont_slice` fails.
445pub(super) fn f64_values(s: &Column) -> PolarsResult<Float64Chunked> {
446    let series = s.as_materialized_series().cast(&DataType::Float64)?;
447    Ok(series.f64()?.rechunk().into_owned())
448}

450#[cfg(test)]
451mod tests {
452    use super::*;
453    use crate::extract::{ExtractionSettings, FeatureSetting, lazy_feature_df};
454    use polars::datatypes::AnyValue;
455
456    #[test]
457    fn test_unit_length_df() {
458        // Create a dataframe with a single row
459        let df = df![
460            "id" => ["a"],
461            "val" => [1.0],
462        ]
463        .unwrap()
464        .lazy();
465
466        // Configure extraction settings
467        let opts = ExtractionSettings {
468            grouping_cols: vec!["id".to_string()],
469            feature_setting: FeatureSetting::Efficient,
470            value_cols: vec!["val".to_string()],
471            config_path: None,
472            dynamic_settings: None,
473            scaling: None,
474        };
475
476        // Extract features
477        let gdf = lazy_feature_df(df, opts).unwrap();
478
479        // Collect the results
480        let fdf = gdf.collect().unwrap();
481
482        // Assert that the resulting dataframe has exactly one row
483        assert_eq!(fdf.shape().0, 1);
484
485        // Also check that the length column has the expected value
486        assert_eq!(
487            fdf.column("length").unwrap().get(0).unwrap(),
488            AnyValue::UInt32(1)
489        );
490    }
