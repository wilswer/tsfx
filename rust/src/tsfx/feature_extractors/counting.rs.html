9/// Count above mean feature.
10///
11/// The number of values strictly greater than the mean of the series:
12/// $$ \sum_{i=1}^{n} \mathbb{1}[x_i > \mu]. $$
13///
14/// # Output column
15/// `{name}__count_above_mean`
16///
17/// # Edge cases
18/// - Nulls are dropped first; a group with no non-null values gives NaN.
19/// - A series containing NaN has a NaN mean and gives 0.
20/// - A constant series gives 0.
21///
22/// # tsfresh
23/// `feature_calculators.count_above_mean` (v0.21.2).
24    CountAboveMean, "count_above_mean", |ctx| {
25        let mean = ctx.mean();
26        ctx.x().mapv(|x| if x > mean { 1.0 } else { 0.0 }).sum()
27    }
28}

98/// Count above feature.
99///
100/// The fraction of values greater than or equal to the threshold $t$:
101/// $$ \frac{1}{n} \sum_{i=1}^{n} \mathbb{1}[x_i \geq t], $$
102/// where $n$ is the number of values in the time series.
103///
104/// # Parameters
105/// - `t`: threshold. Config: `[count_above] parameters = [{ t = 0.0 }]`; one
106///   column per entry.
107///
108/// # Output column
109/// `{name}__count_above__t_{t:.1}`
110///
111/// # Edge cases
112/// - Nulls are dropped first; a group with no non-null values gives NaN.
113/// - NaN values count in $n$ but never satisfy $x_i \geq t$.
114/// - The column name rounds `t` to one decimal, so e.g. 0.25 and 0.2 would
115///   share a name.
116///
117/// # tsfresh
118/// `feature_calculators.count_above` (v0.21.2).
119pub struct CountAbove {
120    pub ts: Vec<f64>,
121}
122
123impl Kernel for CountAbove {
124    fn names(&self) -> Vec<String> {
125        self.ts
126            .iter()
127            .map(|t| format!("count_above__t_{t:.1}"))
128            .collect()
129    }
130
131    fn compute(&self, ctx: &GroupContext, out: &mut Vec<f64>) {
132        // fraction of values >= t, as in tsfresh
133        for &t in &self.ts {
134            let count = ctx.x().mapv(|x| if x >= t { 1.0 } else { 0.0 }).sum();
135            out.push(count / ctx.len() as f64);
136        }
137    }
138}

182/// Number of crossings of m feature.
183///
184/// The number of times the series crosses the level $m$. Each value is
185/// classified as above ($x_i > m$) or not, and every change of class between
186/// neighbouring values counts as one crossing:
187/// $$ \sum_{i=1}^{n-1} \mathbb{1}\big(\mathbb{1}(x_i > m) \neq \mathbb{1}(x_{i+1} > m)\big). $$
188///
189/// # Parameters
190/// - `m`: level. Config: `[number_crossing_m] parameters = [{ m = 0.0 }, ...]`;
191///   one column per entry.
192///
193/// # Output column
194/// `{name}__number_crossing_m__m_{m:.1}`
195///
196/// # Edge cases
197/// - Nulls are dropped first; a group with no non-null values gives NaN.
198/// - A value equal to $m$, and NaN, count as "not above".
199///
200/// # tsfresh
201/// `feature_calculators.number_crossing_m` (v0.21.2).
202pub struct NumberCrossingM {
203    pub ms: Vec<f64>,
204}
205
206impl Kernel for NumberCrossingM {
207    fn names(&self) -> Vec<String> {
208        self.ms
209            .iter()
210            .map(|m| format!("number_crossing_m__m_{m:.1}"))
211            .collect()
212    }
213
214    fn compute(&self, ctx: &GroupContext, out: &mut Vec<f64>) {
215        // tsfresh: binarise as x > m (NaN and x == m are "not above") and
216        // count every change between neighbours
217        for &m in &self.ms {
218            let changes = ctx
219                .values()
220                .windows(2)
221                .filter(|w| (w[0] > m) != (w[1] > m))
222                .count();
223            out.push(changes as f64);
224        }
225    }
226}

228/// Range count feature.
229///
230/// The number of values in the half-open interval $[\text{min}, \text{max})$:
231/// $$ \sum_{i=1}^{n} \mathbb{1}[\text{min} \leq x_i < \text{max}]. $$
232///
233/// # Parameters
234/// - `min`, `max`: interval bounds. Config:
235///   `[range_count] parameters = [{ min = -1.0, max = 1.0 }, ...]`; one column
236///   per entry.
237///
238/// # Output column
239/// `{name}__range_count__min_{min:.1}__max_{max:.1}`
240///
241/// # Edge cases
242/// - Nulls are dropped first; a group with no non-null values gives NaN.
243/// - NaN values are never counted.
244/// - `max <= min` is an empty interval and gives 0.
245///
246/// # tsfresh
247/// `feature_calculators.range_count` (v0.21.2).
248pub struct RangeCount {
249    pub ranges: Vec<(f64, f64)>,
250}
251
252impl Kernel for RangeCount {
253    fn names(&self) -> Vec<String> {
254        self.ranges
255            .iter()
256            .map(|(lower, upper)| format!("range_count__min_{lower:.1}__max_{upper:.1}"))
257            .collect()
258    }
259
260    fn compute(&self, ctx: &GroupContext, out: &mut Vec<f64>) {
261        // Half-open interval [lower, upper), as tsfresh; empty if upper <= lower
262        for &(lower, upper) in &self.ranges {
263            let count = ctx
264                .values()
265                .iter()
266                .filter(|&&x| x >= lower && x < upper)
267                .count();
268            out.push(count as f64);
269        }
270    }
271}

273/// Number of peaks feature.
274///
275/// The number of peaks of support $n$: values strictly greater than the $n$
276/// neighbours on each side, $x_i > x_{i \pm k}$ for all $k = 1, \dots, n$.
277/// Values within $n$ of either end cannot be peaks.
278///
279/// # Parameters
280/// - `n`: support. Config: `[number_peaks] parameters = [{ n = 1 }, ...]`; one
281///   column per entry.
282///
283/// # Output column
284/// `{name}__number_peaks__n_{n}`
285///
286/// # Edge cases
287/// - Nulls are dropped first; a group with no non-null values gives NaN.
288/// - A series shorter than $2n + 1$ gives 0.
289/// - Comparisons with NaN are false, so NaN values are never peaks and
290///   neighbouring a NaN prevents a peak.
291///
292/// # tsfresh
293/// `feature_calculators.number_peaks` (v0.21.2).
294pub struct NumberPeaks {
295    pub ns: Vec<usize>,
296}
297
298impl Kernel for NumberPeaks {
299    fn names(&self) -> Vec<String> {
300        self.ns
301            .iter()
302            .map(|n| format!("number_peaks__n_{n}"))
303            .collect()
304    }
305
306    fn compute(&self, ctx: &GroupContext, out: &mut Vec<f64>) {
307        out.extend(
308            peak_counts(ctx.values(), &self.ns)
309                .into_iter()
310                .map(|c| c as f64),
311        );
312    }
313}

315/// For each `n` in `ns`, the number of positions `j` with `n <= j < len - n`
316/// whose value is strictly greater than its `n` neighbours on each side.
317///
318/// One pass computes each position's support (how many neighbours it beats on
319/// both sides, capped at the largest `n`); the scan stops at the first
320/// neighbour that is not smaller. Comparisons with NaN are false, as in
321/// tsfresh.
322fn peak_counts(x: &[f64], ns: &[usize]) -> Vec<usize> {
323    let len = x.len();
324    let cap = ns.iter().copied().max().unwrap_or(0);
325    let mut counts = vec![0; ns.len()];
326    for j in 0..len {
327        let mut left = 0;
328        while left < cap && left < j && x[j] > x[j - 1 - left] {
329            left += 1;
330        }
331        if left == 0 {
332            continue;
333        }
334        let mut right = 0;
335        while right < left && j + 1 + right < len && x[j] > x[j + 1 + right] {
336            right += 1;
337        }
338        let support = left.min(right);
339        for (count, &n) in counts.iter_mut().zip(ns) {
340            if n <= support && j >= n && j + n < len {
341                *count += 1;
342            }
343        }
344    }
345    counts
