8scalar_kernel! {
9/// First location of maximum feature.
10///
11/// The relative position of the first occurrence of the maximum value, as a
12/// fraction of the series length $n$ (0-based index $i$):
13/// $$ \frac{i}{n}, \quad i = \min\{\,j : x_j = \max_k x_k\,\}. $$
14///
15/// # Output column
16/// `{name}__first_location_of_maximum`
17///
18/// # Edge cases
19/// - Nulls are dropped first; a group with no non-null values gives NaN.
20/// - If the series contains NaN, NaN counts as the maximum (numpy's `np.argmax`
21///   rule), so the result is the position of the first NaN.
22/// - Ties resolve to the first occurrence.
23///
24/// # tsfresh
25/// `feature_calculators.first_location_of_maximum` (v0.21.2).
26    FirstLocationOfMaximum, "first_location_of_maximum", |ctx| np_argmax(ctx.x()).map_or(f64::NAN, |i| i as f64 / ctx.len() as f64)
27}

70        np_argmax(&ctx.x().slice(s![..;-1])).map_or(f64::NAN, |i| 1.0 - i as f64 / ctx.len() as f64)
71    }

98/// Index mass quantile feature.
99///
100/// The relative position at which the cumulative absolute "mass" of the
101/// series first reaches the fraction $q$ of its total:
102/// $$ \frac{i + 1}{n}, \quad i = \min\left\{\, j : \frac{\sum_{k=0}^{j} |x_k|}{\sum_{k=0}^{n-1} |x_k|} \geq q \,\right\}, $$
103/// with 0-based indices and $n$ the number of values.
104///
105/// # Parameters
106/// - `q`: mass fraction in $[0, 1]$. Config:
107///   `[index_mass_quantile] parameters = [{ q = 0.1 }, ...]`; one column per
108///   entry.
109///
110/// # Output column
111/// `{name}__index_mass_quantile__q_{q}` (`q` in Rust's default formatting,
112/// e.g. `q_0.3`, `q_0.99`)
113///
114/// # Edge cases
115/// - Nulls are dropped first; a group with no non-null values gives NaN.
116/// - A series whose values are all 0 has no mass and gives NaN.
117/// - A series containing NaN gives $1/n$, as in tsfresh.
118///
119/// # tsfresh
120/// `feature_calculators.index_mass_quantile` (v0.21.2).
121pub struct IndexMassQuantile {
122    pub qs: Vec<f64>,
123}
124
125impl Kernel for IndexMassQuantile {
126    fn names(&self) -> Vec<String> {
127        self.qs
128            .iter()
129            .map(|q| format!("index_mass_quantile__q_{q:2}"))
130            .collect()
131    }
132
133    fn compute(&self, ctx: &GroupContext, out: &mut Vec<f64>) {
134        // np.cumsum(np.abs(x)) / np.sum(np.abs(x)): a sequential cumulative
135        // sum over numpy's pairwise total
136        let mut mass = ctx.x().mapv(f64::abs);
137        let total = numpy_sum(mass.len(), |i| mass[i]);
138        if total == 0.0 {
139            out.extend(self.qs.iter().map(|_| f64::NAN));
140            return;
141        }
142        mass.accumulate_axis_inplace(Axis(0), |&prev, curr| *curr += prev);
143        let mass = mass.mapv(|x| x / total);
144        for q in &self.qs {
145            let first = mass.iter().position(|x| x >= q).unwrap_or(0);
146            out.push((first + 1) as f64 / ctx.len() as f64);
147        }
148    }
149}
