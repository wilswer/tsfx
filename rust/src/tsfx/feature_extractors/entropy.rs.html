8/// All length-`chunk_size` sliding windows of `x`, one step apart.
9fn _into_subchunks(x: &ArrayView1<f64>, chunk_size: usize) -> Vec<Array1<f64>> {
10    let mut subchunks = Vec::with_capacity(x.len());
11    for chunk in x.axis_windows(Axis(0), chunk_size) {
12        subchunks.push(chunk.to_owned());
13    }
14    subchunks
15}

17/// Number of unordered pairs of templates within Chebyshev distance `r`.
18fn _get_matches(templates: Vec<Array1<f64>>, r: f64) -> usize {
19    let mut matches = 0;
20    for combo in templates.into_iter().combinations(2) {
21        let a = combo[0].to_owned();
22        let b = combo[1].to_owned();
23        let diff = a - b;
24        // tsfresh counts a match at distance <= tolerance
25        let dist_check = diff.mapv(|x| if x.abs() <= r { 1 } else { 0 }).sum();
26        if dist_check == diff.len() {
27            matches += 1;
28        }
29    }
30    matches
31}

34/// Sample entropy feature.
35///
36/// The sample entropy (SampEn) of the time series, a measure of its
37/// complexity, with embedding dimension $m = 2$ and tolerance
38/// $r = 0.2\,\sigma$, where $\sigma$ is the population standard deviation:
39/// $$ \text{SampEn} = -\ln \frac{A}{B}, $$
40/// where $B$ is the number of pairs of length-$m$ windows and $A$ the number
41/// of pairs of length-$(m+1)$ windows whose Chebyshev (maximum) distance is
42/// at most $r$. A regular series scores low, an irregular one high.
43///
44/// # Output column
45/// `{name}__sample_entropy`
46///
47/// # Edge cases
48/// - Unlike the other features, nulls are **not** dropped; a null or NaN
49///   anywhere in the series gives NaN.
50/// - Fewer than 4 values give NaN (no matching pairs of windows).
51/// - When no length-$(m+1)$ windows match, the result is $+\infty$.
52/// - **Deliberate deviation from tsfresh:** a series containing $\pm\infty$
53///   gives NaN. tsfresh's tolerance becomes NaN, every match count turns
54///   negative after subtracting self-matches, and the ratio of two negative
55///   counts returns a finite but meaningless value.
56///
57/// # Cost
58/// $O(n^2)$ in the series length; only computed for
59/// `FeatureSetting::Comprehensive`, where it is always on (the config file
60/// does not switch it off).
61///
62/// # tsfresh
63/// `feature_calculators.sample_entropy` (v0.21.2).
64    SampleEntropy, "sample_entropy", |ctx| {
65        // Nulls count as NaN here, and any NaN makes the result NaN
66        if ctx.has_nulls() {
67            f64::NAN
68        } else {
69            let m = 2;
70            let r = 0.2 * ctx.std();
71            let matches_m = _get_matches(_into_subchunks(ctx.x(), m), r);
72            let matches_m_plus_1 = _get_matches(_into_subchunks(ctx.x(), m + 1), r);
73            ((matches_m as f64) / (matches_m_plus_1 as f64)).ln()
74        }
75    }
76}
