8/// Numbers of unordered pairs of length-`m` and of length-`(m + 1)` sliding
9/// windows of `x` within Chebyshev distance `r` of each other (tsfresh counts
10/// a match at distance <= tolerance).
11///
12/// Both counts come from one pass over the pairs, comparing values in place:
13/// a pair of length-`(m + 1)` windows can only match if its first `m` values
14/// do.
15fn _count_matches(x: &[f64], m: usize, r: f64) -> (usize, usize) {
16    // Numbers of windows of length m and m + 1
17    let n_m = (x.len() + 1).saturating_sub(m);
18    let n_m_plus_1 = x.len().saturating_sub(m);
19    let close = |i: usize, j: usize, k: usize| (x[i + k] - x[j + k]).abs() <= r;
20    let (mut matches_m, mut matches_m_plus_1) = (0, 0);
21    for i in 0..n_m {
22        for j in (i + 1)..n_m {
23            if (0..m).all(|k| close(i, j, k)) {
24                matches_m += 1;
25                if j < n_m_plus_1 && close(i, j, m) {
26                    matches_m_plus_1 += 1;
27                }
28            }
29        }
30    }
31    (matches_m, matches_m_plus_1)
32}

35/// Sample entropy feature.
36///
37/// The sample entropy (SampEn) of the time series, a measure of its
38/// complexity, with embedding dimension $m = 2$ and tolerance
39/// $r = 0.2\,\sigma$, where $\sigma$ is the population standard deviation:
40/// $$ \text{SampEn} = -\ln \frac{A}{B}, $$
41/// where $B$ is the number of pairs of length-$m$ windows and $A$ the number
42/// of pairs of length-$(m+1)$ windows whose Chebyshev (maximum) distance is
43/// at most $r$. A regular series scores low, an irregular one high.
44///
45/// # Output column
46/// `{name}__sample_entropy`
47///
48/// # Edge cases
49/// - Unlike the other features, nulls are **not** dropped; a null or NaN
50///   anywhere in the series gives NaN.
51/// - Fewer than 4 values give NaN (no matching pairs of windows).
52/// - When no length-$(m+1)$ windows match, the result is $+\infty$.
53/// - **Deliberate deviation from tsfresh:** a series containing $\pm\infty$
54///   gives NaN. tsfresh's tolerance becomes NaN, every match count turns
55///   negative after subtracting self-matches, and the ratio of two negative
56///   counts returns a finite but meaningless value.
57///
58/// # Cost
59/// $O(n^2)$ in the series length; only computed for
60/// `FeatureSetting::Comprehensive`, and only with a `[sample_entropy]`
61/// config section (in the default config).
62///
63/// # tsfresh
64/// `feature_calculators.sample_entropy` (v0.21.2).
65    SampleEntropy, "sample_entropy", |ctx| {
66        // Nulls count as NaN here, and any NaN makes the result NaN
67        if ctx.has_nulls() {
68            f64::NAN
69        } else {
70            let m = 2;
71            let r = 0.2 * ctx.std();
72            let (matches_m, matches_m_plus_1) = _count_matches(ctx.values(), m, r);
73            ((matches_m as f64) / (matches_m_plus_1 as f64)).ln()
74        }
75    }
76}

78/// The entropy of the values binned into `max_bins` equal-width bins, as
79/// tsfresh's `binned_entropy`: `np.histogram(values, bins=max_bins)` (bin
80/// edges and the assignment of values to bins reproduced exactly, the last
81/// bin closed), then $-\sum_k p_k \ln p_k$ over the non-empty bins.
82///
83/// NaN if `values` is empty, contains NaN, `max_bins` is 0, or numpy's
84/// histogram raises (an infinite value, or a range too small for
85/// `max_bins` distinct edges).
86pub(crate) fn binned_entropy(values: &[f64], max_bins: usize) -> f64 {
87    if values.is_empty() || max_bins == 0 || values.iter().any(|v| v.is_nan()) {
88        return f64::NAN;
89    }
90    let (mut first, mut last) = extremes(values.iter().copied());
91    // np.histogram: "autodetected range is not finite"
92    if !first.is_finite() || !last.is_finite() {
93        return f64::NAN;
94    }
95    if first == last {
96        first -= 0.5;
97        last += 0.5;
98    }
99    let edges = numpy_linspace(first, last, max_bins + 1);
100    // np.histogram: "Too many bins for data range"
101    if edges.windows(2).any(|w| w[0] >= w[1]) {
102        return f64::NAN;
103    }
104    let denom = last - first;
105    let mut hist = vec![0_usize; max_bins];
106    for &v in values {
107        // numpy's fast path for equal bins, including its ±1 corrections
108        let mut idx = (((v - first) / denom * max_bins as f64) as usize).min(max_bins);
109        if idx == max_bins {
110            idx -= 1;
111        }
112        if idx > 0 && v < edges[idx] {
113            idx -= 1;
114        }
115        if v >= edges[idx + 1] && idx != max_bins - 1 {
116            idx += 1;
117        }
118        hist[idx] += 1;
119    }
120    let n = values.len() as f64;
121    let terms: Vec<f64> = hist
122        .iter()
123        .map(|&count| {
124            let p = count as f64 / n;
125            let p = if p == 0.0 { 1.0 } else { p };
126            p * p.ln()
127        })
128        .collect();
129    -numpy_sum(terms.len(), |i| terms[i])
130}

132/// Binned entropy feature.
133///
134/// Bin the values into `max_bins` equal-width bins between the minimum and
135/// the maximum and return the entropy of the bin frequencies:
136/// $$ -\sum_{k} p_k \ln p_k \cdot \mathbf{1}_{(p_k > 0)}, $$
137/// where $p_k$ is the share of values in bin $k$. Bins follow
138/// `np.histogram` exactly: each bin includes its left edge, the last bin
139/// also its right edge, and a constant series is binned over
140/// $[x - 0.5, x + 0.5]$.
141///
142/// # Parameters
143/// - `max_bins`: number of bins. Config:
144///   `[binned_entropy] parameters = [{ max_bins = 10 }, ...]`; one column per
145///   entry.
146///
147/// # Output column
148/// `{name}__binned_entropy__max_bins_{max_bins}`
149///
150/// # Edge cases
151/// - Nulls are dropped first; a group with no non-null values gives NaN.
152/// - A NaN anywhere gives NaN, as tsfresh.
153/// - `max_bins = 0` gives NaN.
154/// - **Deliberate deviation from tsfresh:** a series containing $\pm\infty$,
155///   or whose range is too small for `max_bins` distinct bin edges, gives
156///   NaN; tsfresh raises numpy's `ValueError` from `np.histogram`.
157///
158/// # tsfresh
159/// `feature_calculators.binned_entropy` (v0.21.2). tsfresh's `np.log`
160/// (numpy's SIMD logarithm) and Rust's `ln` can differ in the last bit, so
161/// results can differ from tsfresh's in the last bits.
162pub struct BinnedEntropy {
163    pub max_bins: Vec<usize>,
164}
165
166impl Kernel for BinnedEntropy {
167    fn names(&self) -> Vec<String> {
168        self.max_bins
169            .iter()
170            .map(|max_bins| format!("binned_entropy__max_bins_{max_bins}"))
171            .collect()
172    }
173
174    fn compute(&self, ctx: &GroupContext, out: &mut Vec<f64>) {
175        let x = ctx.values();
176        out.extend(
177            self.max_bins
178                .iter()
179                .map(|&max_bins| binned_entropy(x, max_bins)),
180        );
181    }
182}

184/// Lempel-Ziv complexity feature.
185///
186/// Bin the values into `bins` equal-width bins between the minimum and the
187/// maximum (bin $k$ holds the values $v$ with $e_k < v \leq e_{k+1}$, as
188/// `np.searchsorted(np.linspace(min, max, bins + 1)[1:], x, side="left")`),
189/// then parse the sequence of bin indices from left to right into the
190/// shortest sub-words not seen before. The complexity is the number of
191/// distinct sub-words divided by the series length; for example "100111"
192/// splits into 1, 0, 01 and 11, giving $4/6$.
193///
194/// # Parameters
195/// - `bins`: number of bins. Config:
196///   `[lempel_ziv_complexity] parameters = [{ bins = 2 }, ...]`; one column per
197///   entry.
198///
199/// # Output column
200/// `{name}__lempel_ziv_complexity__bins_{bins}`
201///
202/// # Edge cases
203/// - Nulls are dropped first; a group with no non-null values gives NaN.
204/// - A NaN makes every bin edge NaN; values are then binned as numpy's
205///   `searchsorted` does with NaN sorted last, exactly as in tsfresh.
206///
207/// # Cost
208/// $O(n^2)$ in the worst case (long repeated sub-words), typically close to
209/// linear.
210///
211/// # tsfresh
212/// `feature_calculators.lempel_ziv_complexity` (v0.21.2).
213pub struct LempelZivComplexity {
214    pub bins: Vec<usize>,
215}
216
217impl Kernel for LempelZivComplexity {
218    fn names(&self) -> Vec<String> {
219        self.bins
220            .iter()
221            .map(|bins| format!("lempel_ziv_complexity__bins_{bins}"))
222            .collect()
223    }
224
225    fn compute(&self, ctx: &GroupContext, out: &mut Vec<f64>) {
226        let x = ctx.values();
227        let (min, max) = (ctx.min(), ctx.max());
228        for &bins in &self.bins {
229            let edges = numpy_linspace(min, max, bins + 1);
230            let sequence = numpy_searchsorted_left(&edges[1..], x);
231            let n = sequence.len();
232            let mut sub_words: HashSet<&[usize]> = HashSet::new();
233            let (mut ind, mut inc) = (0, 1);
234            while ind + inc <= n {
235                let sub_word = &sequence[ind..ind + inc];
236                if sub_words.contains(sub_word) {
237                    inc += 1;
238                } else {
239                    sub_words.insert(sub_word);
240                    ind += inc;
241                    inc = 1;
242                }
243            }
244            out.push(sub_words.len() as f64 / n as f64);
245        }
246    }
247}

249/// Permutation entropy feature.
250///
251/// Cut the series into windows of `dimension` values, starting every `tau`
252/// values, replace each window by the ordinal pattern of its values (the
253/// rank of each value within the window), and return the entropy of the
254/// pattern frequencies, $-\sum_\pi p_\pi \ln p_\pi$ (Bandt & Pompe, 2002).
255///
256/// # Parameters
257/// - `tau`: step between window starts.
258/// - `dimension`: window length.
259///
260/// Config:
261/// `[permutation_entropy] parameters = [{ tau = 1, dimension = 3 }, ...]`; one
262/// column per entry.
263///
264/// # Output column
265/// `{name}__permutation_entropy__tau_{tau}__dimension_{dimension}`
266///
267/// # Edge cases
268/// - Nulls are dropped first; a group with no non-null values gives NaN.
269/// - Fewer than `dimension` values (no complete window) give NaN, as
270///   tsfresh.
271/// - NaN ranks above every number; `-0.0` and `0.0` are equal.
272/// - `dimension < 2` or `tau = 0` give NaN (tsfresh fails an assertion).
273/// - **Deliberate deviation from tsfresh:** equal values within a window
274///   (including two NaN) are ranked by position, as a stable sort does.
275///   tsfresh ranks with `np.argsort`, whose order of ties depends on the
276///   sorting algorithm numpy picks for the CPU (with AVX-512 it is not
277///   stable), so its result for windows with ties varies between machines.
278///
279/// # tsfresh
280/// `feature_calculators.permutation_entropy` (v0.21.2). tsfresh's `np.log`
281/// and Rust's `ln` can differ in the last bit.
282pub struct PermutationEntropy {
283    /// `(tau, dimension)` per config entry.
284    pub params: Vec<(usize, usize)>,
285}
286
287impl Kernel for PermutationEntropy {
288    fn names(&self) -> Vec<String> {
289        self.params
290            .iter()
291            .map(|(tau, dimension)| {
292                format!("permutation_entropy__tau_{tau}__dimension_{dimension}")
293            })
294            .collect()
295    }
296
297    fn compute(&self, ctx: &GroupContext, out: &mut Vec<f64>) {
298        let x = ctx.values();
299        for &(tau, dimension) in &self.params {
300            out.push(permutation_entropy(x, tau, dimension));
301        }
302    }
303}

305fn permutation_entropy(x: &[f64], tau: usize, dimension: usize) -> f64 {
306    if dimension < 2 || tau == 0 || x.len() < dimension {
307        return f64::NAN;
308    }
309    let less = |a: f64, b: f64| a < b || (b.is_nan() && !a.is_nan());
310    let equal = |a: f64, b: f64| a == b || (a.is_nan() && b.is_nan());
311    let windows = (x.len() - dimension) / tau + 1;
312    // The rank of each value of window k
313    let ranks = |k: usize| {
314        let window = &x[k * tau..k * tau + dimension];
315        window.iter().enumerate().map(move |(i, &v)| {
316            window
317                .iter()
318                .enumerate()
319                .filter(|&(j, &w)| less(w, v) || (j < i && equal(w, v)))
320                .count()
321        })
322    };
323    let terms = match (dimension as u64).checked_pow(dimension as u32) {
324        // A pattern as one base-`dimension` number, which orders the
325        // patterns as their rank sequences: no allocation per window
326        Some(_) => pattern_terms(
327            windows,
328            (0..windows).map(|k| ranks(k).fold(0u64, |code, r| code * dimension as u64 + r as u64)),
329        ),
330        None => pattern_terms(windows, (0..windows).map(|k| ranks(k).collect::<Vec<_>>())),
331    };
332    -numpy_sum(terms.len(), |i| terms[i])
333}
334
335/// $p \ln p$ of each pattern's frequency, in increasing pattern order (as
336/// `np.unique(axis=0)` counts them, lexicographically).
337fn pattern_terms<K: Ord>(windows: usize, patterns: impl Iterator<Item = K>) -> Vec<f64> {
338    let mut counts: BTreeMap<K, usize> = BTreeMap::new();
339    for pattern in patterns {
340        *counts.entry(pattern).or_insert(0) += 1;
341    }
342    counts
343        .values()
344        .map(|&count| {
345            let p = count as f64 / windows as f64;
346            p * p.ln()
347        })
348        .collect()
349}

351/// Approximate entropy feature.
352///
353/// The approximate entropy (ApEn) of the series (Pincus, 1991), a measure of
354/// its regularity, with run length `m` and tolerance $r\,\sigma$, where
355/// $\sigma$ is the population standard deviation:
356/// $$ \text{ApEn} = \left| \Phi_m - \Phi_{m+1} \right|, \quad
357///    \Phi_k = \frac{1}{N_k} \sum_{j=1}^{N_k} \ln \frac{C_j^k}{N_k}, $$
358/// where $N_k = n - k + 1$ is the number of length-$k$ windows and $C_j^k$
359/// the number of them (window $j$ itself included) within Chebyshev
360/// (maximum) distance $r\,\sigma$ of window $j$.
361///
362/// # Parameters
363/// - `m`: run length.
364/// - `r`: tolerance as a multiple of the standard deviation.
365///
366/// Config:
367/// `[approximate_entropy] parameters = [{ m = 2, r = 0.1 }, ...]`; one column
368/// per entry.
369///
370/// # Output column
371/// `{name}__approximate_entropy__m_{m}__r_{r}`
372///
373/// # Edge cases
374/// - Nulls are dropped first; a group with no non-null values gives NaN.
375/// - $n \leq m + 1$ gives 0, as tsfresh.
376/// - NaN or $\pm\infty$ values are kept: windows at NaN distance never
377///   match, exactly as in tsfresh (typically giving NaN).
378/// - A negative tolerance $r\,\sigma < 0$ or `m = 0` gives NaN (tsfresh
379///   raises).
380///
381/// # Cost
382/// $O(n^2)$ in the series length; all `r` of one `m` share a single pass
383/// over the pairs of windows. Only computed for
384/// `FeatureSetting::Comprehensive`.
385///
386/// # tsfresh
387/// `feature_calculators.approximate_entropy` (v0.21.2). tsfresh's `np.log`
388/// and Rust's `ln` can differ in the last bit.
389pub struct ApproximateEntropy {
390    /// `(m, r)` per config entry.
391    pub params: Vec<(usize, f64)>,
392}
393
394impl Kernel for ApproximateEntropy {
395    fn names(&self) -> Vec<String> {
396        self.params
397            .iter()
398            .map(|(m, r)| format!("approximate_entropy__m_{m}__r_{r}"))
399            .collect()
400    }
401
402    fn compute(&self, ctx: &GroupContext, out: &mut Vec<f64>) {
403        let x = ctx.values();
404        let n = x.len();
405        let std = ctx.std();
406        let mut results = vec![f64::NAN; self.params.len()];
407        // Parameters that need the pass over the windows, by run length
408        let mut pending: BTreeMap<usize, Vec<(usize, f64)>> = BTreeMap::new();
409        for (k, &(m, r)) in self.params.iter().enumerate() {
410            let r = r * std;
411            if r < 0.0 || m == 0 {
412                continue;
413            }
414            if n <= m + 1 {
415                results[k] = 0.0;
416                continue;
417            }
418            pending.entry(m).or_default().push((k, r));
419        }
420        for (m, params) in pending {
421            let rs: Vec<f64> = params.iter().map(|&(_, r)| r).collect();
422            for (&(k, _), apen) in params.iter().zip(approximate_entropies(x, m, &rs)) {
423                results[k] = apen;
424            }
425        }
426        out.extend(results);
427    }
428}

430/// NaN if either is NaN, else the larger: `np.max` of two values.
431fn nan_max(a: f64, b: f64) -> f64 {
432    if a.is_nan() || b.is_nan() {
433        f64::NAN
434    } else {
435        a.max(b)
436    }
437}

439/// Approximate entropy for run length `m` (with `n > m + 1`) and each
440/// absolute tolerance in `rs`, in one pass over the pairs of windows.
441fn approximate_entropies(x: &[f64], m: usize, rs: &[f64]) -> Vec<f64> {
442    let n = x.len();
443    let (w_m, w_m1) = (n - m + 1, n - m);
444    let mut counts_m = vec![vec![0_u32; w_m]; rs.len()];
445    let mut counts_m1 = vec![vec![0_u32; w_m1]; rs.len()];
446    for i in 0..w_m {
447        for j in i..w_m {
448            let d_m = (0..m).fold(0.0, |d, k| nan_max(d, (x[i + k] - x[j + k]).abs()));
449            for (r, counts) in rs.iter().zip(counts_m.iter_mut()) {
450                if d_m <= *r {
451                    counts[i] += 1;
452                    if i != j {
453                        counts[j] += 1;
454                    }
455                }
456            }
457            if j < w_m1 {
458                let d_m1 = nan_max(d_m, (x[i + m] - x[j + m]).abs());
459                for (r, counts) in rs.iter().zip(counts_m1.iter_mut()) {
460                    if d_m1 <= *r {
461                        counts[i] += 1;
462                        if i != j {
463                            counts[j] += 1;
464                        }
465                    }
466                }
467            }
468        }
469    }
470    let phi = |counts: &[u32]| {
471        let w = counts.len() as f64;
472        numpy_sum(counts.len(), |j| (counts[j] as f64 / w).ln()) / w
473    };
474    counts_m
475        .iter()
476        .zip(&counts_m1)
477        .map(|(c_m, c_m1)| (phi(c_m) - phi(c_m1)).abs())
478        .collect()
479}
