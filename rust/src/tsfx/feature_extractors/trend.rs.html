9/// Linear trend feature.
10///
11/// The intercept $a$ and slope $b$ of the least-squares line through the
12/// values against their 0-based position, $x_i \approx a + b\,i$:
13/// $$ b = \frac{\sum_i (i - \bar{i})(x_i - \bar{x})}{\sum_i (i - \bar{i})^2}, \quad a = \bar{x} - b\,\bar{i}. $$
14/// Only these two of tsfresh's `linear_trend` attributes are computed
15/// (not `pvalue`, `rvalue` or `stderr`).
16///
17/// # Output column
18/// `{name}__linear_trend_intercept` and `{name}__linear_trend_slope`
19///
20/// # Edge cases
21/// - Nulls are dropped first.
22/// - Fewer than 2 values give NaN.
23/// - NaN values are kept and make both results NaN.
24///
25/// # tsfresh
26/// `feature_calculators.linear_trend` (v0.21.2) with `attr` = `"intercept"`
27/// or `"slope"`.
28pub struct LinearTrend;
29
30impl Kernel for LinearTrend {
31    fn names(&self) -> Vec<String> {
32        vec![
33            "linear_trend_intercept".to_string(),
34            "linear_trend_slope".to_string(),
35        ]
36    }
37
38    fn compute(&self, ctx: &GroupContext, out: &mut Vec<f64>) {
39        let (intercept, slope) = calculate_sequential_ols(ctx.values());
40        out.extend([intercept, slope]);
41    }
42}

44/// Aggregated linear trend feature.
45///
46/// Split the series into consecutive chunks of `chunk_size` values (the last
47/// chunk may be shorter), aggregate each chunk to one number, and fit a
48/// least-squares line through the aggregates against the chunk index, as in
49/// [`LinearTrend`]. Returns the intercept and slope of that line.
50///
51/// # Parameters
52/// - `chunk_size`: values per chunk.
53/// - `aggregator`: how each chunk is summarised: `mean`, `median`, `min`,
54///   `max` or `var` (population variance, `ddof = 0`), each exactly as the
55///   numpy method tsfresh's extraction calls on the chunk. `median` is a
56///   tsfx addition (numpy arrays have no `median` method; it follows
57///   `np.median`). Any other name is rejected when the config is loaded.
58///
59/// Config: `[agg_linear_trend] parameters = [{ chunk_size = 5, aggregator = "mean" }, ...]`;
60/// two columns per entry.
61///
62/// # Output column
63/// `{name}__agg_linear_trend_intercept__chunk_size_{chunk_size}__agg_{aggregator}`
64/// and `{name}__agg_linear_trend_slope__chunk_size_{chunk_size}__agg_{aggregator}`
65///
66/// # Edge cases
67/// - Nulls are dropped first. A group with `chunk_size` or fewer values
68///   gives NaN, as tsfresh.
69/// - Fewer than 2 chunks give NaN (no line can be fitted).
70/// - A NaN in a chunk makes that chunk's aggregate NaN (numpy semantics, as
71///   in tsfresh's extraction), and so the fit.
72///
73/// # tsfresh
74/// `feature_calculators.agg_linear_trend` (v0.21.2) with `chunk_len`,
75/// `f_agg` and `attr` = `"intercept"` or `"slope"`, as called by tsfresh's
76/// `extract_features` (with a numpy array, not a pandas Series as in
77/// tsfresh's own unit tests, where `var` uses `ddof = 1` and NaN is skipped).
78pub struct AggLinearTrend {
79    /// `(chunk_size, aggregator)` per config entry.
80    pub params: Vec<(usize, ChunkAggregator)>,
81}
82
83impl Kernel for AggLinearTrend {
84    fn names(&self) -> Vec<String> {
85        self.params
86            .iter()
87            .flat_map(|(chunk_size, aggregator)| {
88                ["intercept", "slope"].map(|attr| {
89                    format!("agg_linear_trend_{attr}__chunk_size_{chunk_size:.1}__agg_{aggregator}")
90                })
91            })
92            .collect()
93    }
94
95    fn compute(&self, ctx: &GroupContext, out: &mut Vec<f64>) {
96        let x = ctx.values();
97        for (chunk_size, aggregator) in &self.params {
98            let chunk_size = *chunk_size;
99            if x.len() <= chunk_size {
100                out.extend([f64::NAN, f64::NAN]);
101                continue;
102            }
103            let aggregate = |chunk: &[f64]| match aggregator {
104                ChunkAggregator::Mean => numpy_mean(chunk),
105                ChunkAggregator::Median => {
106                    if chunk.iter().any(|v| v.is_nan()) {
107                        f64::NAN
108                    } else {
109                        numpy_median(&mut chunk.to_vec())
110                    }
111                }
112                ChunkAggregator::Max => extremes(chunk.iter().copied()).1,
113                ChunkAggregator::Min => extremes(chunk.iter().copied()).0,
114                ChunkAggregator::Var => population_var(&ArrayView1::from(chunk)),
115            };
116            let aggregates = x.chunks(chunk_size).map(aggregate).collect::<Vec<f64>>();
117            let (intercept, slope) = calculate_sequential_ols(&aggregates);
118            out.extend([intercept, slope]);
119        }
120    }
121}
