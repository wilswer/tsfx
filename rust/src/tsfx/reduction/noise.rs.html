1//! Step 4: noise. The Laplacian score (He, Cai and Niyogi, NeurIPS 2005) of
2//! each feature on a nearest-neighbor graph of the rows, compared with the
3//! scores of shuffled "shadow" features scored in the same graph. A feature
4//! that varies no less between neighboring rows than the best shadow carries
5//! no neighbor structure: it is noise.
6

13/// Rows per block in the parallel nearest-neighbor search: each block holds a
14/// 64 × n matrix, so the memory in use is about threads · 64 · n floats.
15const ROW_BLOCK: usize = 64;
16
17/// For each row of `m`, its `k` (at least 1) nearest other rows with their
18/// squared Euclidean distances, sorted by (distance, index): equal distances
19/// go to the lower index. Distances come from ‖a‖² + ‖b‖² − 2a·b; one at most
20/// 1e-12 · (‖a‖² + ‖b‖²) is rounding noise of (nearly) equal rows and counts
21/// as exactly 0. Parallel over blocks of 64 rows; each block's distances to
22/// all rows come from one single-threaded matrix product, so no n × n matrix
23/// is held and the result does not depend on the thread count.
24pub(crate) fn knn(m: &Array2<f64>, k: usize) -> Vec<Vec<(usize, f64)>> {
25    debug_assert!(k >= 1, "knn needs at least one neighbor");
26    let n = m.nrows();
27    let norms: Vec<f64> = m.rows().into_iter().map(|row| row.dot(&row)).collect();
28    let starts: Vec<usize> = (0..n).step_by(ROW_BLOCK).collect();
29    let blocks: Vec<Vec<Vec<(usize, f64)>>> = starts
30        .par_iter()
31        .map(|&start| {
32            let end = (start + ROW_BLOCK).min(n);
33            let gram = m.slice(s![start..end, ..]).dot(&m.t());
34            (start..end)
35                .map(|i| nearest(i, gram.row(i - start), &norms, k))
36                .collect()
37        })
38        .collect();
39    blocks.into_iter().flatten().collect()
40}
41
42/// The `k` nearest rows of row `i`, from its row of the Gram matrix.
43fn nearest(i: usize, gram_row: ArrayView1<f64>, norms: &[f64], k: usize) -> Vec<(usize, f64)> {
44    let by_distance = |a: &(usize, f64), b: &(usize, f64)| a.1.total_cmp(&b.1).then(a.0.cmp(&b.0));
45    let mut candidates: Vec<(usize, f64)> = (0..norms.len())
46        .filter(|&j| j != i)
47        .map(|j| {
48            let scale = norms[i] + norms[j];
49            let d2 = scale - 2.0 * gram_row[j];
50            // Rounding noise where rows are (nearly) equal, of either sign:
51            // snap it to 0, so equal rows tie exactly
52            (j, if d2 <= 1e-12 * scale { 0.0 } else { d2 })
53        })
54        .collect();
55    if k < candidates.len() {
56        candidates.select_nth_unstable_by(k - 1, by_distance);
57        candidates.truncate(k);
58    }
59    candidates.sort_by(by_distance);
60    candidates
61}

63/// The Laplacian score of every column of `m` on the graph of the neighbor
64/// lists `nn` (from [`knn`]): edges are the union of the neighbor pairs, with
65/// weight `exp(-d² / t)`, `t` the lower median of all neighbor distances (1.0
66/// if that is 0). With degrees `D` and `L = D − W`, the score of a column `f`
67/// is `f̃ᵀ L f̃ / f̃ᵀ D f̃`, `f̃ = f − (fᵀD1 / 1ᵀD1) 1`: lower means the column
68/// varies less between neighbors. NaN for a constant column. Parallel over
69/// columns; every sum runs in a fixed order.
70pub(crate) fn laplacian_scores(m: &Array2<f64>, nn: &[Vec<(usize, f64)>]) -> Vec<f64> {
71    let n = m.nrows();
72    let mut distances: Vec<f64> = nn.iter().flatten().map(|&(_, d2)| d2).collect();
73    distances.sort_by(f64::total_cmp);
74    let t = match distances.get(distances.len().saturating_sub(1) / 2) {
75        Some(&median) if median > 0.0 => median,
76        _ => 1.0,
77    };
78
79    // Each undirected edge once, as (lower, upper, weight); a stable sort
80    // keeps the copy found from the lower row first
81    let mut edges: Vec<(usize, usize, f64)> = nn
82        .iter()
83        .enumerate()
84        .flat_map(|(i, row)| row.iter().map(move |&(j, d2)| (i.min(j), i.max(j), d2)))
85        .collect();
86    edges.sort_by_key(|edge| (edge.0, edge.1));
87    edges.dedup_by(|later, first| (later.0, later.1) == (first.0, first.1));
88    for edge in edges.iter_mut() {
89        edge.2 = (-edge.2 / t).exp();
90    }
91    let mut degree = vec![0.0; n];
92    for &(i, j, w) in &edges {
93        degree[i] += w;
94        degree[j] += w;
95    }
96    let total_degree: f64 = degree.iter().sum();
97
98    (0..m.ncols())
99        .into_par_iter()
100        .map(|c| {
101            // A contiguous copy: the edge loop reads it in random order
102            let f = m.column(c).to_vec();
103            if f.iter().all(|&x| x == f[0]) {
104                // The weighted mean can miss a constant by an ulp, which
105                // would score it 0, the best score
106                return f64::NAN;
107            }
108            let mean = degree.iter().zip(&f).map(|(d, x)| d * x).sum::<f64>() / total_degree;
109            // Centering cancels in the differences of the numerator
110            let numerator: f64 = edges
111                .iter()
112                .map(|&(i, j, w)| w * (f[i] - f[j]).powi(2))
113                .sum();
114            let denominator: f64 = degree
115                .iter()
116                .zip(&f)
117                .map(|(d, x)| d * (x - mean).powi(2))
118                .sum();
119            if denominator > 0.0 {
120                numerator / denominator
121            } else {
122                f64::NAN
123            }
124        })
125        .collect()
126}

128/// The noise step's scores.
129#[derive(Clone, Debug)]
130pub(crate) struct NoiseResult {
131    /// Laplacian score of each column of the input matrix.
132    pub scores: Vec<f64>,
133    /// The lowest shadow score: a feature is kept only if its score is below.
134    pub cutoff: f64,
135}
136
137/// Scores the columns of the rank matrix `z` against `n_shadows` shadow
138/// columns, each a permutation (from `rng`, in shadow order) of the
139/// standardized ranks of a tie-free column, which is what any noise column
140/// becomes after the rank transform. The graph is built on the real and the
141/// shadow columns together, with `neighbors` neighbors per row, so that a
142/// pure-noise feature and the shadows are interchangeable: it beats them all
143/// with probability 1 / (shadows + 1).
144pub(crate) fn noise_scores(
145    z: &Array2<f64>,
146    n_shadows: usize,
147    neighbors: usize,
148    rng: &mut SplitMix64,
149) -> NoiseResult {
150    let (n, d) = z.dim();
151    let positions: Vec<f64> = (0..n).map(|i| i as f64).collect();
152    let all_rows: Vec<usize> = (0..n).collect();
153    let base = rank_matrix(&[&positions], &all_rows).column(0).to_vec();
154    let mut m = Array2::zeros((n, d + n_shadows));
155    m.slice_mut(s![.., ..d]).assign(z);
156    for shadow in 0..n_shadows {
157        let mut column = base.clone();
158        rng.shuffle(&mut column);
159        m.column_mut(d + shadow)
160            .assign(&ArrayView1::from(column.as_slice()));
161    }
162    let scores = laplacian_scores(&m, &knn(&m, neighbors));
163    let cutoff = scores[d..].iter().copied().fold(f64::INFINITY, f64::min);
164    NoiseResult {
165        scores: scores[..d].to_vec(),
166        cutoff,
167    }
168}

178        (g.next_u64() >> 11) as f64 / (1u64 << 53) as f64

254    /// Dense references: neighbors from all pairwise distances computed
255    /// directly, and scores from full W, D and L matrices.
256    fn dense_knn(m: &Array2<f64>, k: usize) -> Vec<Vec<(usize, f64)>> {
257        (0..m.nrows())
258            .map(|i| {
259                let mut all: Vec<(usize, f64)> = (0..m.nrows())
260                    .filter(|&j| j != i)
261                    .map(|j| {
262                        let diff = &m.row(i) - &m.row(j);
263                        (j, diff.dot(&diff))
264                    })
265                    .collect();
266                all.sort_by(|a, b| a.1.total_cmp(&b.1).then(a.0.cmp(&b.0)));
267                all.truncate(k);
268                all
269            })
270            .collect()
271    }

273    fn dense_scores(m: &Array2<f64>, nn: &[Vec<(usize, f64)>]) -> Vec<f64> {
274        let n = m.nrows();
275        let mut d2: Vec<f64> = nn.iter().flatten().map(|p| p.1).collect();
276        d2.sort_by(f64::total_cmp);
277        let t = match d2[(d2.len() - 1) / 2] {
278            x if x > 0.0 => x,
279            _ => 1.0,
280        };
281        let mut w = Array2::<f64>::zeros((n, n));
282        for (i, row) in nn.iter().enumerate() {
283            for &(j, dist) in row {
284                let (a, b) = (i.min(j), i.max(j));
285                if w[[a, b]] == 0.0 {
286                    let weight = (-dist / t).exp();
287                    w[[a, b]] = weight;
288                    w[[b, a]] = weight;
289                }
290            }
291        }
292        let deg = w.sum_axis(ndarray::Axis(1));
293        let lap = Array2::from_diag(&deg) - &w;
294        m.columns()
295            .into_iter()
296            .map(|f| {
297                let mean = deg.dot(&f) / deg.sum();
298                let ft = f.mapv(|x| x - mean);
299                let den = (&ft * &deg).dot(&ft);
300                if den > 0.0 {
301                    ft.dot(&lap.dot(&ft)) / den
302                } else {
303                    f64::NAN
304                }
305            })
306            .collect()
307    }

309    #[test]
310    fn test_knn_and_scores_match_dense_reference() {
311        let mut case = 0;
312        // 63, 64, 65 and 300 rows cross the 64-row blocks of knn
313        for n in [3, 11, 63, 64, 65, 300] {
314            for d in [1, 3, 5, 8, 2] {
315                case += 1;
316                let mut g = SplitMix64::new(1000 + case);
317                let mut m = Array2::from_shape_fn((n, d), |_| uniform(&mut g));
318                if case % 3 == 0 {
319                    // duplicated rows: exact ties
320                    for i in (1..n).step_by(4) {
321                        let row = m.row(i - 1).to_owned();
322                        m.row_mut(i).assign(&row);
323                    }
324                }
325                let k = 10.min(n - 1);
326                let got = knn(&m, k);
327                let want = dense_knn(&m, k);
328                for (row_got, row_want) in got.iter().zip(&want) {
329                    let idx_got: Vec<usize> = row_got.iter().map(|p| p.0).collect();
330                    let idx_want: Vec<usize> = row_want.iter().map(|p| p.0).collect();
331                    assert_eq!(idx_got, idx_want, "case {case} (n {n}, d {d})");
332                    for (a, b) in row_got.iter().zip(row_want) {
333                        assert!((a.1 - b.1).abs() <= 1e-9 * (1.0 + b.1), "case {case}");
334                    }
335                }
336                let scores = laplacian_scores(&m, &got);
337                let dense = dense_scores(&m, &got);
338                for (a, b) in scores.iter().zip(&dense) {
339                    assert!(
340                        (a.is_nan() && b.is_nan()) || (a - b).abs() <= 1e-10 * b.abs() + 1e-15,
341                        "case {case}: {a} vs {b}"
342                    );
343                }
344            }
345        }
346    }

348    /// Rare-flag noise (0/1 columns with about 10% ones) among informative
349    /// columns, with average ranks and with ties broken at random: how many
350    /// of 200 (20 columns, 10 seeds) beat every shadow, and whether a boolean
351    /// aligned with the clusters does in every seed.
352    fn rare_flag_noise_kept(tie_break: bool) -> (usize, bool) {
353        let (n, informative, flags) = (300, 40, 20);
354        let (mut kept, mut aligned_always) = (0, true);
355        for seed in 0..10 {
356            let mut g = SplitMix64::new(700 + seed);
357            let mut cols: Vec<Vec<f64>> = (0..informative)
358                .map(|_| {
359                    (0..n)
360                        .map(|i| (i % 3) as f64 + 0.3 * uniform(&mut g))
361                        .collect()
362                })
363                .collect();
364            cols.push((0..n).map(|i| f64::from(i % 3 == 0)).collect());
365            for _ in 0..flags {
366                cols.push((0..n).map(|_| f64::from(uniform(&mut g) < 0.1)).collect());
367            }
368            let refs: Vec<&[f64]> = cols.iter().map(|c| c.as_slice()).collect();
369            let rows: Vec<usize> = (0..n).collect();
370            let z = if tie_break {
371                let seeds: Vec<u64> = (0..refs.len() as u64).map(|j| 1000 * seed + j).collect();
372                tie_broken_rank_matrix(&refs, &rows, &seeds)
373            } else {
374                rank_matrix(&refs, &rows)
375            };
376            let r = noise_scores(&z, 99, 10, &mut SplitMix64::new(seed));
377            aligned_always &= r.scores[informative] < r.cutoff;
378            kept += r.scores[informative + 1..]
379                .iter()
380                .filter(|s| **s < r.cutoff)
381                .count();
382        }
383        (kept, aligned_always)
384    }

386    /// With ties broken at random, rare-flag noise gets through about as
387    /// rarely as continuous noise (1%, ~2 of 200); with average ranks its tie
388    /// blocks let it beat the shadows far more often (a numpy check of this
389    /// setup: 32 of 200), which this test also confirms, so that it detects a
390    /// return to average ranks. A cluster-aligned boolean is kept either way.
391    #[test]
392    fn test_discrete_noise_needs_tie_breaking() {
393        let (with_ties_broken, aligned) = rare_flag_noise_kept(true);
394        assert!(
395            with_ties_broken <= 8,
396            "{with_ties_broken} of 200 flags kept"
397        );
398        assert!(aligned, "the cluster-aligned boolean was dropped");
399        let (with_average_ranks, _) = rare_flag_noise_kept(false);
400        assert!(
401            with_average_ranks >= 16,
402            "only {with_average_ranks} of 200 flags kept with average ranks: \
403             the setup no longer shows why ties are broken"
404        );
405    }
