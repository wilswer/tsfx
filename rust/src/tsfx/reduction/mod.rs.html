1//! Reduction of the extracted features before scaling: drop features that are
2//! mostly missing, almost always one value, near-duplicates of an earlier
3//! feature, or noise (no better than shuffled "shadow" features at respecting
4//! the neighbor structure of the rows).
5//!
6//! The steps decide which columns remain; the values of the columns kept are
7//! not changed. Missing means null, NaN or ±inf. Thresholds are strict: a
8//! feature is dropped when its value is above the threshold.
9

26/// With fewer features than this reaching the noise step, the fit warns: the
27/// shadow features then make up much of the distance between rows, and the
28/// step drops real features by mistake.
29pub const NOISE_WARN_MIN_FEATURES: usize = 50;
30
31/// With fewer rows than this in the noise step, the fit warns: the neighbor
32/// graph is then coarse, and the step drops many real features as noise (in
33/// tests, features of medium strength: 45% at 20 rows, 6% at 40, 2% at 60).
34pub const NOISE_WARN_MIN_ROWS: usize = 50;
35
36/// Fewest rows (groups or windows) any reduction is fitted on.
37const MIN_ROWS: usize = 10;
38
39/// Smallest `noise_sensitivity`: 1 shadow feature.
40const MIN_NOISE_SENSITIVITY: f64 = 0.5;
41
42/// Largest `noise_sensitivity`: 999 shadow features.
43const MAX_NOISE_SENSITIVITY: f64 = 0.999;
44
45/// Settings of the feature reduction. Each step is turned off with `None`.
46#[derive(Clone, Debug, PartialEq)]
47pub struct Reduction {
48    /// Drop a feature whose share of missing values is above this, in [0, 1).
49    pub max_missing: Option<f64>,
50    /// Drop a feature whose most frequent value covers more than this share
51    /// of its non-missing values, in (0, 1).
52    pub max_top_share: Option<f64>,
53    /// Drop a feature whose |Spearman correlation| with an earlier kept
54    /// feature is above this, in (0, 1).
55    pub max_correlation: Option<f64>,
56    /// About this share of pure-noise features is dropped by the noise step,
57    /// in [0.5, 0.999].
58    pub noise_sensitivity: Option<f64>,
59    /// Nearest rows per row in the noise step's neighbor graph, at least 1.
60    pub noise_neighbors: usize,
61    /// Row cap for the correlation and noise steps, above `noise_neighbors`.
62    pub max_samples: usize,
63    /// Seed of the row sample and of the noise step (its tie-breaking and
64    /// shadow features).
65    pub seed: u64,
66}
67
68impl Default for Reduction {
69    fn default() -> Self {
70        Reduction {
71            max_missing: Some(0.05),
72            max_top_share: Some(0.95),
73            max_correlation: Some(0.95),
74            noise_sensitivity: Some(0.99),
75            noise_neighbors: 10,
76            max_samples: 5000,
77            seed: 0,
78        }
79    }

82impl Reduction {
83    /// Checks every setting's range; the error names the field and its value.
84    pub fn validate(&self) -> Result<(), ExtractionError> {
85        let invalid = |name: &str, range: &str, value: f64| {
86            Err(ExtractionError::InvalidSettings(format!(
87                "ReductionSetting {name} must be in {range}, got {value}"
88            )))
89        };
90        // NaN fails every range
91        let open_unit = |v: f64| v > 0.0 && v < 1.0;
92        match self.max_missing {
93            Some(v) if !(0.0..1.0).contains(&v) => return invalid("max_missing", "[0, 1)", v),
94            _ => {}
95        }
96        match self.max_top_share {
97            Some(v) if !open_unit(v) => return invalid("max_top_share", "(0, 1)", v),
98            _ => {}
99        }
100        match self.max_correlation {
101            Some(v) if !open_unit(v) => return invalid("max_correlation", "(0, 1)", v),
102            _ => {}
103        }
104        match self.noise_sensitivity {
105            Some(v) if !(MIN_NOISE_SENSITIVITY..=MAX_NOISE_SENSITIVITY).contains(&v) => {
106                return invalid("noise_sensitivity", "[0.5, 0.999]", v);
107            }
108            _ => {}
109        }
110        if self.noise_neighbors < 1 {
111            return Err(ExtractionError::InvalidSettings(
112                "ReductionSetting noise_neighbors must be at least 1, got 0".to_string(),
113            ));
114        }
115        if self.max_samples <= self.noise_neighbors {
116            return Err(ExtractionError::InvalidSettings(format!(
117                "ReductionSetting max_samples must be larger than noise_neighbors ({}), got {}",
118                self.noise_neighbors, self.max_samples
119            )));
120        }
121        Ok(())
122    }
123
124    /// Number of shadow features of the noise step, `None` without it: a
125    /// pure-noise feature beats all of them with probability
126    /// 1 / (shadows + 1) = 1 - `noise_sensitivity`.
127    pub fn shadow_count(&self) -> Option<usize> {
128        self.noise_sensitivity.map(|s| {
129            ((1.0 / (1.0 - s)).round() as usize)
130                .saturating_sub(1)
131                .max(1)
132        })
133    }
134
135    /// Fewest rows (groups or windows) this reduction is fitted on.
136    pub fn min_rows(&self) -> usize {
137        match self.noise_sensitivity {
138            Some(_) => MIN_ROWS.max(self.noise_neighbors + 1),
139            None => MIN_ROWS,
140        }
141    }
142
143    /// Decides which feature columns of `features` to keep: every column not
144    /// in `key_cols` (the grouping and time columns) is a feature. Steps 3
145    /// and 4 read the rows in the order of `key_cols`, so the decision does
146    /// not depend on the input row order as long as the keys identify the
147    /// rows, as they do for extracted features (one row per group or window).
148    /// Memory: the shares of steps 1 and 2 are computed column by column, and
149    /// steps 3 and 4 hold only the sampled rows (at most `max_samples`) of the
150    /// features left, plus step 3's correlation matrix: 8 · d² bytes for the d
151    /// features left after step 2 (about 1.8 GB at 15,000 features). Errors on
152    /// invalid settings and on fewer rows than
153    /// [`Reduction::min_rows`]: the reduction is fitted across the rows.
154    pub fn fit(
155        &self,
156        features: &DataFrame,
157        key_cols: &[String],
158    ) -> Result<ReductionFit, ExtractionError> {
159        self.validate()?;
160        let n = features.height();
161        let min_rows = self.min_rows();
162        if n < min_rows {
163            return Err(ExtractionError::InvalidSettings(format!(
164                "feature reduction needs at least {min_rows} rows (groups or windows) to fit, \
165                 got {n}; it is fitted across the extracted rows, so it cannot be fitted on \
166                 a single series"
167            )));
168        }
169        let names: Vec<String> = features
170            .get_column_names()
171            .into_iter()
172            .filter(|name| !key_cols.iter().any(|key| key == name.as_str()))
173            .map(|name| name.to_string())
174            .collect();
175        let d = names.len();
176        let mut dropped: Vec<Option<Dropped>> = vec![None; d];
177
178        // Steps 1 and 2 do not depend on the row order: per column, on all
179        // rows, in parallel, keeping only the shares (not the values)
180        let shares: Vec<(f64, Option<f64>)> = names
181            .par_iter()
182            .map(|name| {
183                let values = column_values(features, name)?;
184                let missing = missing_share(&values);
185                let reaches_step_2 = self.max_missing.is_none_or(|max| missing <= max);
186                let top =
187                    (self.max_top_share.is_some() && reaches_step_2).then(|| top_share(&values));
188                Ok((missing, top))
189            })
190            .collect::<PolarsResult<_>>()?;
191        // Step 1: mostly missing
192        if let Some(max) = self.max_missing {
193            for (drop, (missing, _)) in dropped.iter_mut().zip(&shares) {
194                if *missing > max {
195                    *drop = Some(Dropped::Missing);
196                }
197            }
198        }
199        // Step 2: almost one value
200        if let Some(max) = self.max_top_share {
201            for (drop, (_, top)) in dropped.iter_mut().zip(&shares) {
202                if top.is_some_and(|share| share > max) {
203                    *drop = Some(Dropped::OneValue);
204                }
205            }
206        }
207
208        // Steps 3 and 4 run on at most max_samples rows, in the order of the
209        // key columns; the generator draws the sample first, then the noise
210        // step's tie-breaking seeds and shadows
211        let order = key_order(features, key_cols)?;
212        let mut rng = SplitMix64::new(self.seed);
213        let rows: Vec<usize> = if n > self.max_samples {
214            rng.sample_sorted(n, self.max_samples)
215                .into_iter()
216                .map(|position| order[position])
217                .collect()
218        } else {
219            order
220        };
221        // Only the sampled rows of the features left after step 2 are read
222        let candidates = remaining(&dropped);
223        let sampled: Vec<Vec<f64>> =
224            if self.max_correlation.is_some() || self.noise_sensitivity.is_some() {
225                candidates
226                    .par_iter()
227                    .map(|&j| {
228                        let values = column_values(features, &names[j])?;
229                        Ok(rows.iter().map(|&i| values[i]).collect())
230                    })
231                    .collect::<PolarsResult<_>>()?
232            } else {
233                Vec::new()
234            };
235        let mut position: Vec<Option<usize>> = vec![None; d];
236        for (k, &j) in candidates.iter().enumerate() {
237            position[j] = Some(k);
238        }
239        let sampled_columns = |subset: &[usize]| -> Vec<&[f64]> {
240            subset
241                .iter()
242                .map(|&j| sampled[position[j].expect("read after step 2")].as_slice())
243                .collect()
244        };
245        let sample_rows: Vec<usize> = (0..rows.len()).collect();
246
247        // Step 3: near-duplicates
248        let mut duplicate_of: Vec<Option<(usize, f64)>> = vec![None; d];
249        if let Some(max) = self.max_correlation {
250            let alive = remaining(&dropped);
251            if !alive.is_empty() {
252                let corr = correlation_matrix(&rank_matrix(&sampled_columns(&alive), &sample_rows));
253                // Extraction order decides who is kept (the follow-up with an
254                // outcome label orders by relevance instead)
255                let order: Vec<usize> = (0..alive.len()).collect();
256                let duplicates = greedy_duplicates(&corr, &order, max);
257                for (&j, duplicate) in alive.iter().zip(duplicates) {
258                    if let Some(duplicate) = duplicate {
259                        dropped[j] = Some(Dropped::Duplicate);
260                        duplicate_of[j] = Some((alive[duplicate.of], duplicate.correlation));
261                    }
262                }
263            }
264        }
265
266        // Step 4: noise
267        let mut noise_score: Vec<Option<f64>> = vec![None; d];
268        let mut noise_cutoff: Option<f64> = None;
269        let mut warnings = Vec::new();
270        if let Some(n_shadows) = self.shadow_count() {
271            let alive = remaining(&dropped);
272            if !alive.is_empty() {
273                if alive.len() < NOISE_WARN_MIN_FEATURES {
274                    warnings.push(format!(
275                        "the noise step ran on {} features (fewer than \
276                         {NOISE_WARN_MIN_FEATURES}): with few features it may drop real \
277                         features as noise; consider a lower noise_sensitivity, or \
278                         noise_sensitivity=None",
279                        alive.len()
280                    ));
281                }
282                if rows.len() < NOISE_WARN_MIN_ROWS {
283                    warnings.push(format!(
284                        "the noise step ran on {} rows (fewer than {NOISE_WARN_MIN_ROWS}): with \
285                         few rows it drops many real features as noise; consider more groups \
286                         or windows (or a larger max_samples), a lower noise_sensitivity, or \
287                         noise_sensitivity=None",
288                        rows.len()
289                    ));
290                }
291                let columns = sampled_columns(&alive);
292                // Ties broken at random, one seed per column, drawn after the
293                // row sample and before the shadows: a discrete noise feature
294                // is then as random as a shadow (with average ranks, its tie
295                // blocks let it beat the shadows far more often)
296                let seeds: Vec<u64> = columns.iter().map(|_| rng.next_u64()).collect();
297                let result = noise_scores(
298                    &tie_broken_rank_matrix(&columns, &sample_rows, &seeds),
299                    n_shadows,
300                    self.noise_neighbors,
301                    &mut rng,
302                );
303                for (&j, &score) in alive.iter().zip(&result.scores) {
304                    noise_score[j] = Some(score);
305                    // NaN is never below the cutoff
306                    let beats_shadows = score < result.cutoff;
307                    if !beats_shadows {
308                        dropped[j] = Some(Dropped::Noise);
309                    }
310                }
311                noise_cutoff = Some(result.cutoff);
312            }
313        }
314
315        let kept: Vec<String> = names
316            .iter()
317            .zip(&dropped)
318            .filter(|(_, drop)| drop.is_none())
319            .map(|(name, _)| name.clone())
320            .collect();
321        let kept_flags: Vec<bool> = dropped.iter().map(Option::is_none).collect();
322        let dropped_by: Vec<Option<&str>> =
323            dropped.iter().map(|drop| drop.map(Dropped::name)).collect();
324        let missing_share: Vec<Option<f64>> = shares
325            .iter()
326            .map(|&(missing, _)| self.max_missing.map(|_| missing))
327            .collect();
328        let top: Vec<Option<f64>> = shares.iter().map(|&(_, top)| top).collect();
329        let duplicate_names: Vec<Option<&str>> = duplicate_of
330            .iter()
331            .map(|dup| dup.map(|(k, _)| names[k].as_str()))
332            .collect();
333        let correlation: Vec<Option<f64>> =
334            duplicate_of.iter().map(|dup| dup.map(|(_, r)| r)).collect();
335        let cutoff: Vec<Option<f64>> = noise_score
336            .iter()
337            .map(|score| score.and(noise_cutoff))
338            .collect();
339        let report = df![
340            "feature" => names.as_slice(),
341            "kept" => kept_flags,
342            "dropped_by" => dropped_by,
343            "missing_share" => missing_share,
344            "top_share" => top,
345            "duplicate_of" => duplicate_names,
346            "correlation" => correlation,
347            "noise_score" => noise_score,
348            "noise_cutoff" => cutoff,
349        ]?;
350        Ok(ReductionFit {
351            kept,
352            report,
353            warnings,
354        })
355    }
356}

358/// Which step dropped a feature.
359#[derive(Clone, Copy, Debug, PartialEq, Eq)]
360enum Dropped {
361    Missing,
362    OneValue,
363    Duplicate,
364    Noise,
365}
366
367impl Dropped {
368    /// The name in the report's `dropped_by` column.
369    fn name(self) -> &'static str {
370        match self {
371            Dropped::Missing => "missing",
372            Dropped::OneValue => "one_value",
373            Dropped::Duplicate => "duplicate",
374            Dropped::Noise => "noise",
375        }
376    }
377}

379/// Positions of the rows of `features` in the order of `key_cols` (nulls
380/// last; the input order without key columns).
381fn key_order(features: &DataFrame, key_cols: &[String]) -> PolarsResult<Vec<usize>> {
382    if key_cols.is_empty() {
383        return Ok((0..features.height()).collect());
384    }
385    const ROW: &str = "__tsfx_row";
386    let keys = features
387        .select(key_cols.iter().map(String::as_str))?
388        .with_row_index(ROW.into(), None)?
389        .sort(
390            key_cols,
391            SortMultipleOptions::default()
392                .with_nulls_last(true)
393                .with_maintain_order(true),
394        )?;
395    Ok(keys
396        .column(ROW)?
397        .idx()?
398        .into_no_null_iter()
399        .map(|i| i as usize)
400        .collect())
401}
402
403/// Indices of the features no step has dropped yet.
404fn remaining(dropped: &[Option<Dropped>]) -> Vec<usize> {
405    (0..dropped.len())
406        .filter(|&j| dropped[j].is_none())
407        .collect()
408}

410/// A fitted reduction: which feature columns to keep and why. Plain data, so
411/// it can be kept and applied to other frames with the same columns.
412#[derive(Clone, Debug)]
413pub struct ReductionFit {
414    /// The kept feature columns, in the frame's column order.
415    pub kept: Vec<String>,
416    /// One row per feature column, in the frame's column order: `feature`,
417    /// `kept`, `dropped_by` (`missing`, `one_value`, `duplicate`, `noise` or
418    /// null), `missing_share`, `top_share`, `duplicate_of`, `correlation`
419    /// (signed Spearman ρ with `duplicate_of`), `noise_score` and
420    /// `noise_cutoff`. A step that is turned off, or that a feature did not
421    /// reach, leaves its columns null.
422    pub report: DataFrame,
423    /// Warnings of the fit: too few features or rows for the noise step.
424    pub warnings: Vec<String>,
425}
426
427impl ReductionFit {
428    /// `features` with only the `key_cols`, then the kept feature columns (in
429    /// the order they were fitted), dtypes unchanged. A missing column is an
430    /// error.
431    pub fn apply(&self, features: DataFrame, key_cols: &[String]) -> PolarsResult<DataFrame> {
432        features.select(key_cols.iter().chain(&self.kept).map(String::as_str))
433    }
434}

769    /// The report's columns as comparable values: floats as bit patterns
770    /// (any NaN as one value), others as strings.
771    fn report_bits(df: &DataFrame) -> Vec<Vec<Option<String>>> {
772        df.columns()
773            .iter()
774            .map(|c| match c.dtype() {
775                DataType::Float64 => c
776                    .f64()
777                    .unwrap()
778                    .iter()
779                    .map(|v| v.map(|x| if x.is_nan() { u64::MAX } else { x.to_bits() }.to_string()))
780                    .collect(),
781                _ => c
782                    .cast(&DataType::String)
783                    .unwrap()
784                    .str()
785                    .unwrap()
786                    .iter()
787                    .map(|s| s.map(str::to_string))
788                    .collect(),
789            })
790            .collect()
791    }

793    /// Same kept columns and a bit-identical report for 1 and 4 threads and
794    /// any row order, on the sampled path (120 rows > max_samples = 50).
795    #[test]
796    fn test_deterministic_across_threads_and_row_order() {
797        let (n, d) = (120, 60);
798        let mut g = SplitMix64::new(9);
799        let mut columns = vec![Column::new("id".into(), (0..n as i64).collect::<Vec<_>>())];
800        for j in 0..d {
801            let values: Vec<f64> = (0..n)
802                .map(|i| {
803                    let u = uniform(&mut g);
804                    if j % 3 == 0 {
805                        (i % 4) as f64 + u
806                    } else if j % 7 == 1 {
807                        // discrete: the noise step breaks its ties at random
808                        f64::from(u < 0.3)
809                    } else {
810                        u
811                    }
812                })
813                .collect();
814            columns.push(Column::new(format!("f{j}").into(), values));
815        }
816        let df = DataFrame::new_infer_height(columns).unwrap();
817        let mut order: Vec<IdxSize> = (0..n as IdxSize).collect();
818        SplitMix64::new(10).shuffle(&mut order);
819        let shuffled = df.take(&IdxCa::from_vec("idx".into(), order)).unwrap();
820
821        let r = Reduction {
822            noise_neighbors: 5,
823            max_samples: 50,
824            ..Default::default()
825        };
826        let fit_in = |threads: usize, frame: &DataFrame| {
827            rayon::ThreadPoolBuilder::new()
828                .num_threads(threads)
829                .build()
830                .unwrap()
831                .install(|| r.fit(frame, &keys()).unwrap())
832        };
833        let one = fit_in(1, &df);
834        let four = fit_in(4, &shuffled);
835        // Not vacuous: features reached the noise step
836        assert!(one.report.column("noise_score").unwrap().null_count() < d);
837        assert_eq!(one.kept, four.kept);
838        assert_eq!(report_bits(&one.report), report_bits(&four.report));
839        assert_eq!(one.warnings, four.warnings);
840    }

842    /// A constant and an all-missing feature that reach the noise step (the
843    /// earlier steps off) score NaN and are dropped as noise. Regression:
844    /// random tie-breaking turned them into random permutations, which got
845    /// through 1% of the time.
846    #[test]
847    fn test_constant_features_are_noise() {
848        let n = 60;
849        let mut g = SplitMix64::new(11);
850        let mut columns = vec![Column::new("id".into(), (0..n as i64).collect::<Vec<_>>())];
851        for j in 0..50 {
852            let values: Vec<f64> = (0..n)
853                .map(|i| (i % 3) as f64 + 0.3 * uniform(&mut g) + j as f64)
854                .collect();
855            columns.push(Column::new(format!("f{j}").into(), values));
856        }
857        columns.push(Column::new("constant".into(), vec![2.0; n]));
858        columns.push(Column::new("missing".into(), vec![f64::NAN; n]));
859        let df = DataFrame::new_infer_height(columns).unwrap();
860        let r = Reduction {
861            max_missing: None,
862            max_top_share: None,
863            max_correlation: None,
864            ..Default::default()
865        };
866        let fit = r.fit(&df, &keys()).unwrap();
867        let rep = &fit.report;
868        let dropped_by = strs(rep, "dropped_by");
869        let scores = rep.column("noise_score").unwrap().f64().unwrap();
870        for i in [50, 51] {
871            assert_eq!(dropped_by[i].as_deref(), Some("noise"), "row {i}");
872            assert!(scores.get(i).unwrap().is_nan(), "row {i}");
873        }
874    }

899        ]
900        .unwrap();

902            max_correlation: None,
