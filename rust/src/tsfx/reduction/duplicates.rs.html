1//! Step 3: near-duplicates. Spearman correlations between the features (from
2//! the rank matrix), then a greedy walk that keeps a feature unless it is too
3//! correlated with one kept before it.
4
5use ndarray::{Array2, s};
6use rayon::prelude::*;
7
8/// Columns per block in the correlation matrix's parallel product.
9const BLOCK: usize = 64;
10
11/// The correlation matrix `zᵀ z / n` of the columns of `z` (n rows). With
12/// standardized ranks in `z` (see `rank_matrix`) this is Spearman's
13/// correlation. Computed in parallel over blocks of 64 columns, each block of
14/// rows of the result a single-threaded matrix product, placed by index.
15pub(crate) fn correlation_matrix(z: &Array2<f64>) -> Array2<f64> {
16    let (n, d) = z.dim();
17    let starts: Vec<usize> = (0..d).step_by(BLOCK).collect();
18    let blocks: Vec<Array2<f64>> = starts
19        .par_iter()
20        .map(|&start| {
21            let end = (start + BLOCK).min(d);
22            z.slice(s![.., start..end]).t().dot(z)
23        })
24        .collect();
25    let mut corr = Array2::zeros((d, d));
26    for (&start, block) in starts.iter().zip(&blocks) {
27        corr.slice_mut(s![start..start + block.nrows(), ..])
28            .assign(block);
29    }
30    let n = n as f64;
31    corr.mapv_inplace(|x| x / n);
32    corr
33}
34
35/// A feature dropped as a near-duplicate of a kept one.
36#[derive(Clone, Debug, PartialEq)]
37pub(crate) struct Duplicate {
38    /// Index of the kept feature it copies.
39    pub of: usize,
40    /// Their signed correlation.
41    pub correlation: f64,
42}
43
44/// Walks the features in `order` and keeps each unless its |correlation| with
45/// an already kept feature is above `max_correlation`; then it is a
46/// [`Duplicate`] of the kept feature it correlates with most (the one kept
47/// first on ties). Indexed by feature; `None` means kept. `order` is the
48/// priority, a permutation of the feature indices: earlier features win
49/// (extraction order in stage 1).
50pub(crate) fn greedy_duplicates(
51    corr: &Array2<f64>,
52    order: &[usize],
53    max_correlation: f64,
54) -> Vec<Option<Duplicate>> {
55    let mut result: Vec<Option<Duplicate>> = vec![None; corr.nrows()];
56    let mut kept: Vec<usize> = Vec::new();
57    for &j in order {
58        let mut best: Option<Duplicate> = None;
59        for &k in &kept {
60            let r = corr[[j, k]];
61            if r.abs() > max_correlation
62                && best.as_ref().is_none_or(|b| r.abs() > b.correlation.abs())
63            {
64                best = Some(Duplicate {
65                    of: k,
66                    correlation: r,
67                });
68            }
69        }
70        match best {
71            Some(duplicate) => result[j] = Some(duplicate),
72            None => kept.push(j),
73        }
74    }
75    result
76}

85    #[test]
86    fn test_spearman_monotone_and_negative() {
87        let x = [1.0, 2.0, 3.0, 4.0];
88        let sq = [1.0, 4.0, 9.0, 16.0];
89        let rev = [4.0, 3.0, 2.0, 1.0];
90        let z = rank_matrix(&[&x, &sq, &rev], &[0, 1, 2, 3]);
91        let c = correlation_matrix(&z);
92        assert!((c[[0, 1]] - 1.0).abs() < 1e-12);
93        assert!((c[[0, 2]] + 1.0).abs() < 1e-12);
94        assert!((c[[1, 1]] - 1.0).abs() < 1e-12);
95    }

97    /// Blocks of 64 columns put together give the plain product, also for a
98    /// width that is not a multiple of 64.
99    #[test]
100    fn test_correlation_blocks_match_direct_product() {
101        let mut g = SplitMix64::new(5);
102        let (n, d) = (37, 130);
103        let z = Array2::from_shape_fn((n, d), |_| {
104            (g.next_u64() >> 11) as f64 / (1u64 << 53) as f64
105        });
106        let c = correlation_matrix(&z);
107        let direct = z.t().dot(&z) / n as f64;
108        assert_eq!(c.dim(), (d, d));
109        assert!(
110            c.iter()
111                .zip(direct.iter())
112                .all(|(a, b)| (a - b).abs() < 1e-12)
113        );
114    }

116    fn corr3(r01: f64, r02: f64, r12: f64) -> Array2<f64> {
117        array![[1.0, r01, r02], [r01, 1.0, r12], [r02, r12, 1.0]]
118    }
119
120    #[test]
121    fn test_greedy() {
122        let d = greedy_duplicates(&corr3(0.97, 0.5, 0.96), &[0, 1, 2], 0.95);
123        assert!(d[0].is_none() && d[2].is_none());
124        assert_eq!(d[1].as_ref().unwrap().of, 0);
125        // equal to the threshold is kept; negative correlation counts by |ρ|
126        assert!(greedy_duplicates(&corr3(0.95, 0.0, 0.0), &[0, 1, 2], 0.95)[1].is_none());
127        let neg = greedy_duplicates(&corr3(-0.97, 0.0, 0.0), &[0, 1, 2], 0.95);
128        assert_eq!(neg[1].as_ref().unwrap().correlation, -0.97);
129        // the order decides who is kept
130        let rev = greedy_duplicates(&corr3(0.97, 0.5, 0.96), &[2, 1, 0], 0.95);
131        assert!(rev[2].is_none() && rev[0].is_none());
132        assert_eq!(rev[1].as_ref().unwrap().of, 2);
133    }
134
135    #[test]
136    fn test_greedy_names_the_closest_kept_feature() {
137        // 2 copies both 0 (0.96) and 1 (0.99): it is a duplicate of 1
138        let d = greedy_duplicates(&corr3(0.1, 0.96, 0.99), &[0, 1, 2], 0.95);
139        assert_eq!(d[2].as_ref().unwrap().of, 1);
140        // a tie goes to the feature kept first
141        let tie = greedy_duplicates(&corr3(0.1, 0.97, -0.97), &[0, 1, 2], 0.95);
142        assert_eq!(tie[2].as_ref().unwrap().of, 0);
143    }
